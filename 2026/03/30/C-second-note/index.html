2026/3/30 C语言

C语言格式说明符（格式转义符）完整指南

格式说明符表格

格式符 数据类型 说明 示例代码
%d int 十进制有符号整数 printf("%d", 123);
%i int 十进制有符号整数（同%d） printf("%i", -456);
%u unsigned int 十进制无符号整数 printf("%u", 456);
%o unsigned int 八进制无符号整数 printf("%o", 8); // 输出10
%x unsigned int 十六进制无符号整数（小写） printf("%x", 255); // 输出ff
%X unsigned int 十六进制无符号整数（大写） printf("%X", 255); // 输出FF
%f float/double 十进制浮点数 printf("%f", 3.14159);
%lf double 十进制双精度浮点数 printf("%lf", 3.1415926535);
%e float/double 科学计数法（小写e） printf("%e", 123.45); // 1.234500e+02
%E float/double 科学计数法（大写E） printf("%E", 123.45); // 1.234500E+02
%g float/double 自动选择%f或%e printf("%g", 0.00012345);
%c char 单个字符 printf("%c", 'A');
%s char[] 字符串 printf("%s", "Hello");
%p 指针 指针地址 printf("%p", &变量);
%% - 输出百分号本身 printf("%%"); // 输出%

实际代码演示

1. 整数类型演示

#include <stdio.h>

int main() {
    int num = 255;
    unsigned int unum = 255;
  
    printf("=== 整数类型演示 ===\n");
    printf("十进制: %%d = %d\n", num);           // 输出: 255
    printf("八进制: %%o = %o\n", num);           // 输出: 377
    printf("十六进制: %%x = %x\n", num);         // 输出: ff
    printf("十六进制大写: %%X = %X\n", num);     // 输出: FF
    printf("无符号: %%u = %u\n", unum);          // 输出: 255
  
    return 0;
}

2.浮点数类型演示

#include <stdio.h>

int main() {
    float fnum = 3.14159f;
    double dnum = 3.14159265358979;
  
    printf("=== 浮点数类型演示 ===\n");
    printf("标准浮点数: %%f = %f\n", fnum);           // 输出: 3.141590
    printf("指定精度: %%.2f = %.2f\n", fnum);         // 输出: 3.14
    printf("双精度: %%lf = %lf\n", dnum);             // 输出: 3.141593
    printf("科学计数法: %%e = %e\n", fnum);           // 输出: 3.141590e+00
    printf("自动选择: %%g = %g\n", 0.00012345);       // 输出: 0.00012345
  
    return 0;
}

3.字符和字符串演示

#include <stdio.h>

int main() {
    char ch = 'A';
    char str[] = "Hello, World!";
  
    printf("=== 字符和字符串演示 ===\n");
    printf("单个字符: %%c = %c\n", ch);              // 输出: A
    printf("字符ASCII码: %%d = %d\n", ch);           // 输出: 65
    printf("字符串: %%s = %s\n", str);               // 输出: Hello, World!
    printf("字符串截取: %%.5s = %.5s\n", str);       // 输出: Hello
  
    return 0;
}
