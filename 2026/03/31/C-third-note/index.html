什么是数组

想象你要存储全班30个同学的成绩。难道要定义30个变量吗？

int score1, score2, score3, ... score30;  // 太麻烦了！

数组就是解决这个问题的！它就像一排连续的房间，每个房间都有一个编号（下标），可以存放相同类型的数据。

int scores[30];  // 一次性定义30个整型变量
// 下标从0开始：scores[0], scores[1], ..., scores[29]

一维数组

1.1 数组的定义

#include <stdio.h>

int main() {
    // 方式1：只定义不初始化
    int arr1[5];  // 定义了5个int类型的空间，里面的值是随机的
  
    // 方式2：定义时全部初始化为0
    int arr2[5] = {0};  // 所有元素都是0
  
    // 方式3：定义时初始化部分值
    int arr3[5] = {1, 2, 3};  // arr3[0]=1, arr3[1]=2, arr3[2]=3, 后面自动为0
  
    // 方式4：不指定大小，由初始值决定
    int arr4[] = {10, 20, 30, 40};  // 自动确定大小为4
  
    // 方式5：使用宏定义大小（推荐）
    #define SIZE 10
    int arr5[SIZE];
  
    return 0;
}

1.2 数组的访问和赋值


#include <stdio.h>

int main() {
    int scores[5];
  
    // 给数组元素赋值
    scores[0] = 85;
    scores[1] = 92;
    scores[2] = 78;
    scores[3] = 88;
    scores[4] = 95;
  
    // 访问数组元素
    printf("第一个学生的成绩：%d\n", scores[0]);  // 输出85
    printf("第三个学生的成绩：%d\n", scores[2]);  // 输出78
  
    // 修改数组元素
    scores[2] = 80;  // 修改第三个学生的成绩
    printf("修改后的成绩：%d\n", scores[2]);  // 输出80
  
    return 0;
}

1.3 使用循环遍历数组（最重要！）

#include <stdio.h>

int main() {
    int scores[5] = {85, 92, 78, 88, 95};
    
    // 遍历数组并打印
    printf("所有学生的成绩：\n");
    for(int i = 0; i < 5; i++) {
        printf("第%d个学生的成绩：%d\n", i + 1, scores[i]);
    }
    
    // 计算平均分
    int sum = 0;
    for(int i = 0; i < 5; i++) {
        sum += scores[i];
    }
    float average = sum / 5.0;
    printf("平均分：%.2f\n", average);
    
    return 0;
}

#include <stdio.h>

int main() {

}

二维数组

二维数组可以想象成一张表格，有行和列。

2.1 二维数组的定义

#include <stdio.h>

int main() {
    // 定义一个3行4列的二维数组
    // 就像一个3行4列的表格
    int matrix[3][4];
    
    // 定义并初始化
    int scores[3][4] = {
        {85, 92, 78, 88},  // 第1行（0号学生）
        {90, 87, 93, 89},  // 第2行（1号学生）
        {76, 82, 79, 91}   // 第3行（2号学生）
    };
    
    // 也可以这样初始化（不推荐，可读性差）
    int arr[2][3] = {1, 2, 3, 4, 5, 6};
    
    // 部分初始化
    int partial[2][3] = {
        {1, 2},     // 后面自动补0：{1, 2, 0}
        {4}         // {4, 0, 0}
    };
    
    return 0;
}

2.2 二维数组的访问

#include <stdio.h>

int main() {
    int scores[3][4] = {
        {85, 92, 78, 88},  // 学生0的4科成绩
        {90, 87, 93, 89},  // 学生1的4科成绩
        {76, 82, 79, 91}   // 学生2的4科成绩
    };
    
    // 访问单个元素
    printf("学生0的语文成绩：%d\n", scores[0][0]);  // 85
    printf("学生1的数学成绩：%d\n", scores[1][1]);  // 87
    printf("学生2的英语成绩：%d\n", scores[2][2]);  // 79
    
    // 修改元素
    scores[2][1] = 88;  // 修改学生2的数学成绩为88
    printf("修改后的数学成绩：%d\n", scores[2][1]);
    
    return 0;
}

2.3 二维数组的遍历（双重循环）

#include <stdio.h>

int main() {
    int scores[3][4] = {
        {85, 92, 78, 88},
        {90, 87, 93, 89},
        {76, 82, 79, 91}
    };
    
    printf("成绩表：\n");
    printf("    语文 数学 英语 科学\n");
    
    // 双重循环遍历二维数组
    for(int i = 0; i < 3; i++) {      // 外层循环：行（学生）
        printf("学生%d ", i);
        for(int j = 0; j < 4; j++) {  // 内层循环：列（科目）
            printf("%3d ", scores[i][j]);
        }
        printf("\n");
    }
    
    // 计算每个学生的总分
    printf("\n学生总分：\n");
    for(int i = 0; i < 3; i++) {
        int sum = 0;
        for(int j = 0; j < 4; j++) {
            sum += scores[i][j];
        }
        printf("学生%d的总分：%d\n", i, sum);
    }
    
    return 0;
}

实战场景应用

场景1：学生成绩管理系统

#include <stdio.h>

#define STUDENT_NUM 5  // 学生人数
#define SUBJECT_NUM 3  // 科目数

int main() {
    // 存储学生成绩的二维数组
    // 行：学生，列：科目（0语文，1数学，2英语）
    int scores[STUDENT_NUM][SUBJECT_NUM];
    
    // 1. 输入成绩
    printf("========== 学生成绩录入系统 ==========\n");
    for(int i = 0; i < STUDENT_NUM; i++) {
        printf("\n请输入第%d个学生的成绩：\n", i + 1);
        printf("语文：");
        scanf("%d", &scores[i][0]);
        printf("数学：");
        scanf("%d", &scores[i][1]);
        printf("英语：");
        scanf("%d", &scores[i][2]);
    }
    
    // 2. 显示所有成绩
    printf("\n========== 成绩表 ==========\n");
    printf("学号\t语文\t数学\t英语\t总分\n");
    printf("--------------------------------\n");
    
    for(int i = 0; i < STUDENT_NUM; i++) {
        int sum = 0;
        printf("%d\t", i + 1);
        for(int j = 0; j < SUBJECT_NUM; j++) {
            printf("%d\t", scores[i][j]);
            sum += scores[i][j];
        }
        printf("%d\n", sum);
    }
    
    // 3. 计算各科平均分
    printf("\n========== 各科平均分 ==========\n");
    char* subjects[] = {"语文", "数学", "英语"};
    
    for(int j = 0; j < SUBJECT_NUM; j++) {
        int sum = 0;
        for(int i = 0; i < STUDENT_NUM; i++) {
            sum += scores[i][j];
        }
        float avg = sum / (float)STUDENT_NUM;
        printf("%s平均分：%.2f\n", subjects[j], avg);
    }
    
    // 4. 找出最高分和最低分
    int max_score = scores[0][0], min_score = scores[0][0];
    int max_stu = 0, max_sub = 0, min_stu = 0, min_sub = 0;
    
    for(int i = 0; i < STUDENT_NUM; i++) {
        for(int j = 0; j < SUBJECT_NUM; j++) {
            if(scores[i][j] > max_score) {
                max_score = scores[i][j];
                max_stu = i;
                max_sub = j;
            }
            if(scores[i][j] < min_score) {
                min_score = scores[i][j];
                min_stu = i;
                min_sub = j;
            }
        }
    }
    
    printf("\n最高分：%d分 (学生%d，%s)\n", 
           max_score, max_stu + 1, subjects[max_sub]);
    printf("最低分：%d分 (学生%d，%s)\n", 
           min_score, min_stu + 1, subjects[min_sub]);
    
    return 0;
}

场景2：矩阵运算

#include <stdio.h>

#define ROW 3
#define COL 3

int main() {
    int matrix1[ROW][COL] = {
        {1, 2, 3},
        {4, 5, 6},
        {7, 8, 9}
    };
    
    int matrix2[ROW][COL] = {
        {9, 8, 7},
        {6, 5, 4},
        {3, 2, 1}
    };
    
    int result[ROW][COL];  // 存储运算结果
    
    // 矩阵相加
    printf("========== 矩阵相加 ==========\n");
    for(int i = 0; i < ROW; i++) {
        for(int j = 0; j < COL; j++) {
            result[i][j] = matrix1[i][j] + matrix2[i][j];
            printf("%3d ", result[i][j]);
        }
        printf("\n");
    }
    
    // 矩阵转置
    printf("\n========== 矩阵转置 ==========\n");
    printf("原矩阵：\n");
    for(int i = 0; i < ROW; i++) {
        for(int j = 0; j < COL; j++) {
            printf("%3d ", matrix1[i][j]);
        }
        printf("\n");
    }
    
    printf("转置后：\n");
    for(int i = 0; i < COL; i++) {
        for(int j = 0; j < ROW; j++) {
            printf("%3d ", matrix1[j][i]);
        }
        printf("\n");
    }
    
    return 0;
}

场景3：字符数组（字符串）

#include <stdio.h>
#include <string.h>

int main() {
    // 一维字符数组 = 字符串
    char name[20] = "张三";
    char greeting[50];
    
    // 字符串操作
    strcpy(greeting, "你好，");  // 复制字符串
    strcat(greeting, name);      // 连接字符串
    printf("%s\n", greeting);
    
    // 二维字符数组 = 多个字符串
    char students[3][20] = {
        "张三",
        "李四",
        "王五"
    };
    
    printf("\n学生名单：\n");
    for(int i = 0; i < 3; i++) {
        printf("第%d个学生：%s\n", i + 1, students[i]);
    }
    
    // 从键盘输入多个字符串
    char names[5][30];
    printf("\n请输入5个名字：\n");
    for(int i = 0; i < 5; i++) {
        printf("第%d个：", i + 1);
        scanf("%s", names[i]);  // 注意：字符串不需要&
    }
    
    printf("\n你输入的名字：\n");
    for(int i = 0; i < 5; i++) {
        printf("%s\n", names[i]);
    }
    
    return 0;
}

场景4：数据统计与分析

#include <stdio.h>

#define DAYS 7  // 一周7天

int main() {
    // 存储一周的温度
    int temperatures[DAYS];
    
    // 输入温度
    printf("请输入一周7天的温度：\n");
    for(int i = 0; i < DAYS; i++) {
        printf("第%d天：", i + 1);
        scanf("%d", &temperatures[i]);
    }
    
    // 统计信息
    int sum = 0;
    int max_temp = temperatures[0];
    int min_temp = temperatures[0];
    int max_day = 0, min_day = 0;
    
    for(int i = 0; i < DAYS; i++) {
        sum += temperatures[i];
        
        if(temperatures[i] > max_temp) {
            max_temp = temperatures[i];
            max_day = i;
        }
        
        if(temperatures[i] < min_temp) {
            min_temp = temperatures[i];
            min_day = i;
        }
    }
    
    float avg = sum / (float)DAYS;
    
    printf("\n========== 温度统计 ==========\n");
    printf("最高温度：%d℃ (第%d天)\n", max_temp, max_day + 1);
    printf("最低温度：%d℃ (第%d天)\n", min_temp, min_day + 1);
    printf("平均温度：%.1f℃\n", avg);
    
    // 找出高于平均温度的天数
    int above_avg = 0;
    printf("\n高于平均温度的天数：");
    for(int i = 0; i < DAYS; i++) {
        if(temperatures[i] > avg) {
            above_avg++;
            printf(" %d", i + 1);
        }
    }
    printf("\n共%d天\n", above_avg);
    
    return 0;
}
