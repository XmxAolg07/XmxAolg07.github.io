C语言结构体学习笔记

什么是结构体

结构体可以将不同类型的数据组合在一起，形成一个新的数据类型。

#include <stdio.h>
#include <string.h>

// 定义结构体
struct Student {
    int id;
    char name[20];
    int age;
    float score;
};

int main() {
    // 声明结构体变量
    struct Student stu1;
    
    // 赋值
    stu1.id = 1001;
    strcpy(stu1.name, "张三");
    stu1.age = 20;
    stu1.score = 88.5;
    
    // 输出
    printf("学号：%d\n", stu1.id);
    printf("姓名：%s\n", stu1.name);
    printf("年龄：%d\n", stu1.age);
    printf("成绩：%.1f\n", stu1.score);
    
    return 0;
}

结构体的初始化

#include <stdio.h>

struct Book {
    char title[50];
    char author[20];
    float price;
};

int main() {
    // 方式1：按顺序初始化
    struct Book book1 = {"C语言程序设计", "谭浩强", 49.8};
    
    // 方式2：指定成员初始化（C99）
    struct Book book2 = {.title = "Python入门", .price = 59.9};
    
    // 方式3：先定义后赋值
    struct Book book3;
    strcpy(book3.title, "数据结构");
    strcpy(book3.author, "严蔚敏");
    book3.price = 68.0;
    
    printf("%s - %s - %.1f\n", book1.title, book1.author, book1.price);
    printf("%s - %.1f\n", book2.title, book2.price);
    printf("%s - %s - %.1f\n", book3.title, book3.author, book3.price);
    
    return 0;
}

typedef 重命名

#include <stdio.h>

// 使用typedef简化结构体名称
typedef struct {
    int x;
    int y;
} Point;

typedef struct Student {
    int id;
    char name[20];
} Stu;

int main() {
    Point p1 = {10, 20};      // 不用写struct
    Stu s1 = {1001, "李四"};
    
    printf("坐标：(%d, %d)\n", p1.x, p1.y);
    printf("学生：%d - %s\n", s1.id, s1.name);
    
    return 0;
}

结构体数组

#include <stdio.h>

typedef struct {
    char name[20];
    int chinese;
    int math;
    int english;
} Student;

int main() {
    // 结构体数组
    Student class[3] = {
        {"张三", 85, 92, 78},
        {"李四", 90, 88, 95},
        {"王五", 78, 85, 82}
    };
    
    // 遍历输出
    for(int i = 0; i < 3; i++) {
        int total = class[i].chinese + class[i].math + class[i].english;
        printf("%s：语文%d 数学%d 英语%d 总分%d\n", 
               class[i].name, class[i].chinese, 
               class[i].math, class[i].english, total);
    }
    
    return 0;
}

结构体指针

#include <stdio.h>

typedef struct {
    char name[20];
    int age;
} Person;

int main() {
    Person p1 = {"小明", 18};
    Person *ptr = &p1;
    
    // 两种访问方式
    printf("姓名：%s，年龄：%d\n", p1.name, p1.age);
    printf("姓名：%s，年龄：%d\n", ptr->name, ptr->age);  // 箭头运算符
    
    // 通过指针修改
    ptr->age = 19;
    printf("修改后年龄：%d\n", p1.age);
    
    return 0;
}

结构体作为函数参数

#include <stdio.h>

typedef struct {
    int width;
    int height;
} Rectangle;

// 值传递（会复制整个结构体）
int getArea(Rectangle r) {
    return r.width * r.height;
}

// 指针传递（效率高，可修改原数据）
void enlarge(Rectangle *r, int factor) {
    r->width *= factor;
    r->height *= factor;
}

int main() {
    Rectangle rect = {10, 20};
    
    int area = getArea(rect);
    printf("面积：%d\n", area);
    
    enlarge(&rect, 2);
    printf("放大后：宽%d 高%d 面积%d\n", 
           rect.width, rect.height, getArea(rect));
    
    return 0;
}

结构体嵌套

#include <stdio.h>

typedef struct {
    int year;
    int month;
    int day;
} Date;

typedef struct {
    char name[20];
    Date birthday;  // 嵌套结构体
    float height;
} Student;

int main() {
    Student stu = {"小红", {2005, 5, 15}, 165.5};
    
    printf("姓名：%s\n", stu.name);
    printf("生日：%d年%d月%d日\n", 
           stu.birthday.year, stu.birthday.month, stu.birthday.day);
    printf("身高：%.1f\n", stu.height);
    
    return 0;
}

常见使用场景

#include <stdio.h>

// 场景1：点坐标
typedef struct {
    double x;
    double y;
} Point;

// 场景2：商品信息
typedef struct {
    int id;
    char name[50];
    double price;
    int stock;
} Product;

// 场景3：日期时间
typedef struct {
    int year;
    int month;
    int day;
    int hour;
    int minute;
    int second;
} DateTime;

int main() {
    Point p = {3.5, 7.2};
    Product book = {1001, "C语言教程", 49.9, 100};
    DateTime now = {2024, 3, 20, 14, 30, 0};
    
    printf("坐标：(%.1f, %.1f)\n", p.x, p.y);
    printf("商品：%s 价格：%.1f 库存：%d\n", 
           book.name, book.price, book.stock);
    printf("时间：%d-%02d-%02d %02d:%02d:%02d\n",
           now.year, now.month, now.day,
           now.hour, now.minute, now.second);
    
    return 0;
}
