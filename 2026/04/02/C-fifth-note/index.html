C语言指针学习笔记

什么是指针

指针就是存放变量地址的变量。

#include <stdio.h>

int main() {
    int num = 10;      // 普通变量
    int *p = #     // 指针变量，存放num的地址
  
    printf("num的值：%d\n", num);      // 10
    printf("num的地址：%p\n", &num);   // 地址值
    printf("p的值：%p\n", p);          // 和上面一样
    printf("p指向的值：%d\n", *p);     // 10（解引用）
  
    return 0;
}

指针的基本操作

#include <stdio.h>

int main() {
    int a = 5, b = 10;
    int *ptr = &a;
  
    // 通过指针修改变量
    *ptr = 20;
    printf("a = %d\n", a);  // 20
  
    // 改变指针指向
    ptr = &b;
    *ptr = 99;
    printf("b = %d\n", b);  // 99
  
    return 0;
}

指针与数组

#include <stdio.h>

int main() {
    int arr[5] = {10, 20, 30, 40, 50};
    int *p = arr;  // 数组名就是首地址
  
    // 通过指针访问数组
    for(int i = 0; i < 5; i++) {
        printf("%d ", *(p + i));  // 10 20 30 40 50
    }
  
    printf("\n");
  
    // 指针移动
    printf("*p = %d\n", *p);       // 10
    p++;
    printf("*p = %d\n", *p);       // 20
  
    return 0;
}

指针作为函数参数

#include <stdio.h>

// 交换两个数（值传递不行，必须用指针）
void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

int main() {
    int x = 5, y = 10;
    printf("交换前：x=%d, y=%d\n", x, y);
  
    swap(&x, &y);
    printf("交换后：x=%d, y=%d\n", x, y);
  
    return 0;
}

指针与字符串

#include <stdio.h>

int main() {
    char str[] = "hello";
    char *p = str;
  
    // 用指针遍历字符串
    while(*p != '\0') {
        printf("%c ", *p);
        p++;
    }
  
    return 0;
}

指针数组

#include <stdio.h>

int main() {
    int arr1[] = {1, 2, 3};
    int arr2[] = {4, 5, 6};
    int arr3[] = {7, 8, 9};
  
    int *parr[3] = {arr1, arr2, arr3};  // 指针数组
  
    // 访问第一个数组的第二个元素
    printf("%d\n", parr[0][1]);  // 2
  
    // 打印所有
    for(int i = 0; i < 3; i++) {
        for(int j = 0; j < 3; j++) {
            printf("%d ", parr[i][j]);
        }
        printf("\n");
    }
  
    return 0;
}

动态内存分配

#include <stdio.h>
#include <stdlib.h>

int main() {
    int *p = (int*)malloc(5 * sizeof(int));  // 分配空间
  
    if(p == NULL) {
        printf("分配失败\n");
        return -1;
    }
  
    // 使用指针
    for(int i = 0; i < 5; i++) {
        p[i] = i * 10;
        printf("%d ", p[i]);
    }
  
    free(p);  // 释放内存
    p = NULL; // 避免野指针
  
    return 0;
}

常见指针用法速查

写法 含义
int *p 声明一个int指针
p = &a p指向a的地址
*p 取p指向的值
p++ 指针移动到下一个位置
p[0] 等价于*p
arr 数组名就是指针
NULL 空指针
