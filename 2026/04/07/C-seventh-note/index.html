题目：有 1、2、3、4 四个数字，能组成多少个互不相同且无重复数字的三位数？都是多少？

解答：

#include <stdio.h>

int main(void)
{
    //定义三个变量作为百位十位个位
    int i,j,k;
    int total =0;
    //利用for循环把所有不重复数字可能打印一遍
    for(int i=1;i<5;i++)
    {
        for (int j = 1;j<5; j++)
        {
            for(int k =1;k<5;k++)
            {
                if(i!=j&&j!=k&&k!=i)
                {
                    printf("%d%d%d\n",i,j,k);
                    //每成功一次加一
                    total++;
                }
            }
        }  
    }
    //打印符合组合数
    printf("%d",total);
    return 0;
}

题目：企业发放的奖金根据利润提成。

利润(I)低于或等于10万元时，奖金可提10%；

利润高于10万元，低于20万元时，低于10万元的部分按10%提成，高于10万元的部分，可提成7.5%；

20万到40万之间时，高于20万元的部分，可提成5%；

40万到60万之间时高于40万元的部分，可提成3%；

60万到100万之间时，高于60万元的部分，可提成1.5%；

高于100万元时，超过100万元的部分按1%提成。

解答：

#include <stdio.h>

int main(void)
{   
    //定义一个双精度浮点数
    double i;
    printf("Please input your number");
    scanf("%lf",&i);

    //利用if语句计算奖励的值
    double bonus;
    if(i<=100000&&i>0)
    {
        bonus = i * 0.1;
    }
    else if (i>100000&&i<=200000)
    {
        bonus = (i-100000)*0.075+10000;
    }
    else if (i>=200000&&i<400000)
    {
        bonus =(i-200000)*0.05+17500;
    }
    else if (i>=400000&&i<600000)
    {
        bonus =(i-400000)*0.03+27500;
    }
    else if (i>=600000&&i<1000000)
    {
        bonus =(i-600000)*0.015+33500;
    }
    else if (i>=1000000)
    {
        bonus =(i-1000000)*0.01+39500;
    }
    else
    {  
        printf("Please input illgal number");
    }
    //打印奖励的提成
    printf("Bonus:%.2lf",bonus);
  
    return 0;
}

一个整数，它加上100后是一个完全平方数，再加上168又是一个完全平方数，请问该数是多少？

解答：

此题可以调用math函数也可以不调用

调用math函数

#include <stdio.h>
#include <math.h>   // 使用 sqrt 函数

int main(void)
{
    int x = 0;

    // 暴力枚举
    for (x = 0; x < 10000; x++)
    {
        int a = x + 100;
        int b = x + 100 + 168;

        // 判断是否是完全平方数
        int sqrt_a = (int)sqrt(a);
        int sqrt_b = (int)sqrt(b);

        if (sqrt_a * sqrt_a == a && sqrt_b * sqrt_b == b)
        {
            printf("%d\n", x);
            printf("%d + 100 = %d = %d^2\n", x, a, sqrt_a);
            printf("%d + 268 = %d = %d^2\n", x, b, sqrt_b);
            return 0;
        }
    }
    return 0;
}

不调用math函数
分析此题

  1. 则：x + 100 = n2, x + 100 + 168 = m2
  2. 计算等式：m2 - n2 = (m + n)(m - n) = 168
  3. 设置： m + n = i，m - n = j，i * j =168，i 和 j 至少一个是偶数
  4. 可得： m = (i + j) / 2， n = (i - j) / 2，i 和 j 要么都是偶数，要么都是奇数。
  5. 从 3 和 4 推导可知道，i 与 j 均是大于等于 2 的偶数。
  6. 由于 i * j = 168， j>=2，则 1 < i < 168 / 2 + 1。
  7. 接下来将 i 的所有数字循环计算即可。
#include <stdio.h>
 
int main (void)
{
    int  i, j, m, n, x;
     // a * b = 168，a 和 b 同奇偶，且 a < b
    for (i = 1; i < 168 / 2 + 1; i++)
    {
        if (168 % i == 0)
        {
            j = 168 / i;
            if ( i > j && (i + j) % 2 == 0 && (i - j) % 2 == 0)
            {
                 // 同奇偶判断
                m = (i + j) / 2;
                n = (i - j) / 2;
                // x = m*m - 100
                x = n * n - 100;
                printf ("%d + 100 = %d * %d\n", x, n, n);
                printf ("%d + 268 = %d * %d\n", x, m, m);
            }
        }
    }
    return 0;
}

输入某年某月某日，判断这一天是这一年的第几天？

解答：

#include <stdio.h>
//创建一个判断是否为闰年的函数
int leapyear(int year)
{
    if((year % 4 == 0 && year % 100 != 0) || (year % 400 == 0))
    {
        return 1;
    }
    return 0;
}

int main(void)
{
    //定义并输入年月日
    int year, month, day;
    //建立一个12月对应天数的数组
    int daysinmonth[12]={31,28,31,30,31,30,31,31,30,31,30,31};
    int total = 0;

    printf("Please input years month and days\n");
    scanf_s("%d %d %d",&year,&month,&day);
    if(leapyear(year))
    {
        //判断是否为闰年如果是就把2月变成29天
        daysinmonth[1]=29;
    }
    //计算天数
    for(int i=0;i<month;i++)
    {
        total += daysinmonth[i];
    }

    total += day;
    //打印天数
    printf("This %d days\n",total);

    return 0;
}

输入三个整数 x、y、z，请把这三个数由小到大输出。

解答

#include <stdio.h>

int main(void)
{
    int x,y,z,j;
    //输入三个数字
    printf("Please input x y z");
    scanf("%d %d %d",&x,&y,&z);
    //交换x y的值
    if(x>y)
    {
        j=y;
        y=x;
        x=j;
    }
    //交换x z的值
    if(x>z)
    {
        j=z;
        z=x;
        x=j;
    }
    //交换y z的值
    if (y>z)
    {
        j=z;
        z=y;
        y=j;
    }
    //打印排序后的
    printf("%d %d %d",x,y,z);
    return 0;
}
