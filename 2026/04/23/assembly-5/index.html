汇编笔记5

条件标志位

跳转指令根据标志位决定是否跳转

标志位 含义
ZF 零标志（结果为0时置1）
CF 进位标志
SF 符号标志
OF 溢出标志

常用跳转指令

指令 条件 含义
jmp 无条件 无条件跳转
je 或 jz ZF = 1 相等/为零则跳转
jne 或 jnz ZF = 0 不相等/不为零则跳转
jg 或 jnle ZF=0 且 SF=OF 有符号大于
jge 或 jnl SF=OF 有符号大于等于
jl 或 jnge SF≠OF 有符号小于
jle 或 jng ZF=1 或 SF≠OF 有符号小于等于
ja 或 jnbe CF=0 且 ZF=0 无符号大于
jae 或 jnb CF=0 无符号大于等于
jb 或 jnae CF=1 无符号小于
jbe 或 jna CF=1 或 ZF=1 无符号小于等于

jmp无条件跳转

.intel_syntax noprefix
.global main
.section .text

main:
    push rbp
    mov rbp, rsp
  
    # 循环计数
    mov ecx, 0          # ecx = 0
  
start_loop:
    cmp ecx, 5          # 比较 ecx 和 5
    jge end_loop        # 如果 ecx >= 5，跳过循环
  
    # 循环体可以在这里添加代码
    # ...
  
    inc ecx             # ecx++
    jmp start_loop      # 无条件跳回循环开始
  
end_loop:
    # 程序结束
    mov eax, 0
    pop rbp
    ret

je / jz 条件跳转（相等则跳转）

.intel_syntax noprefix
.global main
.section .text

main:
    push rbp
    mov rbp, rsp
    sub rsp, 16
  
    # 局部变量
    mov dword ptr [rbp-4], 10    # a = 10
    mov dword ptr [rbp-8], 10    # b = 10
    mov dword ptr [rbp-12], 0    # result = 0
  
    # 比较 a 和 b
    mov eax, dword ptr [rbp-4]    # eax = a
    cmp eax, dword ptr [rbp-8]    # 比较 a 和 b
    jne not_equal                  # 不相等则跳转
  
    # 相等的情况
    mov dword ptr [rbp-12], 1     # result = 1
    jmp done                       # 跳过else部分
  
not_equal:
    mov dword ptr [rbp-12], 0     # result = 0
  
done:
    # 打印结果
    # printf("result: %d\n", result)
    lea rdi, [rip+fmt]            # 格式化字符串
    mov esi, dword ptr [rbp-12]   # 结果值
    xor eax, eax
    call printf
  
    mov eax, 0
    leave
    ret

.section .rodata
fmt:
    .string "result: %d\n"

简单循环（while循环）

.intel_syntax noprefix
.global main
.section .text

main:
    push rbp
    mov rbp, rsp
  
    xor ecx, ecx          # ecx = 0（计数器）
    xor eax, eax          # eax = 0（累加器）
  
while_loop:
    cmp ecx, 10           # while (ecx < 10)
    jge end_while         # ecx >= 10 则退出
  
    add eax, ecx          # sum += ecx
    inc ecx               # ecx++
    jmp while_loop        # 继续循环
  
end_while:
    # 打印结果（eax中是1+2+...+9=45）
    lea rdi, [rip+fmt]
    mov esi, eax
    xor eax, eax
    call printf
  
    mov eax, 0
    pop rbp
    ret

.section .rodata
fmt:
    .string "Sum = %d\n"

for循环（1到10求和）

.intel_syntax noprefix
.global main
.section .text

main:
    push rbp
    mov rbp, rsp
    sub rsp, 16
  
    # 局部变量
    mov dword ptr [rbp-4], 0      # sum = 0
    mov dword ptr [rbp-8], 0      # i = 0
  
    # for (i = 1; i <= 10; i++)
    mov dword ptr [rbp-8], 1      # i = 1
  
for_loop:
    cmp dword ptr [rbp-8], 10     # i <= 10 ?
    jg end_for                     # i > 10 则退出
  
    # sum += i
    mov eax, dword ptr [rbp-4]    # eax = sum
    add eax, dword ptr [rbp-8]    # eax = sum + i
    mov dword ptr [rbp-4], eax    # sum = eax
  
    # i++
    inc dword ptr [rbp-8]
    jmp for_loop
  
end_for:
    # 打印结果 sum = 55 (1+...+10)
    lea rdi, [rip+fmt]
    mov esi, dword ptr [rbp-4]
    xor eax, eax
    call printf
  
    mov eax, 0
    leave
    ret

.section .rodata
fmt:
    .string "Sum 1-10 = %d\n"

do-while循环（先执行后判断）

.intel_syntax noprefix
.global main
.section .text

main:
    push rbp
    mov rbp, rsp
  
    xor ecx, ecx          # ecx = 0
    xor eax, eax          # eax = 0
  
do_while_start:
    add eax, ecx          # sum += ecx
    inc ecx               # ecx++
    cmp ecx, 10           # while (ecx < 10)
    jl do_while_start     # ecx < 10 则继续循环
  
    # eax = 0+1+2+...+9 = 45
    lea rdi, [rip+fmt]
    mov esi, eax
    xor eax, eax
    call printf
  
    mov eax, 0
    pop rbp
    ret

.section .rodata
fmt:
    .string "Sum = %d\n"

多重条件判断（if-else if-else）

.intel_syntax noprefix
.global main
.section .text

main:
    push rbp
    mov rbp, rsp
    sub rsp, 16
  
    mov dword ptr [rbp-4], 75    # score = 75
    mov dword ptr [rbp-8], 0     # grade = 0
  
    # if (score >= 90)
    cmp dword ptr [rbp-4], 90
    jl check_80
    mov dword ptr [rbp-8], 'A'
    jmp print_grade
  
check_80:
    # else if (score >= 80)
    cmp dword ptr [rbp-4], 80
    jl check_70
    mov dword ptr [rbp-8], 'B'
    jmp print_grade
  
check_70:
    # else if (score >= 70)
    cmp dword ptr [rbp-4], 70
    jl check_60
    mov dword ptr [rbp-8], 'C'
    jmp print_grade
  
check_60:
    # else if (score >= 60)
    cmp dword ptr [rbp-4], 60
    jl else_branch
    mov dword ptr [rbp-8], 'D'
    jmp print_grade
  
else_branch:
    mov dword ptr [rbp-8], 'F'
  
print_grade:
    lea rdi, [rip+fmt]
    mov esi, dword ptr [rbp-8]
    xor eax, eax
    call printf
  
    mov eax, 0
    leave
    ret

.section .rodata
fmt:
    .string "Grade: %c\n"

使用loop指令（ECX作为循环计数器）

.intel_syntax noprefix
.global main
.section .text

# loop指令：执行 ecx--，如果ecx != 0则跳转

main:
    push rbp
    mov rbp, rsp
  
    mov ecx, 10           # 循环10次
    xor eax, eax          # sum = 0
  
loop_start:
    add eax, ecx          # sum += ecx (10+9+...+1)
    loop loop_start       # ecx--, if ecx != 0 goto loop_start
  
    # eax = 55
    lea rdi, [rip+fmt]
    mov esi, eax
    xor eax, eax
    call printf
  
    mov eax, 0
    pop rbp
    ret

.section .rodata
fmt:
    .string "Sum with LOOP: %d\n"
