什么是函数

函数是一段可以重复使用的代码块，用来完成特定的任务。

#include <stdio.h>

// 一个简单的函数：打印欢迎信息
void printWelcome() {
    printf("欢迎来到C语言世界！\n");
}

int main() {
    printWelcome();  // 调用函数
    printf("正在学习函数...\n");
    printWelcome();  // 再次调用
    return 0;
}

不同返回类型的函数

#include <stdio.h>

// 1. 无返回值，无参数
void sayHello() {
    printf("你好！\n");
}

// 2. 无返回值，有参数
void printMax(int a, int b) {
    if (a > b) {
        printf("较大值是：%d\n", a);
    } else {
        printf("较大值是：%d\n", b);
    }
}

// 3. 有返回值，无参数
int getCurrentYear() {
    return 2024;
}

// 4. 有返回值，有参数
int multiply(int x, int y) {
    return x * y;
}

// 5. 返回浮点数
float getAverage(int a, int b) {
    return (a + b) / 2.0;
}

int main() {
    sayHello();
    printMax(15, 8);
  
    int year = getCurrentYear();
    printf("当前年份：%d\n", year);
  
    int product = multiply(6, 7);
    printf("6 × 7 = %d\n", product);
  
    float avg = getAverage(85, 92);
    printf("平均分：%.2f\n", avg);
  
    return 0;
}

函数的参数

值传递

#include <stdio.h>

// 值传递：函数内部修改参数不会影响外部变量
void changeValue(int num) {
    printf("函数内部修改前：num = %d\n", num);
    num = 100;  // 只修改了副本
    printf("函数内部修改后：num = %d\n", num);
}

int main() {
    int a = 10;
    printf("调用函数前：a = %d\n", a);
    changeValue(a);
    printf("调用函数后：a = %d\n", a);  // a还是10，没有改变
    return 0;
}

地址传递（指针）

#include <stdio.h>

// 地址传递：可以修改外部变量
void changeByPointer(int *ptr) {
    printf("函数内部修改前：*ptr = %d\n", *ptr);
    *ptr = 100;  // 修改指针指向的值
    printf("函数内部修改后：*ptr = %d\n", *ptr);
}

int main() {
    int a = 10;
    printf("调用函数前：a = %d\n", a);
    changeByPointer(&a);  // 传递地址
    printf("调用函数后：a = %d\n", a);  // a变成了100
    return 0;
}

数组作为参数

#include <stdio.h>

// 计算数组元素之和
int sumArray(int arr[], int size) {
    int sum = 0;
    for (int i = 0; i < size; i++) {
        sum += arr[i];
    }
    return sum;
}

// 修改数组元素（地址传递）
void multiplyBy2(int arr[], int size) {
    for (int i = 0; i < size; i++) {
        arr[i] *= 2;
    }
}

// 打印数组
void printArray(int arr[], int size) {
    printf("[");
    for (int i = 0; i < size; i++) {
        printf("%d", arr[i]);
        if (i < size - 1) {
            printf(", ");
        }
    }
    printf("]\n");
}

int main() {
    int scores[] = {85, 90, 78, 92, 88};
    int size = sizeof(scores) / sizeof(scores[0]);
  
    printf("原数组：");
    printArray(scores, size);
  
    int total = sumArray(scores, size);
    printf("总分：%d\n", total);
  
    multiplyBy2(scores, size);
    printf("乘以2后：");
    printArray(scores, size);
  
    return 0;
}

函数的返回值

#include <stdio.h>

// 返回整数的函数
int getMax(int a, int b) {
    return (a > b) ? a : b;

// 返回浮点数
float getCircleArea(float radius) {
    return 3.14159 * radius * radius;

// 返回字符
char getGrade(int score) {
    if (score >= 90) return 'A';
    if (score >= 80) return 'B';

递归函数

阶乘计算

#include <stdio.h>

// 递归计算阶乘
int factorial(int n) {
    if (n <= 1) {  // 递归终止条件
        return 1;
    }
    return n * factorial(n - 1);  // 递归调用
}

// 非递归方式计算阶乘
int factorialLoop(int n) {
    int result = 1;
    for (int i = 1; i <= n; i++) {
        result *= i;
    }
    return result;
}

int main() {
    printf("5的阶乘（递归）：%d\n", factorial(5));
    printf("5的阶乘（循环）：%d\n", factorialLoop(5));
  
    printf("7的阶乘：%d\n", factorial(7));
    printf("10的阶乘：%d\n", factorial(10));
  
    return 0;
}

斐波那契数列

#include <stdio.h>

// 递归计算斐波那契数列
int fibonacci(int n) {
    if (n <= 1) {
        return n;
    }
    return fibonacci(n - 1) + fibonacci(n - 2);
}

// 打印斐波那契数列前n项
void printFibonacci(int n) {
    printf("斐波那契数列前 %d 项：\n", n);
    for (int i = 0; i < n; i++) {
        printf("%d ", fibonacci(i));
    }
    printf("\n");
}

int main() {
    printFibonacci(10);
    printf("第8项的值：%d\n", fibonacci(8));
    return 0;
}

实际应用场景

学生成绩管理系统

#include <stdio.h>

#define STUDENT_NUM 5  // 学生人数
#define SUBJECT_NUM 3  // 科目数

int main() {
    // 存储学生成绩的二维数组
    // 行：学生，列：科目（0语文，1数学，2英语）
    int scores[STUDENT_NUM][SUBJECT_NUM];
  
    // 1. 输入成绩
    printf("========== 学生成绩录入系统 ==========\n");
    for(int i = 0; i < STUDENT_NUM; i++) {
        printf("\n请输入第%d个学生的成绩：\n", i + 1);
        printf("语文：");
        scanf("%d", &scores[i][0]);
        printf("数学：");
        scanf("%d", &scores[i][1]);
        printf("英语：");
        scanf("%d", &scores[i][2]);
    }
  
    // 2. 显示所有成绩
    printf("\n========== 成绩表 ==========\n");
    printf("学号\t语文\t数学\t英语\t总分\n");
    printf("--------------------------------\n");
  
    for(int i = 0; i < STUDENT_NUM; i++) {
        int sum = 0;
        printf("%d\t", i + 1);
        for(int j = 0; j < SUBJECT_NUM; j++) {
            printf("%d\t", scores[i][j]);
            sum += scores[i][j];
        }
        printf("%d\n", sum);
    }
  
    // 3. 计算各科平均分
    printf("\n========== 各科平均分 ==========\n");
    char* subjects[] = {"语文", "数学", "英语"};
  
    for(int j = 0; j < SUBJECT_NUM; j++) {
        int sum = 0;
        for(int i = 0; i < STUDENT_NUM; i++) {
            sum += scores[i][j];
        }
        float avg = sum / (float)STUDENT_NUM;
        printf("%s平均分：%.2f\n", subjects[j], avg);
    }
  
    // 4. 找出最高分和最低分
    int max_score = scores[0][0], min_score = scores[0][0];
    int max_stu = 0, max_sub = 0, min_stu = 0, min_sub = 0;
  
    for(int i = 0; i < STUDENT_NUM; i++) {
        for(int j = 0; j < SUBJECT_NUM; j++) {
            if(scores[i][j] > max_score) {
                max_score = scores[i][j];
                max_stu = i;
                max_sub = j;
            }
            if(scores[i][j] < min_score) {
                min_score = scores[i][j];
                min_stu = i;
                min_sub = j;
            }
        }
    }
  
    printf("\n最高分：%d分 (学生%d，%s)\n", 
           max_score, max_stu + 1, subjects[max_sub]);
    printf("最低分：%d分 (学生%d，%s)\n", 
           min_score, min_stu + 1, subjects[min_sub]);
  
    return 0;
}

常用函数模板

#include <stdio.h>

// 1. 交换两个整数
void swap(int* a, int* b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

// 2. 判断是否为闰年
int isLeapYear(int year) {
    return (year % 4 == 0 && year % 100 != 0) || (year % 400 == 0);
}

// 3. 计算字符串长度
int stringLength(char str[]) {
    int len = 0;
    while (str[len] != '\0') {
        len++;
    }
    return len;
}

// 4. 复制字符串
void stringCopy(char dest[], char src[]) {
    int i = 0;
    while (src[i] != '\0') {
        dest[i] = src[i];
        i++;
    }
    dest[i] = '\0';
}

// 5. 计算幂运算
int power(int base, int exponent) {
    int result = 1;
    for (int i = 0; i < exponent; i++) {
        result *= base;
    }
    return result;
}

int main() {
    int x = 10, y = 20;
    printf("交换前：x=%d, y=%d\n", x, y);
    swap(&x, &y);
    printf("交换后：x=%d, y=%d\n", x, y);
  
    int year = 2024;
    if (isLeapYear(year)) {
        printf("%d年是闰年\n", year);
    } else {
        printf("%d年不是闰年\n", year);
    }
  
    char name[] = "张三";
    printf("字符串\"%s\"的长度：%d\n", name, stringLength(name));
  
    int num = 2;
    int exp = 10;
    printf("%d的%d次方：%d\n", num, exp, power(num, exp));
  
    return 0;
}
