一、6 种常用Debug指令

指令 全称 作用说明 GDB 示例
int 3 Interrupt 3 软件断点。程序执行到此会触发调试器暂停。常用于手动插入断点。 无需输入，CPU遇到即停。
nop No Operation 空操作。占用一个字节 (0x90)，什么都不做。常用于对齐代码或预留补丁位置。 x/10i $rip (查看后续指令)
ud2 Undefined Instruction 未定义指令。强制产生 SIGILL 非法指令异常，调试器会捕获。用于标记不该执行的代码路径。 run 后自动停在异常处。
pushf / pushfq Push Flags 保存标志寄存器。将 RFLAGS/EFLAGS 压栈，调试时用于查看条件码（如 ZF, CF）。 print $eflags
cpuid CPU Identification 获取CPU信息。常用于检测是否支持特定指令集（如 AVX），调试时用来确认环境。 info registers eax
hlt Halt 停机。使CPU进入低功耗暂停状态，直到收到中断。在用户态程序中使用会触发 SIGSEGV 或异常。 通常用于内核调试，用户态极少用。

三、寄存器

CPU是由运算器，控制器，寄存器等构成

  • 运算器进行信息处理
  • 寄存器进行信息储存
  • 控制器控制器件进行工作

  • 四个通用寄存器

64位 低32位 低16位 低8位 特殊用途 / 备注
RAX EAX AX AL 累加器。mul/div 隐式使用；函数返回值通常存于此。
RBX EBX BX BL 基址寄存器。被调用者保存（Callee-saved），函数内用需先压栈。
RCX ECX CX CL 计数器。loop 指令；shl/shr 移位位数非立即数时存于此。
RDX EDX DX DL 数据寄存器。mul 乘法结果高64位存此；div 除法余数存此；系统调用第3参数。

5.具体实例

.inter_syntax noprefix
.global _start
_start:
    mov rax, 1337
    mov rbx, rax

    mov rcx, 10
    add rcx, 5          ; RCX = 15
    sub rcx, 3          ; RCX = 12

    mov rdx, 100
    inc rdx             ; RDX = 101
    dec rdx             ; RDX = 100

    mov al, 0b11001100
    and al, 0b11110000  ; AL = 0b11000000
    mov bl, 0b00001111
    or  bl, al          ; BL = 0b11001111

    mov rax, 1
    shl rax, 4          ; RAX = 16
    shr rax, 1          ; RAX = 8

    mov rax, 6
    mov rcx, 7
    mul rcx             ; RAX = 42, RDX = 0

    mov rdx, 0          ; 高位清零
    mov rax, 20
    mov rcx, 6
    div rcx             ; RAX = 3 (商), RDX = 2 (余数)

    mov rax, 111
    mov rbx, 222
    xchg rax, rbx       ; RAX = 222, RBX = 111

    mov rax, 5
    neg rax             ; RAX = -5 (0xFFFFFFFFFFFFFFFB)

    nop                 ; 空操作，单步时直接跳过
    pushfq              ; 保存标志位到栈
    popfq               ; 恢复标志位（保持栈平衡）
    int3                ; 断点，GDB 会自动停下

 
    mov rax, 60         ; sys_exit
    xor rdi, rdi        ; 返回 0
    syscall

五、简单程序编写

环境WSL-Ubuntu，gcc

.inter_syntax noprefix
.global _start
_start:
        mov rdi, 42     ;退出状态码
        mov rax, 60     ;系统调用约定 退出
        syscall
as ./program.s -o ./program.o
ld ./program.o -o ./program

> ./program
> echo $?
42

Hello World

.inter_syntax noprefix
.data
msg:
    .ascii "Helko_World\n"
    len = . - msg

.text
    .global _start

_start:
    # 写入
    mov rax, 1 # 系统调用
    mov rdi, 1
    lea rsi, [msg]
    mov rdx, len
    syscall

    #exit
    mov rax, 60
    mov rdi, 0
    syscall
