问题

古典问题（兔子生崽）：有一对兔子，从出生后第3个月起每个月都生一对兔子，小兔子长到第三个月后每个月又生一对兔子，假如兔子都不死，问每个月的兔子总数为多少？（输出前40个月即可）
这题就是斐波那契数列的原定义很经典的题目

解答

#include <stdio.h>

int main(void)
{
    //先定义一些函数
    int a=1,b=1,c,d=2,e;
    //题目是对所以×2，先输出前两行
    printf("%d\n",d);
    printf("%d\n",d);
    //斐波那契数列第三项开始每项是前两项的和
    for(int i = 3;i<41;i++)
    {
        c = a + b;
        d = c * 2;
        e = c;
        a = b;
        b = c;
        c = e;
        //打印数字并换行
        printf("%d\n",d);
    }
    return 0;
}

问题

判断 101 到 200 之间的素数。

解答

#include <stdio.h>

int main(void)
{
    int i,j;
    //遍历101到200的数字
    for(i=101;i<201;i++)
    {
        //根据定义如过果这个数除2到它本身能除开就跳出循环
        for(j=2;j<i;j++)
        {
            if(i%j==0)
                break;
        }
        //判断是否提前跳出循环
        if(j>=i)
        {
            //打印素数
            printf("%d\n",i);
        }
    }
    return 0;
}

问题

打印出所有的”水仙花数”，所谓”水仙花数”是指一个三位数，其各位数字立方和等于该数 本身。例如：153是一个”水仙花数”，因为153=1的三次方＋5的三次方＋3的三次方。

解答

#include <stdio.h>

int main(void)
{
    //定义一些数字
    int i,a,b,c,j;
    //遍历100-999的数字
    for(i =100;i<1000;i++)
    {
        //个位
        a = i % (10);
        //十位
        b = (i / 10) % (10);
        //百位
        c = (i / 100) % (10);
        j = a*a*a+b*b*b+c*c*c;
        //判断是否相等
        if(i==j)
        {
            printf("%d\n",j);
        }
    }
    return 0;
}

问题

将一个正整数分解质因数。例如：输入 90,打印出 90 = 2 * 3 * 3 *5

解答

#include <stdio.h>

int main(void)
{
    int a;
    //随便输入一个a
    printf("Please input number\n");
    scanf("%d", &a);
    //打印出a=
    printf("%d=", a);
    //利用for循环相除，能除开就打印那个数和*并且把除上那个数
    for(int i = 2; i <= a; i++)
    {
        while(a % i == 0)
        {
            //当a=i时打印他本身并且结束循环
            printf("%d", i);
            a = a / i;
            if(a != 1)
            {
                printf("*");
            }
        }
    }
    
    return 0;
}

题目

利用条件运算符的嵌套来完成此题：学习成绩>=90分的同学用A表示，60-89分之间的用B表示，60分以下的用C表示。

解答

#include<stdio.h>
int main()
{
    //(a>b)?a:b这个结构是最基本的结构，于是利用这个结构来判断
    int score;
    char grade;
    printf("Please input score: ");
    scanf("%d",&score);
    //具体演示
    grade=(score>=90)?'A':((score>=60)?'B':'C');
    printf("%c\n",grade);
    return 0;
}
