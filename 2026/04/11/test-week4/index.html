//2. 建立逆S盒（解密查找表）
static unsigned char Inv_SBOX[256];

// 构建逆S盒：建立加密值到原始值的反向映射
static void bliud_Inv_SBOX(void)
{
    int x,y;
    // 遍历高4位（0-15）
    for(x = 0; x < 16; x++)
    {
        // 遍历低4位（0-15）
        for(y = 0; y < 16; y++)
        {
            // 组合成完整字节（0x00-0xFF）
            unsigned char input = (x<<4) | y;
            // 获取S盒加密后的值
            unsigned char output = SBOX[input];
            // 建立反向映射：加密值 -> 原始值
            Inv_SBOX[output] = input;
        }
    }
}

// 3. 加密函数
void SubBytes(unsigned char *input, size_t length)
{
    size_t i;
    for(i = 0; i < length; i++)
    {
        input[i] = Inv_SBOX[SBOX[input[i]]];
    }
}

// 4. 解密函数
// 使用逆S盒将加密字节恢复为原始字节
void Inv_SubBytes(unsigned char *input, size_t length)
{
    size_t i;
    for(i = 0; i < length; i++)
    {
        // 直接查逆S盒表进行解密
        input[i] = Inv_SBOX[input[i]];
    }
}

// 5. 主程序入口
int main(void)
{
    // 文件路径定义
    const char *src_path = "/assets/secret.png.hacked";  // 加密的源文件
    const char *dst_path = "/assets/secret.png";         // 解密后的目标文件

    FILE *fin, *fout;        // 文件指针
    unsigned char *buf;      // 数据缓冲区
    long file_size;          // 文件大小

    // 步骤1：构建解密用的逆S盒
    bliud_Inv_SBOX();

    // 步骤2：打开加密文件
    fin = fopen(src_path, "rb");
    if(!fin)
    {
        printf("failed please retry\n");
        return 1;
    }

    // 步骤3：获取文件大小
    fseek(fin, 0, SEEK_END);    // 移动到文件末尾
    file_size = ftell(fin);      // 获取位置 = 文件大小
    rewind(fin);                 // 回到文件开头

    // 步骤4：分配内存缓冲区
    buf = (unsigned char *)malloc(file_size);
    if (!buf) {
        printf("failed please retry\n");
        fclose(fin);
        return 1;
    }

    // 步骤5：读取整个加密文件到内存
    fread(buf, 1, file_size, fin);
    fclose(fin);  // 关闭源文件

    // 步骤6：执行解密操作（核心步骤）
    Inv_SubBytes(buf, (size_t)file_size);

    // 步骤7：创建并写入解密后的文件
    fout = fopen(dst_path, "wb");
    if (!fout) {
        printf("failed please retry\n");
        free(buf);
        return 1;
    }

    // 步骤8：将解密数据写入新文件
    fwrite(buf, 1, file_size, fout);
    fclose(fout);  // 关闭目标文件

    // 步骤9：释放内存并输出成功信息
    free(buf);
    printf("succesed %s -> %s (%ld bytes)\n", src_path, dst_path, file_size);
    
    return 0;
}
