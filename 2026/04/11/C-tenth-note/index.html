C语言文件读写学习笔记

文件操作基本步骤

  1. 打开文件 → 2. 读写文件 → 3. 关闭文件
#include <stdio.h>

int main() {
    FILE *fp = fopen("test.txt", "w");  // 打开文件
    if (fp == NULL) {
        return -1;  // 打开失败
    }
    fprintf(fp, "Hello");  // 写入数据
    fclose(fp);  // 关闭文件
    return 0;
}

打开文件的方式

模式 含义
“r” 只读，文件必须存在
“w” 只写，文件不存在创建，存在则清空
“a” 追加，文件不存在创建
“rb” 二进制只读
“wb” 二进制只写
“r+” 读写，文件必须存在

常用读写函数

1.字符读写（fputc/fgetc）

int char_io() {
    char *str = "hello world";
    FILE *fp = fopen("char_io.txt", "w");
  
    // 逐个字符写入
    while (*str != '\0') {
        fputc(*str, fp);
        str++;
    }
    fclose(fp);
  
    // 逐个字符读取
    fp = fopen("char_io.txt", "r");
    while (!feof(fp)) {
        char ch = fgetc(fp);
        printf("%c", ch);
    }
    fclose(fp);
    return 0;
}

2.字符串读写（fputs/fgets）

int str_io() {
    char *data[] = {"hello world\n", "hello china\n"};
    FILE *fp = fopen("str_io.txt", "w");
  
    // 逐行写入
    for (int i = 0; i < 2; i++) {
        fputs(data[i], fp);
    }
    fclose(fp);
  
    // 逐行读取
    fp = fopen("str_io.txt", "r");
    char buf[128];
    while (fgets(buf, sizeof(buf), fp) != NULL) {
        printf("%s", buf);
    }
    fclose(fp);
    return 0;
}

3.格式化读写

int format_io() {
    FILE *fp = fopen("format_io.txt", "w");
  
    // 格式化写入
    fprintf(fp, "%s %d", "hello-world", 100);
    fclose(fp);
  
    // 格式化读取
    fp = fopen("format_io.txt", "r");
    char buf[128];
    int value;
    fscanf(fp, "%s %d", buf, &value);
    printf("buf:%s, value:%d\n", buf, value);
    fclose(fp);
    return 0;
}

二进制读写（fwrite / fread）

int binary_io() {
    FILE *fp = fopen("binary_io.txt", "wb");
  
    // 写入字符串和整数
    fwrite("hello world", 12, 1, fp);  // 12包含\0
    int data = 100;
    fwrite(&data, sizeof(data), 1, fp);
    fclose(fp);
  
    // 二进制读取
    fp = fopen("binary_io.txt", "rb");
    char buf[128] = {0};
    int value = 0;
    fread(buf, 12, 1, fp);
    fread(&value, sizeof(value), 1, fp);
    printf("buf:%s, value:%d\n", buf, value);
    fclose(fp);
    return 0;
}

文件指针定位

int seek_demo() {
    FILE *fp = fopen("seek_demo.txt", "w+");
  
    fwrite("hello world", 12, 1, fp);
  
    // rewind：回到文件开头
    rewind(fp);
    fwrite("hello china", 12, 1, fp);
  
    // fseek：移动指针
    fseek(fp, 6, SEEK_SET);  // 从开头偏移6字节
  
    char buf[128];
    fread(buf, 6, 1, fp);    // 读取"china"
    printf("buf:%s\n", buf);
  
    fclose(fp);
    return 0;
}

fseek参数说明

参数 含义
SEEK_SET 从文件开头偏移
SEEK_CUR 从当前位置偏移
SEEK_END 从文件末尾偏移

获取文件大小

long get_filesize(const char *filepath) {
    FILE *fp = fopen(filepath, "r");
    if (fp == NULL) return 0;
  
    fseek(fp, 0, SEEK_END);      // 指针移到末尾
    long size = ftell(fp);        // 获取当前位置
    fclose(fp);
    return size;
}

学生信息储存演示

#include <stdio.h>

typedef struct {
    int id;
    char name[20];
    float score;
} Student;

// 保存学生信息到文件
void save_students(Student stu[], int n) {
    FILE *fp = fopen("students.dat", "wb");
    fwrite(stu, sizeof(Student), n, fp);
    fclose(fp);
}

// 从文件读取学生信息
void load_students(Student stu[], int n) {
    FILE *fp = fopen("students.dat", "rb");
    fread(stu, sizeof(Student), n, fp);
    fclose(fp);
}

int main() {
    Student students[3] = {
        {1001, "张三", 88.5},
        {1002, "李四", 92.0},
        {1003, "王五", 78.5}
    };
  
    save_students(students, 3);
  
    Student read_stu[3];
    load_students(read_stu, 3);
  
    for (int i = 0; i < 3; i++) {
        printf("%d %s %.1f\n", read_stu[i].id, 
               read_stu[i].name, read_stu[i].score);
    }
  
    return 0;
}
