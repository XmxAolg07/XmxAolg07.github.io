unsigned __int16 __fastcall vm::step(VM *this)
{
  unsigned int n0xFC; // eax
  unsigned __int16 v2; // dx
  unsigned __int16 sp; // ax
  unsigned __int16 v4; // cx
  unsigned __int16 v5; // cx
  unsigned __int16 v6; // ax
  unsigned __int16 result; // ax
  unsigned __int16 v8; // [rsp+1Eh] [rbp-2h]

  v8 = this->regs[this->mem[this->pc + 2]];
  n0xFC = this->mem[this->pc];
  if ( n0xFC > 0xFC )
    return 0;
  if ( this->mem[this->pc] < 0xB0u )
  {
    if ( this->mem[this->pc] <= 0xA0u )
    {
      if ( this->mem[this->pc] >= 0x80u )
      {
        switch ( this->mem[this->pc] )
        {
          case 0x80u:
            this->regs[this->mem[this->pc + 1]] = v8;
            this->pc += 3;
            break;
          case 0x81u:
            this->regs[this->mem[this->pc + 1]] = (this->mem[this->pc + 3] << 8) | this->mem[this->pc + 2];
            this->pc += 4;
            break;
          case 0x90u:
            this->regs[this->mem[this->pc + 1]] += v8;
            this->pc += 3;
            break;
          case 0x91u:
            this->regs[this->mem[this->pc + 1]] += this->mem[this->pc + 2] | (this->mem[this->pc + 3] << 8);
            this->pc += 4;
            break;
          case 0xA0u:
            this->regs[this->mem[this->pc + 1]] -= v8;
            this->pc += 3;
            break;
          default:
            return 0;
        }
      }
      else if ( n0xFC == 0x4C )
      {
        sp = this->sp;
        this->sp = sp - 1;
        this->regs[this->mem[this->pc + 1]] = this->stack[sp];
        this->pc += 2;
      }
      else if ( this->mem[this->pc] <= 0x4Cu )
      {
        if ( this->mem[this->pc] > 0x29u )
        {
          if ( n0xFC == 0x48 )
          {
            v2 = this->regs[this->mem[this->pc + 1]];
            this->stack[++this->sp] = v2;
            this->pc += 2;
          }
        }
        else if ( this->mem[this->pc] >= 8u )
        {
          switch ( this->mem[this->pc] )
          {
            case 8u:
              this->regs[this->mem[this->pc + 1]] = this->mem[v8];
              this->pc += 3;
              break;
            case 0xAu:
              this->regs[this->mem[this->pc + 1]] = (this->mem[v8 + 1] << 8) | this->mem[v8];
              this->pc += 3;
              break;
            case 0x18u:
              this->mem[this->regs[this->mem[this->pc + 1]]] = v8;
              this->pc += 3;
              break;
            case 0x1Au:
              this->mem[this->regs[this->mem[this->pc + 1]]] = v8;
              this->mem[this->regs[this->mem[this->pc + 1]] + 1] = HIBYTE(v8);
              this->pc += 3;
              break;
            case 0x28u:
              this->flags = v8 == this->regs[this->mem[this->pc + 1]];
              this->pc += 3;
              break;
            case 0x29u:
              this->flags = this->regs[this->mem[this->pc + 1]] == (this->mem[this->pc + 2]
                                                                  | (this->mem[this->pc + 3] << 8));
              this->pc += 4;
              break;
            default:
              return 0;
          }
        }
      }
    }
    return 0;
  }
  switch ( this->mem[this->pc] )
  {
    case 0xB0u:
      this->regs[this->mem[this->pc + 1]] ^= v8;
      this->pc += 3;
      return 0;
    case 0xC1u:
      this->regs[this->mem[this->pc + 1]] *= this->mem[this->pc + 2] | (unsigned __int16)(this->mem[this->pc + 3] << 8);
      this->pc += 4;
      return 0;
    case 0xE0u:
      this->pc = this->mem[this->pc + 1] | (this->mem[this->pc + 2] << 8);
      return 0;
    case 0xE8u:
      if ( this->flags )
        this->pc = this->mem[this->pc + 1] | (this->mem[this->pc + 2] << 8);
      else
        this->pc += 3;
      return 0;
    case 0xECu:
      if ( this->flags )
        this->pc += 3;
      else
        this->pc = this->mem[this->pc + 1] | (this->mem[this->pc + 2] << 8);
      return 0;
    case 0xEEu:
      v4 = this->pc + 2;
      this->stack[++this->sp] = v4;
      this->pc = this->regs[this->mem[this->pc + 1]];
      return 0;
    case 0xEFu:
      v5 = this->pc + 3;
      this->stack[++this->sp] = v5;
      this->pc = this->mem[this->pc + 1] | (this->mem[this->pc + 2] << 8);
      return 0;
    case 0xF0u:
      v6 = this->sp;
      this->sp = v6 - 1;
      this->pc = this->stack[v6];
      return 0;
    case 0xF8u:
      result = -1;
      break;
    case 0xFCu:
      if ( this->regs[0] )
        this->regs[0] = read(0, &this->mem[this->regs[1]], this->regs[2]);
      else
        write(1, &this->mem[this->regs[1]], this->regs[2]);
      ++this->pc;
      return 0;
    default:
      return 0;
  }
  return result;
}

__int64 __fastcall sub_16BC(unsigned __int16 *this)
{
  unsigned int n0xFC; // eax
  unsigned __int16 v2; // dx
  unsigned __int16 v3; // ax
  unsigned __int16 v4; // cx
  unsigned __int16 v5; // cx
  unsigned __int16 v6; // ax
  __int64 result; // rax
  unsigned __int16 v8; // [rsp+1Eh] [rbp-2h]

  v8 = *(this + *((unsigned __int8 *)this + *(this + 2304) + 2) + 2307);
  n0xFC = *((unsigned __int8 *)this + *(this + 2304));
  if ( n0xFC > 0xFC )
    return 0;
  if ( *((unsigned __int8 *)this + *(this + 2304)) < 0xB0u )
  {
    if ( *((unsigned __int8 *)this + *(this + 2304)) <= 0xA0u )
    {
      if ( *((unsigned __int8 *)this + *(this + 2304)) >= 0x80u )
      {
        switch ( *((_BYTE *)this + *(this + 2304)) )
        {
          case 0x80:
            *(this + *((unsigned __int8 *)this + *(this + 2304) + 1) + 2307) = v8;
            *(this + 2304) += 3;
            break;
          case 0x81:
            *(this + *((unsigned __int8 *)this + *(this + 2304) + 1) + 2307) = (*((unsigned __int8 *)this
                                                                                + *(this + 2304)
                                                                                + 3) << 8)
                                                                             | *((unsigned __int8 *)this
                                                                               + *(this + 2304)
                                                                               + 2);
            *(this + 2304) += 4;
            break;
          case 0x90:
            *(this + *((unsigned __int8 *)this + *(this + 2304) + 1) + 2307) += v8;
            *(this + 2304) += 3;
            break;
          case 0x91:
            *(this + *((unsigned __int8 *)this + *(this + 2304) + 1) + 2307) += *((unsigned __int8 *)this
                                                                                + *(this + 2304)
                                                                                + 2)
                                                                              | (*((unsigned __int8 *)this
                                                                                 + *(this + 2304)
                                                                                 + 3) << 8);
            *(this + 2304) += 4;
            break;
          case 0xA0:
            *(this + *((unsigned __int8 *)this + *(this + 2304) + 1) + 2307) -= v8;
            *(this + 2304) += 3;
            break;
          default:
            return 0;
        }
      }
      else if ( n0xFC == 76 )
      {
        v3 = *(this + 2305);
        *(this + 2305) = v3 - 1;
        *(this + *((unsigned __int8 *)this + *(this + 2304) + 1) + 2307) = *(this + v3 + 2048);
        *(this + 2304) += 2;
      }
      else if ( *((unsigned __int8 *)this + *(this + 2304)) <= 0x4Cu )
      {
        if ( *((unsigned __int8 *)this + *(this + 2304)) > 0x29u )
        {
          if ( n0xFC == 72 )
          {
            v2 = *(this + *((unsigned __int8 *)this + *(this + 2304) + 1) + 2307);
            *(this + ++*(this + 2305) + 2048) = v2;
            *(this + 2304) += 2;
          }
        }
        else if ( *((unsigned __int8 *)this + *(this + 2304)) >= 8u )
        {
          switch ( *((_BYTE *)this + *(this + 2304)) )
          {
            case 8:
              *(this + *((unsigned __int8 *)this + *(this + 2304) + 1) + 2307) = *((unsigned __int8 *)this + v8);
              *(this + 2304) += 3;
              break;
            case 0xA:
              *(this + *((unsigned __int8 *)this + *(this + 2304) + 1) + 2307) = (*((unsigned __int8 *)this + v8 + 1) << 8)
                                                                               | *((unsigned __int8 *)this + v8);
              *(this + 2304) += 3;
              break;
            case 0x18:
              *((_BYTE *)this + *(this + *((unsigned __int8 *)this + *(this + 2304) + 1) + 2307)) = v8;
              *(this + 2304) += 3;
              break;
            case 0x1A:
              *((_BYTE *)this + *(this + *((unsigned __int8 *)this + *(this + 2304) + 1) + 2307)) = v8;
              *((_BYTE *)this + *(this + *((unsigned __int8 *)this + *(this + 2304) + 1) + 2307) + 1) = HIBYTE(v8);
              *(this + 2304) += 3;
              break;
            case 0x28:
              *(this + 2306) = v8 == *(this + *((unsigned __int8 *)this + *(this + 2304) + 1) + 2307);
              *(this + 2304) += 3;
              break;
            case 0x29:
              *(this + 2306) = *(this + *((unsigned __int8 *)this + *(this + 2304) + 1) + 2307) == (*((unsigned __int8 *)this + *(this + 2304) + 2)
                                                                                                  | (*((unsigned __int8 *)this + *(this + 2304) + 3) << 8));
              *(this + 2304) += 4;
              break;
            default:
              return 0;
          }
        }
      }
    }
    return 0;
  }
  switch ( *((_BYTE *)this + *(this + 2304)) )
  {
    case 0xB0:
      *(this + *((unsigned __int8 *)this + *(this + 2304) + 1) + 2307) ^= v8;
      *(this + 2304) += 3;
      return 0;
    case 0xC1:
      *(this + *((unsigned __int8 *)this + *(this + 2304) + 1) + 2307) *= *((unsigned __int8 *)this + *(this + 2304) + 2)
                                                                        | (unsigned __int16)(*((unsigned __int8 *)this
                                                                                             + *(this + 2304)
                                                                                             + 3) << 8);
      *(this + 2304) += 4;
      return 0;
    case 0xE0:
      *(this + 2304) = *((unsigned __int8 *)this + *(this + 2304) + 1)
                     | (*((unsigned __int8 *)this + *(this + 2304) + 2) << 8);
      return 0;
    case 0xE8:
      if ( *(this + 2306) )
        *(this + 2304) = *((unsigned __int8 *)this + *(this + 2304) + 1)
                       | (*((unsigned __int8 *)this + *(this + 2304) + 2) << 8);
      else
        *(this + 2304) += 3;
      return 0;
    case 0xEC:
      if ( *(this + 2306) )
        *(this + 2304) += 3;
      else
        *(this + 2304) = *((unsigned __int8 *)this + *(this + 2304) + 1)
                       | (*((unsigned __int8 *)this + *(this + 2304) + 2) << 8);
      return 0;
    case 0xEE:
      v4 = *(this + 2304) + 2;
      *(this + ++*(this + 2305) + 2048) = v4;
      *(this + 2304) = *(this + *((unsigned __int8 *)this + *(this + 2304) + 1) + 2307);

      return 0;

      v6 = *(this + 2305);
      *(this + 2305) = v6 - 1;
      *(this + 2304) = *(this + v6 + 2048);
      return 0;

      result = 0xFFFFFFFFLL;
      break;

        *(this + 2307) = read(0, (char *)this + *(this + 2308), *(this + 2309));

对比正确代码与ida原始输出整理出

正确代码 ida原始偏移 字节偏移 含义
this->mem[addr] *((unsigned __int8 *)this + addr) 可变 内存字节
this->pc *((_WORD *)this + 2304) 4608 程序计数器
this->sp *((_WORD *)this + 2305) 4610 栈指针
this->flags *((_WORD *)this + 2306) 4612 标志位

字节偏移 4612:         FLAGS

第二题

2.将一个2行3列的二维数组转置为3行2列的数组

示例：

Input 6 integers:
123 43 12
64 234 12
转换后的数组:
  123   64
   43  234
   12   12

解析

#include <stdio.h>
//先判断此题如何解决，要把一个数组行和列换一下
int main(void)
{
    //定义二维数组
    int test[2][3]={
        {123,43,12},
        {64,234,12}
    };
    //定义转换后的数组
    int temp[3][2]={0};
    //利用for循环把test的行和temp的列互换
    for(int i=0;i<2;i++)
    {
        for(int j=0;j<3;j++)
        {
            temp[j][i]=test[i][j];
        }
    }
    //遍历转换后的数组
    for(int i=0;i<3;i++)
    {
        for(int j=0;j<2;j++)
        {
            printf("%d ",temp[i][j]);
        }
        //每行打印一个换行符
        printf("\n");
    }
    return 0;
    
}

第三题

3.编写一个C语言程序，使用指针将数组{10, 20, 30, 40, 50}反转并输出结果。

实现如下：

原数组: 10 20 30 40 50
反转后: 50 40 30 20 10

解析

#include <stdio.h>
//这题要求用指针反转一维数组
int main(void)
{
    //定义一维数组
    int test[5]={10,20,30,40,50};
    //定义指针地址从头和尾
    int *p =test;
    int *q =test+4;
    //使用临时数据
    int temp;
    //使用循环然数组的头和尾互换，然后指针指向下/上一个
    while (p<q)
    {
        temp=*p;
        *p=*q;
        *q=temp;
        p++;
        q--;
    }
    //遍历数组
    for(int i=0;i<5;i++)
    {
        printf("%d ",test[i]);
    }

    return 0;
}
