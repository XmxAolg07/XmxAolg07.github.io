题目：用*号输出字母C的图案

解答

#include <stdio.h>

int main(void)
{
    //使用空格和换行符打印
    printf("    ******    \n");
    printf("   *      *\n");
    printf("  *\n");
    printf(" *\n");
    printf(" *\n");
    printf(" *\n");
    printf(" *\n");
    printf(" *\n");
    printf("  *\n");
    printf("   *      *\n");
    printf("    ******    \n");
    return 0;
}

#include <stdio.h>

    printf("%c%c%c%c%c\n",a,a,b,a,a);

注意：如果输出为乱码把扩展的ASCLL编码改为437 OEM美国就好，具体操作如下

  1. 点击运行界面左上角标题栏图标【c:\】，选择默认值一项
  2. 修改默认代码页，936（ANSI/OEM-简体中文GBK）为437 OEM-美国
  3. 关闭后重新运行一下即可

题目：输出 9*9 口诀

#include <stdio.h>

int main(void)
{
    //定义三个函数
    int i,j,k;
    //利用for循环打印
    for(i=1;i<10;i++)
    {
        for(j=1;j<=i;j++)
        {
            k = i * j;
            printf("%d * %d = %d   ",i,j,k);
        }
        //每一行一换行
        printf("\n");
    }
    return 0;
}

题目：要求输出国际象棋棋盘

解答

int main()
{
    int i,j;
    for(i=0;i<8;i++)
    {
        for(j=0;j<8;j++)
            if((i+j)%2==0)
                printf("%c%c",219,219);
            else printf("  ");
        printf("\n");
    }
    return 0;
}

题目：打印楼梯，同时在楼梯上方打印两个笑脸。

解答

#include<stdio.h>
 
int main()
{
    int i,j;
    printf("\1\1\n");
    {
        for(j=1;j<=i;j++)
            printf("%c%c",219,219);
        printf("\n");
    }
    return 0;
}
