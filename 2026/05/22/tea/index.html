TEA系列算法及其魔改

2026/5/22 算法 CTF解题

TEA算法和魔改

TEA算法原理

TEA核心参数和原理

TEA的算法主要依靠一下几个核心参数

参数 说明
算法类型 对称分组加密算法
明文长度 64位
密钥长度 128位
迭代轮数 32，64，轮数越多越安全
核心操作 异或（xor）移位（shift)模2^32加法
算法重构 feistel结构

128位密钥被等分为4个32位子密钥K[0],K[1],K[2],K[3]

64位明文被分为左右两个32位块

算法定义一个特殊的常数
δ=0x9E3779B9

选择该常数是为了让每轮运算使用不同的累加值，打破轮与轮之间的对称性，从而抵抗差分密码分析

加密/解密过程

加密

加密循环执行32轮（或64轮），每轮操作如下：

设当前累加和为S（初始S=），每轮执行：

  1. 更新累加和：S=S+δ
  2. 更新左半部：V0=V0+(((V1≪4)+K[0])⊕(V1+S)⊕((V1≫5)+K[1]))
  3. 更新右半部：V1=V1+(((V0≪4)+K[2])⊕(V0+S)⊕((V0≫5)+K[3]))

轮函数的设计思路在于非线性混淆：

  • 左移4位与右移5位的非对称移位，使得数据的高低比特位充分混合
  • 三个分支分别引入子密钥和累加和，形成三路异或结构
  • 累加和SS逐轮递增，确保每轮变换均不相同

其结构可形式化表示为Feistel结构的变体：

F(x,S,Ka,Kb)=((x≪4)+Ka)⊕(x+S)⊕((x≫5)+Kb)

解密

解密是加密的严格逆过程。累加和初始化为S=δ×rounds，每轮执行：

  1. V1=V1−(((V0≪4)+K[2])⊕(V0+S)⊕((V0≫5)+K[3]))
  2. V0=V0−(((V1≪4)+K[0])⊕(V1+S)⊕((V1≫5)+K[1]))
  3. S=S−δ

注意解密时更新顺序与加密相反，先恢复V1再恢复V0。

# TEA加密算法实现
# 64位分组，128位密钥，32轮Feistel结构

def tea_encrypt(v, k):
    # 拆分明文为左右两个32位块
    v0 = v[0]
    v1 = v[1]
    # 黄金分割常数，确保每轮变换不同
    delta = 0x9E3779B9
    # 累加和初始为0
    sum_val = 0

    for i in range(32):
        # 累加和逐轮递增
        sum_val = (sum_val + delta) & 0xFFFFFFFF
        # 左半部更新：三路异或结构
        v0 = (v0 + (((v1 << 4) + k[0]) ^ (v1 + sum_val) ^ ((v1 >> 5) + k[1]))) & 0xFFFFFFFF
        # 右半部更新：使用不同的子密钥
        v1 = (v1 + (((v0 << 4) + k[2]) ^ (v0 + sum_val) ^ ((v0 >> 5) + k[3]))) & 0xFFFFFFFF

    return [v0, v1]

def tea_decrypt(v, k):
    v0 = v[0]
    v1 = v[1]
    delta = 0x9E3779B9
    # 解密时累加和从最大值开始递减
    sum_val = (delta * 32) & 0xFFFFFFFF

    for i in range(32):
        # 先恢复右半部，与加密顺序相反
        v1 = (v1 - (((v0 << 4) + k[2]) ^ (v0 + sum_val) ^ ((v0 >> 5) + k[3]))) & 0xFFFFFFFF
        # 再恢复左半部
        v0 = (v0 - (((v1 << 4) + k[0]) ^ (v1 + sum_val) ^ ((v1 >> 5) + k[1]))) & 0xFFFFFFFF
        # 累加和递减
        sum_val = (sum_val - delta) & 0xFFFFFFFF

    return [v0, v1]

def tea_encrypt_string(data, key):
    # 将16字节密钥拆分为4个32位整数（小端序）
    key_ints = []
    for i in range(0, 16, 4):
        key_ints.append(int.from_bytes(key[i:i+4], 'little'))

    # 零填充至8字节对齐
    data = bytearray(data)
    while len(data) % 8 != 0:
        data.append(0)

    # 逐分组加密（ECB模式）
    result = bytearray()
    for i in range(0, len(data), 8):
        # 将8字节分为左右两个32位字
        v0 = int.from_bytes(data[i:i+4], 'little')
        v1 = int.from_bytes(data[i+4:i+8], 'little')
        enc = tea_encrypt([v0, v1], key_ints)
        # 拼接密文（小端序）
        result.extend(enc[0].to_bytes(4, 'little'))
        result.extend(enc[1].to_bytes(4, 'little'))

    return bytes(result)

def tea_decrypt_string(data, key):
    # 密钥处理同加密
    key_ints = []
    for i in range(0, 16, 4):
        key_ints.append(int.from_bytes(key[i:i+4], 'little'))

    # 逐分分解密
    result = bytearray()
    for i in range(0, len(data), 8):
        v0 = int.from_bytes(data[i:i+4], 'little')
        v1 = int.from_bytes(data[i+4:i+8], 'little')
        dec = tea_decrypt([v0, v1], key_ints)
        result.extend(dec[0].to_bytes(4, 'little'))
        result.extend(dec[1].to_bytes(4, 'little'))

    # 去除填充的零字节
    return bytes(result).rstrip(b'\x00')

{
    // 拆分明文为左右两个32位块
    uint32_t v0 = v[0];
    uint32_t v1 = v[1];

    uint32_t sum = 0;
    int i;

    {
        // 累加和逐轮递增
        sum += delta;
        // 左半部更新：三路异或结构 (左移4位 + k0) ^ (原值 + sum) ^ (右移5位 + k1)
        v0 += ((v1 << 4) + k[0]) ^ (v1 + sum) ^ ((v1 >> 5) + k[1]);
        // 右半部更新：使用不同的子密钥k2、k3
        v1 += ((v0 << 4) + k[2]) ^ (v0 + sum) ^ ((v0 >> 5) + k[3]);
    }
    
    // 写回加密结果
    v[0] = v0;
    v[1] = v1;
}

// 核心解密函数：加密的严格逆运算
void tea_decrypt(uint32_t v[2], uint32_t k[4])
{
    uint32_t v0 = v[0];
    uint32_t v1 = v[1];
    uint32_t delta = 0x9e3779b9;
    // 解密时累加和从最大值开始递减
    uint32_t sum = delta * 32;
    int i;

    for ( i = 0; i < 32; i++)
    {
        // 先恢复右半部，与加密顺序相反

        v0 -= ((v1 << 4) + k[0]) ^ (v1 + sum) ^ ((v1 >> 5) + k[1]);
        // 累加和递减
        sum -= delta;
    }
    
    v[0] = v0;
    v[1] = v1;
}

XTEA算法原理

改进

  1. V0=V0+((((V1≪4)⊕(V1≫5))+V1)⊕(S+K[S mod 4]))
  2. S=S+δ
  3. V1=V1+((((V0≪4)⊕(V0≫5))+V0)⊕(S+K[(S≫11) mod 4]))

与原版TEA的关键区别：

  • 移位结果先异或后加原值：((x≪4)⊕(x≫5))+x，代替原来的((x≪4)+Ka)⊕⋯结构。这种设计使得非线性特性更强，且不再依赖子密钥参与该部分运算
  • 动态密钥索引：V0的更新使用K[S mod 4]，V1V1的更新使用K[(S≫11) mod 4]。子密钥的选择随SS的递增而变化，使得每轮使用的子密钥组合不可预测，彻底消除了等效密钥
  • 累加和更新时机：S在V0更新之后、V1更新之前递增，使得左右两半的更新使用不同的S值（V0用旧S，V1用新S），增加了轮内不对称性

XXTEA算法原理

改进

XXTEA的加密轮数不再固定，而是根据数据长度nn动态调整：

R=6+⌊52n⌋

该公式的设计逻辑在于：当nn较小时（如n=2，仅64位数据），轮数自动增加至6+26=32轮，以弥补短数据混淆不充分的弱点；当nn较大时，轮数相应减少，因为长数据本身的字间扩散已经提供了足够的混淆效果。这种自适应机制在安全性与效率之间取得了平衡。

XXTEA定义了一个名为MX的混合函数

mx(sum_val, y, z, p, e, k, round_num):
        shift1 = (z >> 7) ^ (y << 1)
        shift2 = (y >> 5) ^ (z << 3)
        mixed = (shift1 + shift2) ^ ((sum_val + y) ^ (k[((p & 3) ^ e + round_num) & 3] | z))
        return mixed ^ round_const[round_num & 7]

def xxtea_encrypt(data, key):

    def mx(sum_val, y, z, p, e, k):
        # 前一部分：z和y的移位交叉组合
        # 后一部分：累加和与子密钥的异或混合
        return (((z >> 5) ^ (y << 2)) + ((y >> 3) ^ (z << 4))) ^ ((sum_val ^ y) + (k[(p & 3) ^ e] ^ z))
    
    # 零填充至4字节对齐
    data = bytearray(data)
    while len(data) % 4 != 0:
        data.append(0)

    # 将数据转换为32位字数组
    n = len(data) // 4
    v = []
    for i in range(n):
        v.append(int.from_bytes(data[i*4:(i+1)*4], 'little'))

    # 密钥填充至16字节后拆分为4个32位整数
    key = bytearray(key)
    while len(key) < 16:
        key.append(0)

    k = []
    for i in range(0, 16, 4):
        k.append(int.from_bytes(key[i:i+4], 'little'))

    # 数据不足2个字时直接返回
    if n < 2:
        return data
    
    # 自适应轮数：数据越短轮数越多
    q = 6 + 52 // n
    sum_val = 0
    z = v[n - 1]

    # 外层循环：共q轮
    while q > 0:
        # 累加和递增
        sum_val = (sum_val + 0x9e3779b9) & 0xffffffff
        # 密钥索引偏移
        e = (sum_val >> 2) & 3

            y = v[p + 1]
            v[p] = (v[p] + mx(sum_val, y, z, p, e, k)) & 0xffffffff
            z = v[p]

        y = v[0]
        v[n - 1] = (v[n - 1] + mx(sum_val, y, z, n - 1, e, k)) & 0xffffffff
        z = v[n - 1]

        q -= 1

    # 将字数组转换回字节数组（小端序）
    result = bytearray()
    for val in v:
        result.extend(val.to_bytes(4, 'little'))

    return bytes(result)

    result = bytearray()
    for val in v:
        result.extend(val.to_bytes(4, 'little'))

void xxtea_encrypt(unsigned char *data, int data_len, unsigned char *key, unsigned char *output, int *output_len)
{
    uint32_t k[4];
    int i;
    
    // 将16字节密钥转换为4个32位整数（小端序）
    for (i = 0; i < 4; i++)
    {
        k[i] = ((uint32_t)key[i*4]) |
               ((uint32_t)key[i*4+1] << 8) |
               ((uint32_t)key[i*4+2] << 16) |
               ((uint32_t)key[i*4+3] << 24);
    }
    
    // 零填充至4字节对齐
    int padding = (4 - data_len % 4) % 4;
    int padded_len = data_len + padding;
    unsigned char *padded_data = (unsigned char*)malloc(padded_len);
    memcpy(padded_data, data, data_len);
    memset(padded_data + data_len, 0, padding);
    
    // 字节数组转32位字数组（小端序）
    int n = padded_len / 4;
    uint32_t *v = (uint32_t*)malloc(n * sizeof(uint32_t));
    for (i = 0; i < n; i++)
    {
        v[i] = ((uint32_t)padded_data[i*4]) |
               ((uint32_t)padded_data[i*4+1] << 8) |
               ((uint32_t)padded_data[i*4+2] << 16) |
               ((uint32_t)padded_data[i*4+3] << 24);
    }
    
    // 数据过短时直接拷贝返回
    if (n < 2)
    {
        memcpy(output, padded_data, padded_len);
        *output_len = padded_len;
        free(padded_data);
        free(v);
        return;
    }
    
    // 自适应轮数：6 + 52/n，n越小轮数越多
    int q = 6 + 52 / n;
    uint32_t sum_val = 0;
    uint32_t z = v[n - 1];
    uint32_t y;
    int p;
    
    // 外层循环：共q轮
    while (q > 0)
    {
        // 累加和递增，使用黄金分割常数
        sum_val += 0x9E3779B9;
        // 根据累加和高位计算密钥索引偏移
        uint32_t e = (sum_val >> 2) & 3;

        for (p = 0; p < n - 1; p++)
        {
            y = v[p + 1];
            v[p] += mx(sum_val, y, z, p, e, k);
            z = v[p];
        }
        
        // 处理最后一个字：后继为v[0]，前驱为z
        y = v[0];
        v[n - 1] += mx(sum_val, y, z, n - 1, e, k);
        z = v[n - 1];
        
        q--;
    }

    *output_len = n * 4;

    {

    }
    
    free(padded_data);
    free(v);
}

        sum_val -= 0x9E3779B9;
        q--;

XXTEA魔改

前言

XXTEA算法在实战中同样也会衣魔改的身份存在接下给出如下魔改方式的代码作为演示

双混合函数：前2轮用标准MX，后续轮用魔改MX（不同移位位数）

负delta值：使用0x61C88647（-0x9E3779B9）改变累加方式

循环遍历：所有字统一处理，消除边界特殊处理

轮数调整：基础轮数8 + 52/n（标准为6 + 52/n）

# XXTEA魔改变体加密算法实现
def _bytes_to_words(data):
    """将字节数组转换为32位字数组（小端序），自动零填充至4字节对齐"""
    n = len(data)
    if n % 4:
        data += b'\x00' * (4 - n % 4)
    words = []
    for i in range(0, len(data), 4):
        # 4字节小端序组装为32位整数
        w = data[i] | (data[i+1] << 8) | (data[i+2] << 16) | (data[i+3] << 24)
        words.append(w & 0xFFFFFFFF)
    return words

def _words_to_bytes(words):
    """将32位字数组转换回字节数组（小端序）"""
    data = bytearray()
    for w in words:
        data.append(w & 0xFF)
        data.append((w >> 8) & 0xFF)
        data.append((w >> 16) & 0xFF)
        data.append((w >> 24) & 0xFF)
    return bytes(data)

def _mx(z, y, total, key, p, e):
    """标准MX混合函数：右移5左移2 + 右移3左移4"""
    return ((z >> 5 ^ y << 2) + (y >> 3 ^ z << 4)) ^ ((total ^ y) + (key[(p & 3) ^ e] ^ z))

def _magic_mx(z, y, total, key, p, e):
    """魔改MX混合函数：改变移位位数（右移7左移6 + 右移5左移3），增强非线性扩散"""
    h = (z >> 7 ^ y << 6) + (y >> 5 ^ z << 3)
    return h ^ ((total ^ y) + (key[(p & 3) ^ e] ^ z))

def xxtea_encrypt(data, key):
    if not data or len(data) == 0:
        return b''
    
    # 密钥处理：截取或填充至16字节
    key_bytes = key[:16]
    while len(key_bytes) < 16:
        key_bytes += b'\x00'
    
    # 密钥转换为4个32位字（小端序）
    key_words = []
    for i in range(0, 16, 4):
        w = key_bytes[i] | (key_bytes[i+1] << 8) | (key_bytes[i+2] << 16) | (key_bytes[i+3] << 24)
        key_words.append(w & 0xFFFFFFFF)
    
    # 明文转换为字数组
    v = _bytes_to_words(data)
    n = len(v)
    
    # 魔改轮数：8 + 52/n（标准XXTEA为6 + 52/n）
    rounds = 8 + 52 // n
    
    total = 0
    # 魔改delta：使用负值0x61C88647（标准为0x9E3779B9）
    delta = 0x61C88647
    
    for r in range(rounds):
        total = (total + delta) & 0xFFFFFFFF
        
        e = (total >> 2) & 3
        if r < 2:
            # 前2轮使用标准MX函数进行初始扩散
            for p in range(n):
                y = v[(p + 1) % n]  # 后继字（循环取模）
                v[p] = (v[p] + _mx(v[(p - 1) % n], y, total, key_words, p, e)) & 0xFFFFFFFF
        else:
            # 后续轮使用魔改MX函数增强混淆
            for p in range(n):
                y = v[(p + 1) % n]  # 后继字（循环取模）
                v[p] = (v[p] + _magic_mx(v[(p - 1) % n], y, total, key_words, p, e)) & 0xFFFFFFFF
    
    return _words_to_bytes(v)

def xxtea_decrypt(data, key):
    if not data or len(data) == 0:
        return b''
    
    # 去除多余字节保证4字节对齐
    if len(data) % 4:
        data = data[:-(len(data) % 4)]
    
    # 密钥处理同加密
    key_bytes = key[:16]
    while len(key_bytes) < 16:
        key_bytes += b'\x00'
    
    key_words = []
    for i in range(0, 16, 4):
        w = key_bytes[i] | (key_bytes[i+1] << 8) | (key_bytes[i+2] << 16) | (key_bytes[i+3] << 24)
        key_words.append(w & 0xFFFFFFFF)
    
    v = _bytes_to_words(data)
    n = len(v)
    
    rounds = 8 + 52 // n
    
    # 解密时total从最大值开始递减
    total = (0x61C88647 * rounds) & 0xFFFFFFFF
    
    for r in range(rounds):
        e = (total >> 2) & 3
        if r < rounds - 2:
            # 最后2轮对应加密的前2轮，使用魔改MX（因为加密时这两轮用了魔改MX）
            for p in range(n - 1, -1, -1):
                y = v[(p + 1) % n]
                v[p] = (v[p] - _magic_mx(v[(p - 1) % n], y, total, key_words, p, e)) & 0xFFFFFFFF
        else:
            # 前若干轮对应加密的后2轮，使用标准MX（因为加密时后2轮用了标准MX）
            for p in range(n - 1, -1, -1):
                y = v[(p + 1) % n]
                v[p] = (v[p] - _mx(v[(p - 1) % n], y, total, key_words, p, e)) & 0xFFFFFFFF
        
        total = (total - 0x61C88647) & 0xFFFFFFFF
    
    result = _words_to_bytes(v)
    
    # 去除零填充
    end = len(result)
    while end > 0 and result[end-1] == 0:

// XXTEA魔改变体加密算法C语言实现
#include <stdio.h>
#include <string.h>
#include <stdlib.h>
#include <stdint.h>

// 魔改delta：使用负值（标准为0x9E3779B9）
#define DELTA 0x61C88647

// 标准MX混合函数：右移5左移2 + 右移3左移4
uint32_t _mx(uint32_t z, uint32_t y, uint32_t total, uint32_t key[4], uint32_t p, uint32_t e) {
    return ((z >> 5 ^ y << 2) + (y >> 3 ^ z << 4)) ^ ((total ^ y) + (key[(p & 3) ^ e] ^ z));
}

// 魔改MX混合函数：改变移位位数（右移7左移6 + 右移5左移3），增强非线性扩散
uint32_t _magic_mx(uint32_t z, uint32_t y, uint32_t total, uint32_t key[4], uint32_t p, uint32_t e) {
    uint32_t h = (z >> 7 ^ y << 6) + (y >> 5 ^ z << 3);
    return h ^ ((total ^ y) + (key[(p & 3) ^ e] ^ z));
}

// 字节数组转32位字数组（小端序），自动零填充至4字节对齐
void _bytes_to_words(const uint8_t* data, size_t data_len, uint32_t* words, size_t* words_len) {
    size_t n = data_len;
    size_t padding = 0;
    if (n % 4) {
        padding = 4 - n % 4;
    }
    
    *words_len = (n + padding) / 4;
    
    for (size_t i = 0; i < *words_len; i++) {
        words[i] = 0;
        for (int j = 0; j < 4; j++) {
            size_t idx = i * 4 + j;
            if (idx < n) {
                words[i] |= ((uint32_t)data[idx]) << (j * 8);
            }
        }
    }
}

// 32位字数组转字节数组（小端序）
void _words_to_bytes(const uint32_t* words, size_t words_len, uint8_t* data) {
    for (size_t i = 0; i < words_len; i++) {
        data[i * 4] = words[i] & 0xFF;
        data[i * 4 + 1] = (words[i] >> 8) & 0xFF;
        data[i * 4 + 2] = (words[i] >> 16) & 0xFF;
        data[i * 4 + 3] = (words[i] >> 24) & 0xFF;
    }
}

// XXTEA魔改加密
uint8_t* xxtea_encrypt(const uint8_t* data, size_t data_len, const uint8_t* key, size_t key_len, size_t* out_len) {
    if (!data || data_len == 0) {
        *out_len = 0;
        return NULL;
    }
    
    // 密钥处理：转换为4个32位字（小端序），不足16字节自动补零
    uint32_t key_words[4] = {0};
    for (size_t i = 0; i < 16 && i < key_len; i++) {
        key_words[i / 4] |= ((uint32_t)key[i]) << ((i % 4) * 8);
    }
    
    // 明文转换为字数组
    size_t words_len;
    uint32_t* v = (uint32_t*)malloc((data_len / 4 + 2) * sizeof(uint32_t));
    _bytes_to_words(data, data_len, v, &words_len);
    
    size_t n = words_len;
    // 魔改轮数：8 + 52/n（标准XXTEA为6 + 52/n）
    size_t rounds = 8 + 52 / n;
    
    uint32_t total = 0;
    
    for (size_t r = 0; r < rounds; r++) {
        // 累加和递增（使用负delta）
        total = (total + DELTA) & 0xFFFFFFFF;
        
        // 根据累加和高位计算密钥索引偏移
        uint32_t e = (total >> 2) & 3;
        
        if (r < 2) {
            // 前2轮使用标准MX函数进行初始扩散
            for (size_t p = 0; p < n; p++) {
                uint32_t y = v[(p + 1) % n];  // 后继字（循环取模）
                // 所有字统一循环处理，消除边界特殊处理
                v[p] = (v[p] + _mx(v[(p - 1 + n) % n], y, total, key_words, p, e)) & 0xFFFFFFFF;
            }
        } else {
            // 后续轮使用魔改MX函数增强混淆
            for (size_t p = 0; p < n; p++) {
                uint32_t y = v[(p + 1) % n];  // 后继字（循环取模）
                v[p] = (v[p] + _magic_mx(v[(p - 1 + n) % n], y, total, key_words, p, e)) & 0xFFFFFFFF;
            }
        }
    }
    
    // 输出结果
    *out_len = words_len * 4;
    uint8_t* result = (uint8_t*)malloc(*out_len);
    _words_to_bytes(v, words_len, result);
    
    free(v);
    return result;
}

// XXTEA魔改解密
uint8_t* xxtea_decrypt(const uint8_t* data, size_t data_len, const uint8_t* key, size_t key_len, size_t* out_len) {
    if (!data || data_len == 0) {
        *out_len = 0;
        return NULL;
    }
    
    // 对齐到4字节边界
    size_t aligned_len = data_len;
    if (data_len % 4) {
        aligned_len = data_len - (data_len % 4);
    }
    
    // 密钥处理同加密
    uint32_t key_words[4] = {0};
    for (size_t i = 0; i < 16 && i < key_len; i++) {
        key_words[i / 4] |= ((uint32_t)key[i]) << ((i % 4) * 8);
    }
    
    // 密文字节转字
    size_t words_len = aligned_len / 4;
    uint32_t* v = (uint32_t*)malloc(words_len * sizeof(uint32_t));
    
    for (size_t i = 0; i < words_len; i++) {
        v[i] = data[i * 4] | ((uint32_t)data[i * 4 + 1] << 8) | 
               ((uint32_t)data[i * 4 + 2] << 16) | ((uint32_t)data[i * 4 + 3] << 24);
    }
    
    size_t n = words_len;
    size_t rounds = 8 + 52 / n;
    
    // 解密时total从最大值开始递减
    uint32_t total = (DELTA * rounds) & 0xFFFFFFFF;
    
    for (size_t r = 0; r < rounds; r++) {
        uint32_t e = (total >> 2) & 3;
        
        if (r < rounds - 2) {
            // 对应加密的后n-2轮，使用魔改MX逆运算
            for (size_t p_idx = 0; p_idx < n; p_idx++) {
                size_t p = n - 1 - p_idx;  // 逆序遍历
                uint32_t y = v[(p + 1) % n];
                v[p] = (v[p] - _magic_mx(v[(p - 1 + n) % n], y, total, key_words, p, e)) & 0xFFFFFFFF;
            }
        } else {
            // 对应加密的前2轮，使用标准MX逆运算
            for (size_t p_idx = 0; p_idx < n; p_idx++) {
                size_t p = n - 1 - p_idx;  // 逆序遍历
                uint32_t y = v[(p + 1) % n];
                v[p] = (v[p] - _mx(v[(p - 1 + n) % n], y, total, key_words, p, e)) & 0xFFFFFFFF;
            }
        }
        
        // 累加和递减
        total = (total - DELTA) & 0xFFFFFFFF;
    }
    
    // 字转字节
    uint8_t* temp_result = (uint8_t*)malloc(n * 4);
    _words_to_bytes(v, n, temp_result);
    
    // 去除零填充
    size_t actual_len = n * 4;
    while (actual_len > 0 && temp_result[actual_len - 1] == 0) {
        actual_len--;
    }
    
    *out_len = actual_len;
    uint8_t* result = (uint8_t*)malloc(actual_len);
    memcpy(result, temp_result, actual_len);
    
    free(temp_result);
    free(v);
    return result;
}
