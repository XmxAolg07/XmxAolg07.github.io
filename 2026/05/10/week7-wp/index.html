    import struct

DELTA = 0x11451400 # DELTA常量
# 目标数据
enc_hex = "b4 b5 5a 42 a6 79 0b ac 0e 23 78 de e1 2d c6 1d bb 29 8c e2 94 fe 14 d9 aa 03 e3 8a 14 92 1c 64"
enc_bytes = bytes.fromhex(enc_hex.replace(" ", ""))
# 32 字节 → 8 个 DWORD（小端序，每个 DWORD 4 字节）
v = list(struct.unpack('<8I', enc_bytes))
# 密钥 从gdb调试中的 dword_40A090提取的
key = [11, 45, 14, 777]

def btea_decrypt(v, key):
    n = len(v) # DWORD 8个
    rounds = 16 #16轮
    sum_val = (rounds * DELTA) & 0xFFFFFFFF
    y = v[0]
    #外循环
    for _ in range(rounds):
        e = (sum_val >> 2) & 3
        # 内循环
        for p in range(n - 1, 0, -1):
            z = v[p - 1]
            # 核心运算
            mx = (((z >> 5) ^ (y << 2)) + ((y >> 3) ^ (z << 4))) ^ ((sum_val ^ y) + (key[(p & 3) ^ e] ^ z))
            v[p] = (v[p] - mx) & 0xFFFFFFFF
            y = v[p]
        z = v[n - 1]
        mx = (((z >> 5) ^ (y << 2)) + ((y >> 3) ^ (z << 4))) ^ ((sum_val ^ y) + (key[(0 & 3) ^ e] ^ z))
        v[0] = (v[0] - mx) & 0xFFFFFFFF
        y = v[0]
        sum_val = (sum_val - DELTA) & 0xFFFFFFFF
    return v
#执行解密
v = btea_decrypt(v, key)
result = b''
for val in v:
    result += struct.pack('<I', val)
#输出解密结果
print(result.decode('utf-8'))

完整加密流程

输入：两个 DWORD (a, b)

1. a ^= 0x125                    // 初始轮密钥加

2. 循环 16 轮：
   temp = a
   t0 = T0[a.byte2] + T1[a.byte3]
   t1 = T2[a.byte1]
   t2 = T3[a.byte0]
   a = (t2 + t1) ^ t0 ^ b       // 查表 + 异或
   b = temp                       // 轮间交换

3. b ^= 0x243F6BA8               // 最终异或

4. swap(a, b)                    // 输出交换

解密思路

输入：加密后的 (a, b)

1. swap(a, b)                    // 逆交换
2. b ^= 0x243F6BA8               // 逆最终异或
3. 循环 16 轮（逆序）：
   temp = b
   b = a
   a = temp ^ (T3[b.byte0] + T2[b.byte1]) ^ (T0[b.byte2] + T1[b.byte3])
4. a ^= 0x125                    // 逆初始异或

输出：原始 (a, b)

解密脚本

import struct
# 目标数据：从 GDB dump 的 byte_40A070（8 个 DWORD）
# 小端序读取
enc = [0x8DF0B770, 0x4B1EB28E, 0x73C65C1C, 0x53DE48FD,
       0x982EDF34, 0xD8229123, 0x90271182, 0x1F4271E7]

def read_box(filename):
    with open(filename, 'rb') as f:
        data = f.read()
    return [struct.unpack('<I', data[i:i+4])[0] for i in range(0, len(data), 4)]

box_BYTE2 = read_box('box1.bin')  # 0x40A500
box_BYTE3 = read_box('box2.bin')  # 0x40A100
box_BYTE1 = read_box('box3.bin')  # 0x40A900
box_BYTE0 = read_box('box4.bin')  # 0x40AD00
# 解密：4 对 DWORD，每对独立解密
for i in range(0, 8, 2):
    # 逆swap：加密最后做了交换，这里换回来
    enc[i], enc[i+1] = enc[i+1], enc[i]
    # 逆最终异或：加密最后 b ^= 0x243F6BA8
    enc[i+1] ^= 0x243F6BA8
    # 逆 16 轮：加密是正向 16 轮，解密逆序 16 轮
    for _ in range(16):
        tmp = enc[i]
        enc[i] = enc[i+1]
        enc[i+1] = ((box_BYTE0[enc[i] & 0xFF] + box_BYTE1[(enc[i] >> 8) & 0xFF]) ^
                    (box_BYTE2[(enc[i] >> 16) & 0xFF] + box_BYTE3[enc[i] >> 24]) ^ tmp) & 0xFFFFFFFF
        # 逆初始异或：加密最开始时 a ^= 0x125
    enc[i] ^= 0x125

result = b''
for val in enc:
    result += struct.pack('<I', val)
try:
    print(result.decode('utf-8'))
except:
    print(result.decode('latin-1'))
