AES算法和魔改

一、AES算法核心原理

AES（Advanced Encryption Standard）是目前应用最广泛的对称加密算法，本质是在有限域GF(2⁸)上进行的多轮置换-代换网络。

基本参数

参数 AES-128 AES-192 AES-256
密钥长度 16字节 24字节 32字节
分组长度 16字节 16字节 16字节
加密轮数 10轮 12轮 14轮

明文和密文都表示为4×4字节的State矩阵。

单轮加密的四个步骤

每轮（除最后一轮）依次执行以下操作：

SubBytes（字节代换）

  • 使用S盒对每个字节进行非线性替换。
  • S盒基于GF(2⁸)上的乘法逆元加仿射变换构造，是AES唯一的非线性部分。

ShiftRows（行移位）

  • State矩阵第0行不移，第1行左移1字节，第2行左移2字节，第3行左移3字节。

MixColumns（列混合）

  • 将State的每一列视为GF(2⁸)上的多项式，与固定多项式 3x³ + x² + x + 2 相乘（模 x⁴+1）。
  • 最后一轮省略此步骤。

AddRoundKey（轮密钥加）

  • State矩阵与轮密钥逐字节异或。

密钥扩展

从原始密钥扩展出各轮所需的轮密钥。使用RotWord、SubWord、Rcon等操作保证每轮密钥不同。

# AES轮常量数组：密钥扩展时使用的轮常量，每轮递增
rcon = [0x01, 0x02, 0x04, 0x08, 0x10, 0x20, 0x40, 0x80, 0x1B, 0x36]

# 字节代换（SubBytes）：使用S盒对状态的每个字节进行非线性替换
def sub_bytes(state):
    for i in range(16):
        state[i] = s_box[state[i]]

# 逆字节代换（InvSubBytes）：解密时使用逆S盒恢复原始字节
def inv_sub_bytes(state):
    for i in range(16):
        state[i] = inv_s_box[state[i]]

# 行移位（ShiftRows）：将状态矩阵的每一行循环左移不同字节数
# 第0行不移，第1行左移1，第2行左移2，第3行左移3
def shift_rows(state):
    # 第1行（索引1,5,9,13）左移1位
    state[1], state[5], state[9], state[13] = state[5], state[9], state[13], state[1]
    # 第2行（索引2,6,10,14）左移2位
    state[2], state[6], state[10], state[14] = state[10], state[14], state[2], state[6]
    # 第3行（索引3,7,11,15）左移3位
    state[3], state[7], state[11], state[15] = state[15], state[3], state[7], state[11]

# 逆行移位（InvShiftRows）：解密时将各行右移恢复原位
def inv_shift_rows(state):
    # 第1行右移1位（等价于左移3位）
    state[1], state[5], state[9], state[13] = state[13], state[1], state[5], state[9]
    # 第2行右移2位（等价于左移2位）
    state[2], state[6], state[10], state[14] = state[10], state[14], state[2], state[6]
    # 第3行右移3位（等价于左移1位）
    state[3], state[7], state[11], state[15] = state[7], state[11], state[15], state[3]

# GF(2^8)有限域乘法：AES列混合操作的核心运算
# 使用不可约多项式 x^8 + x^4 + x^3 + x + 1 (0x11B)
def gmul(a, b):
    p = 0
    # 循环处理b的每一位（共8位）
    for _ in range(8):
        # 如果b的当前最低位为1，将a异或累加到结果
        if b & 1:
            p ^= a
        # 检查a左移后是否会溢出（最高位是否为1）
        hi = a & 0x80
        # a左移1位，相当于乘以x
        a = (a << 1) & 0xFF
        # 如果溢出，需要模不可约多项式0x1B进行约简
        if hi:
            a ^= 0x1B
        # b右移1位，继续处理下一位
        b >>= 1
    return p

# 列混合（MixColumns）：将状态的每一列与固定矩阵相乘
# 使用矩阵 [2,3,1,1] 在GF(2^8)上进行乘法运算
def mix_columns(state):
    # 每次处理一列（4个字节）
    for i in range(0, 16, 4):
        s0, s1, s2, s3 = state[i], state[i+1], state[i+2], state[i+3]
        # 矩阵乘法：每行与列向量做点积
        state[i] = gmul(2, s0) ^ gmul(3, s1) ^ s2 ^ s3
        state[i+1] = s0 ^ gmul(2, s1) ^ gmul(3, s2) ^ s3
        state[i+2] = s0 ^ s1 ^ gmul(2, s2) ^ gmul(3, s3)
        state[i+3] = gmul(3, s0) ^ s1 ^ s2 ^ gmul(2, s3)

# 逆列混合（InvMixColumns）：使用逆矩阵 [14,11,13,9] 恢复原始列
def inv_mix_columns(state):
    for i in range(0, 16, 4):
        s0, s1, s2, s3 = state[i], state[i+1], state[i+2], state[i+3]
        # 逆矩阵乘法：使用9,11,13,14作为系数
        state[i] = gmul(14, s0) ^ gmul(11, s1) ^ gmul(13, s2) ^ gmul(9, s3)
        state[i+1] = gmul(9, s0) ^ gmul(14, s1) ^ gmul(11, s2) ^ gmul(13, s3)
        state[i+2] = gmul(13, s0) ^ gmul(9, s1) ^ gmul(14, s2) ^ gmul(11, s3)
        state[i+3] = gmul(11, s0) ^ gmul(13, s1) ^ gmul(9, s2) ^ gmul(14, s3)

# 轮密钥加（AddRoundKey）：状态与轮密钥逐字节异或
def add_round_key(state, round_key):
    for i in range(16):
        state[i] ^= round_key[i]

# 密钥扩展算法：从原始密钥生成各轮加密所需的轮密钥
def key_expansion(key):
    key_size = len(key)
    # 根据密钥长度确定参数
    # nk: 密钥字数（1字=4字节），nr: 加密轮数
    if key_size == 16:
        nk = 4
        nr = 10  # AES-128: 10轮
    elif key_size == 24:
        nk = 6
        nr = 12  # AES-192: 12轮
    else:
        nk = 8
        nr = 14  # AES-256: 14轮

    # w数组存储所有扩展密钥字
    w = []
    # 将原始密钥按4字节分组填入w数组的前nk个字
    for i in range(nk):
        w.append([key[4*i], key[4*i+1], key[4*i+2], key[4*i+3]])

    # 生成后续的密钥字
    for i in range(nk, 4 * (nr + 1)):
        # 取前一个密钥字作为临时值
        temp = w[i-1][:]
        if i % nk == 0:
            # 每nk个字进行一次特殊处理：
            # 1. RotWord: 循环左移1个字节
            temp = temp[1:] + temp[:1]
            # 2. SubWord: 使用S盒对每个字节进行替换
            for j in range(4):
                temp[j] = s_box[temp[j]]
            # 3. 与轮常量Rcon异或
            temp[0] ^= rcon[i // nk - 1]
        elif nk > 6 and i % nk == 4:
            # AES-256额外处理：每4个字做一次SubWord
            for j in range(4):
                temp[j] = s_box[temp[j]]

        # 与nk个字前的密钥字异或，得到新的密钥字
        w.append([w[i-nk][j] ^ temp[j] for j in range(4)])

    # 将w数组整理为轮密钥列表（每轮16字节）
    round_keys = []
    for i in range(nr + 1):
        rk = []
        for j in range(4):
            rk.extend(w[i*4+j])
        round_keys.append(rk)

    return round_keys

# 加密单个16字节（128位）明文分组
def aes_encrypt_block(plaintext, key):
    # 执行密钥扩展，获取所有轮密钥
    round_keys = key_expansion(key)
    # 将明文转换为可变的状态列表
    state = list(plaintext)

    # 初始轮密钥加
    add_round_key(state, round_keys[0])

    # 前nr-1轮：SubBytes → ShiftRows → MixColumns → AddRoundKey
    for round_num in range(1, len(round_keys) - 1):
        sub_bytes(state)
        shift_rows(state)
        mix_columns(state)
        add_round_key(state, round_keys[round_num])

    # 最后一轮：没有MixColumns
    sub_bytes(state)
    shift_rows(state)
    add_round_key(state, round_keys[-1])

    # 返回密文字节
    return bytes(state)

# 解密单个16字节密文分组
def aes_decrypt_block(ciphertext, key):
    # 使用相同的密钥扩展
    round_keys = key_expansion(key)
    state = list(ciphertext)

    # 逆向最后一轮
    add_round_key(state, round_keys[-1])
    inv_shift_rows(state)
    inv_sub_bytes(state)

    # 逆向第nr-1到第1轮：AddRoundKey → InvMixColumns → InvShiftRows → InvSubBytes
    for round_num in range(len(round_keys) - 2, 0, -1):
        add_round_key(state, round_keys[round_num])
        inv_mix_columns(state)
        inv_shift_rows(state)
        inv_sub_bytes(state)

    # 逆向初始轮密钥加
    add_round_key(state, round_keys[0])

    return bytes(state)

# PKCS7填充：将明文填充到block_size（16字节）的整数倍
def pkcs7_pad(data, block_size=16):
    # 计算需要填充的字节数（1到16之间）
    padding = block_size - len(data) % block_size
    # 每个填充字节的值等于填充长度
    return data + bytes([padding] * padding)

# PKCS7去填充：根据最后一个字节的值去除填充
def pkcs7_unpad(data):
    # 读取最后一个字节作为填充长度
    padding = data[-1]
    # 去掉末尾的填充字节
    return data[:-padding]

# ECB模式加密：逐块独立加密，相同明文块产生相同密文块
def aes_encrypt_ecb(plaintext, key):
    # 先对明文进行PKCS7填充
    plaintext = pkcs7_pad(plaintext)
    ciphertext = b""
    # 逐块加密，每块16字节
    for i in range(0, len(plaintext), 16):
        ciphertext += aes_encrypt_block(plaintext[i:i+16], key)
    return ciphertext

# ECB模式解密：逐块独立解密
def aes_decrypt_ecb(ciphertext, key):
    plaintext = b""
    for i in range(0, len(ciphertext), 16):
        plaintext += aes_decrypt_block(ciphertext[i:i+16], key)
    return pkcs7_unpad(plaintext)

# CBC模式加密：前一个密文块与当前明文块异或后再加密
def aes_encrypt_cbc(plaintext, key, iv):
    plaintext = pkcs7_pad(plaintext)
    ciphertext = b""
    # 初始化向量iv作为第一个"前密文块"
    prev = iv
    for i in range(0, len(plaintext), 16):
        # 明文块与前一个密文块异或
        block = bytes(x ^ y for x, y in zip(plaintext[i:i+16], prev))
        # 加密异或结果，作为下一个块的前密文块
        prev = aes_encrypt_block(block, key)
        ciphertext += prev
    return ciphertext

# CBC模式解密：解密后与前一个密文块异或得到明文
def aes_decrypt_cbc(ciphertext, key, iv):
    plaintext = b""
    prev = iv
    for i in range(0, len(ciphertext), 16):
        # 解密当前密文块
        block = aes_decrypt_block(ciphertext[i:i+16], key)
        # 与前一个密文块（或iv）异或得到明文
        plaintext += bytes(x ^ y for x, y in zip(block, prev))
        # 当前密文块成为下一个块的"前密文块"
        prev = ciphertext[i:i+16]
    return pkcs7_unpad(plaintext)

/* AES轮常量数组：密钥扩展时每轮使用的常量值 */

};

/* SubBytes：使用S盒对状态的每个字节进行非线性替换 */
static void sub_bytes(unsigned char state[16]) {
    for (int i = 0; i < 16; i++) {
        state[i] = s_box[state[i]];
    }
}

/* InvSubBytes：解密时使用逆S盒恢复原始字节值 */
static void inv_sub_bytes(unsigned char state[16]) {
    for (int i = 0; i < 16; i++) {
        state[i] = inv_s_box[state[i]];
    }
}

/* ShiftRows：将状态矩阵各行循环左移
   第0行不移，第1行左移1字节，第2行左移2字节，第3行左移3字节 */
static void shift_rows(unsigned char state[16]) {
    unsigned char temp;

    /* 第1行（索引1,5,9,13）循环左移1字节 */
    temp = state[1];
    state[1] = state[5];
    state[5] = state[9];
    state[9] = state[13];
    state[13] = temp;

    /* 第2行（索引2,6,10,14）循环左移2字节 */
    temp = state[2];
    state[2] = state[10];
    state[10] = temp;
    temp = state[6];
    state[6] = state[14];
    state[14] = temp;

    /* 第3行（索引3,7,11,15）循环左移3字节（等价于右移1字节） */
    temp = state[15];
    state[15] = state[11];
    state[11] = state[7];
    state[7] = state[3];
    state[3] = temp;
}

/* InvShiftRows：解密时将各行循环右移恢复原位 */
static void inv_shift_rows(unsigned char state[16]) {
    unsigned char temp;

    /* 第1行循环右移1字节（等价于左移3字节） */
    temp = state[13];
    state[13] = state[9];
    state[9] = state[5];
    state[5] = state[1];
    state[1] = temp;

    /* 第2行循环右移2字节（等价于左移2字节） */
    temp = state[2];
    state[2] = state[10];
    state[10] = temp;
    temp = state[6];
    state[6] = state[14];
    state[14] = temp;

    /* 第3行循环右移3字节（等价于左移1字节） */
    temp = state[3];
    state[3] = state[7];
    state[7] = state[11];
    state[11] = state[15];
    state[15] = temp;
}

/* GF(2^8)有限域乘法：AES列混合操作的核心运算
   使用不可约多项式 x^8 + x^4 + x^3 + x + 1 (0x11B) */
static unsigned char gmul(unsigned char a, unsigned char b) {
    unsigned char p = 0;
    /* 逐位处理b的每一位（共8位） */
    for (int i = 0; i < 8; i++) {
        /* 如果b的最低位为1，将a异或累加到结果 */
        if (b & 1) {
            p ^= a;
        }
        /* 检查a左移后是否会溢出（最高位是否为1） */
        unsigned char hi = a & 0x80;
        /* a左移1位，相当于乘以x */
        a <<= 1;
        /* 如果溢出，需要模不可约多项式0x1B进行约简 */
        if (hi) {
            a ^= 0x1B;
        }
        /* b右移1位，继续处理下一位 */
        b >>= 1;
    }
    return p;
}

/* MixColumns：将状态每列与固定矩阵 [2,3,1,1] 在GF(2^8)上相乘 */
static void mix_columns(unsigned char state[16]) {
    for (int i = 0; i < 16; i += 4) {
        /* 取出当前列的4个字节 */
        unsigned char s0 = state[i];
        unsigned char s1 = state[i+1];
        unsigned char s2 = state[i+2];
        unsigned char s3 = state[i+3];

        /* 矩阵乘法：每行与列向量做点积 */
        state[i]   = gmul(2, s0) ^ gmul(3, s1) ^ s2 ^ s3;
        state[i+1] = s0 ^ gmul(2, s1) ^ gmul(3, s2) ^ s3;
        state[i+2] = s0 ^ s1 ^ gmul(2, s2) ^ gmul(3, s3);
        state[i+3] = gmul(3, s0) ^ s1 ^ s2 ^ gmul(2, s3);
    }
}

/* InvMixColumns：使用逆矩阵 [14,11,13,9] 恢复原始列 */
static void inv_mix_columns(unsigned char state[16]) {
    for (int i = 0; i < 16; i += 4) {
        unsigned char s0 = state[i];
        unsigned char s1 = state[i+1];
        unsigned char s2 = state[i+2];
        unsigned char s3 = state[i+3];

        /* 逆矩阵乘法：使用9,11,13,14作为系数 */
        state[i]   = gmul(14, s0) ^ gmul(11, s1) ^ gmul(13, s2) ^ gmul(9, s3);
        state[i+1] = gmul(9, s0) ^ gmul(14, s1) ^ gmul(11, s2) ^ gmul(13, s3);
        state[i+2] = gmul(13, s0) ^ gmul(9, s1) ^ gmul(14, s2) ^ gmul(11, s3);
        state[i+3] = gmul(11, s0) ^ gmul(13, s1) ^ gmul(9, s2) ^ gmul(14, s3);
    }
}

/* AddRoundKey：状态与轮密钥逐字节异或 */
static void add_round_key(unsigned char state[16], const unsigned char round_key[16]) {
    for (int i = 0; i < 16; i++) {
        state[i] ^= round_key[i];
    }
}

/* 密钥扩展算法：从原始密钥生成各轮所需的轮密钥
   输入：key-原始密钥，key_size-密钥长度
   输出：round_keys-轮密钥数组，nr-加密轮数 */
static void key_expansion(const unsigned char *key, int key_size,
                          unsigned char round_keys[][16], int *nr) {
    int nk, nk_words;
    /* 根据密钥长度确定参数 */
    if (key_size == 16) {
        nk = 4;        /* 密钥字数 */
        nk_words = 4;
        *nr = 10;      /* AES-128: 10轮 */
    } else if (key_size == 24) {
        nk = 6;
        nk_words = 6;
        *nr = 12;      /* AES-192: 12轮 */
    } else {
        nk = 8;
        nk_words = 8;
        *nr = 14;      /* AES-256: 14轮 */
    }

    /* w数组存储所有扩展密钥字（每字4字节） */
    unsigned char w[4 * 15][4];

    /* 将原始密钥按4字节分组填入w数组的前nk个字 */
    for (int i = 0; i < nk_words; i++) {
        w[i][0] = key[4*i];
        w[i][1] = key[4*i+1];
        w[i][2] = key[4*i+2];
        w[i][3] = key[4*i+3];
    }

    /* 生成后续的密钥字 */
    for (int i = nk_words; i < 4 * (*nr + 1); i++) {
        unsigned char temp[4];
        /* 取前一个密钥字作为临时值 */
        temp[0] = w[i-1][0];
        temp[1] = w[i-1][1];
        temp[2] = w[i-1][2];
        temp[3] = w[i-1][3];

        if (i % nk_words == 0) {
            /* 每nk个字进行一次特殊处理 */
            /* 1. RotWord: 循环左移1个字节 */
            unsigned char t = temp[0];
            temp[0] = temp[1];
            temp[1] = temp[2];
            temp[2] = temp[3];
            temp[3] = t;

            /* 2. SubWord: 使用S盒对每个字节进行替换 */
            for (int j = 0; j < 4; j++) {
                temp[j] = s_box[temp[j]];
            }
            /* 3. 与轮常量Rcon异或 */
            temp[0] ^= rcon[i / nk_words - 1];
        } else if (nk_words > 6 && i % nk_words == 4) {
            /* AES-256额外处理：每4个字做一次SubWord */
            for (int j = 0; j < 4; j++) {
                temp[j] = s_box[temp[j]];
            }
        }

        /* 与nk个字前的密钥字异或，得到新的密钥字 */
        w[i][0] = w[i - nk_words][0] ^ temp[0];
        w[i][1] = w[i - nk_words][1] ^ temp[1];
        w[i][2] = w[i - nk_words][2] ^ temp[2];
        w[i][3] = w[i - nk_words][3] ^ temp[3];
    }

    /* 将w数组整理为轮密钥列表（每轮16字节） */
    for (int i = 0; i <= *nr; i++) {
        for (int j = 0; j < 4; j++) {
            round_keys[i][4*j]   = w[i*4+j][0];
            round_keys[i][4*j+1] = w[i*4+j][1];
            round_keys[i][4*j+2] = w[i*4+j][2];
            round_keys[i][4*j+3] = w[i*4+j][3];
        }
    }
}

/* 加密单个16字节（128位）明文分组 */
static void aes_encrypt_block(const unsigned char plaintext[16],
                              const unsigned char *key, int key_size,
                              unsigned char ciphertext[16]) {
    int nr;
    unsigned char round_keys[15][16];
    unsigned char state[16];

    /* 执行密钥扩展，获取所有轮密钥 */
    key_expansion(key, key_size, round_keys, &nr);
    /* 将明文复制到状态数组 */
    memcpy(state, plaintext, 16);

    /* 初始轮密钥加 */
    add_round_key(state, round_keys[0]);

    /* 前nr-1轮：SubBytes → ShiftRows → MixColumns → AddRoundKey */
    for (int round = 1; round < nr; round++) {
        sub_bytes(state);
        shift_rows(state);
        mix_columns(state);
        add_round_key(state, round_keys[round]);
    }

    /* 最后一轮：没有MixColumns操作 */
    sub_bytes(state);
    shift_rows(state);
    add_round_key(state, round_keys[nr]);

    /* 将状态复制到密文输出 */
    memcpy(ciphertext, state, 16);
}

/* 解密单个16字节密文分组 */
static void aes_decrypt_block(const unsigned char ciphertext[16],
                              const unsigned char *key, int key_size,
                              unsigned char plaintext[16]) {
    int nr;
    unsigned char round_keys[15][16];
    unsigned char state[16];

    /* 使用相同的密钥扩展 */
    key_expansion(key, key_size, round_keys, &nr);
    memcpy(state, ciphertext, 16);

    /* 逆向最后一轮（先AddRoundKey，然后逆ShiftRows和逆SubBytes） */
    add_round_key(state, round_keys[nr]);
    inv_shift_rows(state);
    inv_sub_bytes(state);

    /* 逆向第nr-1到第1轮：AddRoundKey → InvMixColumns → InvShiftRows → InvSubBytes */
    for (int round = nr - 1; round > 0; round--) {
        add_round_key(state, round_keys[round]);
        inv_mix_columns(state);
        inv_shift_rows(state);
        inv_sub_bytes(state);
    }

    /* 逆向初始轮密钥加 */
    add_round_key(state, round_keys[0]);

    memcpy(plaintext, state, 16);
}

/* PKCS7填充：将数据填充到block_size（16字节）的整数倍
   填充的每个字节值等于填充长度 */
static int pkcs7_pad(unsigned char *data, int len, int block_size) {
    int padding = block_size - len % block_size;
    for (int i = 0; i < padding; i++) {
        data[len + i] = padding;
    }
    return len + padding;
}

/* PKCS7去填充：根据最后一个字节的值去除填充 */
static int pkcs7_unpad(unsigned char *data, int len) {
    int padding = data[len - 1];
    return len - padding;
}

/* ECB模式加密：逐块独立加密，相同明文块产生相同密文块 */
static void aes_encrypt_ecb(const unsigned char *plaintext, int plaintext_len,
                            const unsigned char *key, int key_size,
                            unsigned char *ciphertext, int *ciphertext_len) {
    unsigned char padded[512];
    /* 复制明文并填充 */
    memcpy(padded, plaintext, plaintext_len);
    int padded_len = pkcs7_pad(padded, plaintext_len, 16);

    *ciphertext_len = 0;
    /* 逐块加密，每块16字节 */
    for (int i = 0; i < padded_len; i += 16) {
        aes_encrypt_block(padded + i, key, key_size, ciphertext + *ciphertext_len);
        *ciphertext_len += 16;
    }
}

/* ECB模式解密：逐块独立解密 */
static void aes_decrypt_ecb(const unsigned char *ciphertext, int ciphertext_len,
                            const unsigned char *key, int key_size,
                            unsigned char *plaintext, int *plaintext_len) {
    *plaintext_len = 0;
    for (int i = 0; i < ciphertext_len; i += 16) {
        aes_decrypt_block(ciphertext + i, key, key_size, plaintext + *plaintext_len);
        *plaintext_len += 16;
    }
    /* 去除PKCS7填充 */
    *plaintext_len = pkcs7_unpad(plaintext, *plaintext_len);
}

/* CBC模式加密：前一个密文块与当前明文块异或后再加密
   需要初始化向量IV作为第一个"前密文块" */
static void aes_encrypt_cbc(const unsigned char *plaintext, int plaintext_len,
                            const unsigned char *key, int key_size,
                            const unsigned char iv[16],
                            unsigned char *ciphertext, int *ciphertext_len) {
    unsigned char padded[512];
    memcpy(padded, plaintext, plaintext_len);
    int padded_len = pkcs7_pad(padded, plaintext_len, 16);

    unsigned char prev[16];
    unsigned char block[16];
    /* 初始化向量IV作为第一个前密文块 */
    memcpy(prev, iv, 16);

    *ciphertext_len = 0;
    for (int i = 0; i < padded_len; i += 16) {
        /* 明文块与前一个密文块（或IV）逐字节异或 */
        for (int j = 0; j < 16; j++) {
            block[j] = padded[i+j] ^ prev[j];
        }
        /* 加密异或结果，并将密文作为下一个块的前密文块 */
        aes_encrypt_block(block, key, key_size, prev);
        memcpy(ciphertext + *ciphertext_len, prev, 16);
        *ciphertext_len += 16;
    }
}

/* CBC模式解密：解密后与前一个密文块异或得到明文 */
static void aes_decrypt_cbc(const unsigned char *ciphertext, int ciphertext_len,
                            const unsigned char *key, int key_size,
                            const unsigned char iv[16],
                            unsigned char *plaintext, int *plaintext_len) {
    unsigned char prev[16];
    unsigned char block[16];
    /* 初始化向量IV作为第一个前密文块 */
    memcpy(prev, iv, 16);

    *plaintext_len = 0;
    for (int i = 0; i < ciphertext_len; i += 16) {
        /* 解密当前密文块 */
        aes_decrypt_block(ciphertext + i, key, key_size, block);
        /* 解密结果与前一个密文块（或IV）异或得到明文 */
        for (int j = 0; j < 16; j++) {
            plaintext[*plaintext_len + j] = block[j] ^ prev[j];
        }
        /* 当前密文块成为下一个块的前密文块 */
        memcpy(prev, ciphertext + i, 16);
        *plaintext_len += 16;
    }
    *plaintext_len = pkcs7_unpad(plaintext, *plaintext_len);
}

static void print_hex(const unsigned char *data, int len) {
    for (int i = 0; i < len; i++) {
        printf("%02x", data[i]);
    }
    printf("\n");
}

AES的魔改

CTF中的AES题目通常不会使用标准算法，而是进行各种魔改来增加逆向难度。

常见魔改方式

  1. S盒魔改 - 替换自定义S盒

  2. 轮数修改 - 8轮AES-128比较常见

  3. ShiftRows修改 - 改变行移位偏移量

  4. MixColumns修改 - 使用不同的列混合矩阵

  5. 密钥扩展魔改 - 修改RotWord、Rcon或SubWord

  6. 额外变换 - 在首尾添加异或/置换操作

# GF(2^8)有限域乘法运算
def gmul(a, b):
    p = 0
    # 循环8次，逐位处理乘数b的每一位
    for _ in range(8):
        # 如果b的最低位为1，将a累加到结果p中
        if b & 1:
            p ^= a
        # 检查a的最高位是否为1（即a >= 0x80）
        hi = a & 0x80
        # a左移1位，相当于乘以x
        a = (a << 1) & 0xFF
        # 如果a溢出（最高位为1），需要模不可约多项式0x1B
        if hi:
            a ^= 0x1B
        # b右移1位，处理下一位
        b >>= 1
    return p

# 轮密钥加操作：状态数组与轮密钥逐字节异或
def add_round_key(state, round_key):
    for i in range(16):
        state[i] ^= round_key[i]

# PKCS7填充：将数据填充到block_size的整数倍
def pkcs7_pad(data, block_size=16):
    # 计算需要填充的字节数
    padding = block_size - len(data) % block_size
    # 每个填充字节的值等于填充长度
    return data + bytes([padding] * padding)

# PKCS7去填充：去除PKCS7填充的数据
def pkcs7_unpad(data):
    # 最后一个字节的值就是填充长度
    padding = data[-1]
    # 返回去掉填充后的数据
    return data[:-padding]


# 魔改AES类型2：线性S盒 + 简化密钥扩展 + 8轮 + 末尾异或
class ModifiedAES_Type2:
    def __init__(self, key):
        self.key = key
        # 生成线性S盒：S[i] = (i * 7 + 13) mod 256
        # 这种线性变换没有GF逆元，安全性较弱但可逆
        self.xor_sbox = [((i * 7 + 13) & 0xFF) for i in range(256)]
        # 生成逆S盒：通过遍历找到每个值对应的原始索引
        self.xor_inv_sbox = [0] * 256
        for i in range(256):
            self.xor_inv_sbox[self.xor_sbox[i]] = i
        # 执行密钥扩展，生成轮密钥
        self.round_keys = self._key_expansion()

    # 字节代换（SubBytes）：使用自定义线性S盒
    def _sub_bytes(self, state):
        for i in range(16):
            state[i] = self.xor_sbox[state[i]]

    # 逆字节代换（InvSubBytes）：使用逆S盒还原
    def _inv_sub_bytes(self, state):
        for i in range(16):
            state[i] = self.xor_inv_sbox[state[i]]

    # 行移位（ShiftRows）：标准AES的行移位操作
    # 第1行左移1字节，第2行左移2字节，第3行左移3字节
    def _shift_rows(self, state):
        state[1], state[5], state[9], state[13] = state[5], state[9], state[13], state[1]
        state[2], state[6], state[10], state[14] = state[10], state[14], state[2], state[6]
        state[3], state[7], state[11], state[15] = state[15], state[3], state[7], state[11]

    # 逆行移位（InvShiftRows）：标准AES的逆行移位操作
    # 第1行右移1字节，第2行右移2字节，第3行右移3字节
    def _inv_shift_rows(self, state):
        state[1], state[5], state[9], state[13] = state[13], state[1], state[5], state[9]
        state[2], state[6], state[10], state[14] = state[10], state[14], state[2], state[6]
        state[3], state[7], state[11], state[15] = state[7], state[11], state[15], state[3]

    # 列混合（MixColumns）：标准AES的列混合操作
    # 使用矩阵 [2,3,1,1] 在GF(2^8)上进行乘法运算
    def _mix_columns(self, state):
        for i in range(0, 16, 4):
            # 取出当前列的4个字节
            s0, s1, s2, s3 = state[i], state[i+1], state[i+2], state[i+3]
            # 矩阵乘法：每行与列向量做点积
            state[i]   = gmul(2, s0) ^ gmul(3, s1) ^ s2 ^ s3
            state[i+1] = s0 ^ gmul(2, s1) ^ gmul(3, s2) ^ s3
            state[i+2] = s0 ^ s1 ^ gmul(2, s2) ^ gmul(3, s3)
            state[i+3] = gmul(3, s0) ^ s1 ^ s2 ^ gmul(2, s3)

    # 逆列混合（InvMixColumns）：使用逆矩阵 [14,11,13,9]
    def _inv_mix_columns(self, state):
        for i in range(0, 16, 4):
            s0, s1, s2, s3 = state[i], state[i+1], state[i+2], state[i+3]
            state[i]   = gmul(14, s0) ^ gmul(11, s1) ^ gmul(13, s2) ^ gmul(9, s3)
            state[i+1] = gmul(9, s0) ^ gmul(14, s1) ^ gmul(11, s2) ^ gmul(13, s3)
            state[i+2] = gmul(13, s0) ^ gmul(9, s1) ^ gmul(14, s2) ^ gmul(11, s3)
            state[i+3] = gmul(11, s0) ^ gmul(13, s1) ^ gmul(9, s2) ^ gmul(14, s3)

    # 密钥扩展：生成8轮加密所需的轮密钥
    def _key_expansion(self):
        key_size = len(self.key)
        # 根据密钥长度确定参数（nk: 密钥字数，nr: 轮数）
        if key_size == 16:
            nk = 4
            nr = 8  # 魔改点：标准AES-128为10轮，这里减少为8轮
        elif key_size == 24:
            nk = 6
            nr = 10
        else:
            nk = 8
            nr = 12

        w = []
        # 将密钥按4字节分组填入初始w数组
        for i in range(nk):
            w.append([self.key[4*i], self.key[4*i+1], self.key[4*i+2], self.key[4*i+3]])

        # 魔改点：简化的密钥扩展算法
        for i in range(nk, 4 * (nr + 1)):
            temp = w[i-1][:]
            if i % nk == 0:
                # 魔改点：每nk个字，不是RotWord+SubWord+Rcon，而是直接异或0xAA
                for j in range(4):
                    temp[j] ^= 0xAA
            elif nk > 6 and i % nk == 4:
                # 对于AES-256，每4个字额外做一次S盒替换（与标准AES一致）
                for j in range(4):
                    temp[j] = self.xor_sbox[temp[j]]

            # 与nk个字前的值异或得到新的4字节密钥字
            w.append([w[i-nk][j] ^ temp[j] for j in range(4)])

        # 将w数组整理为round_keys格式（每轮16字节）
        round_keys = []
        for i in range(nr + 1):
            rk = []
            for j in range(4):
                rk.extend(w[i*4+j])
            round_keys.append(rk)

        return round_keys

    # 加密单个16字节分组
    def encrypt_block(self, plaintext):
        state = list(plaintext)
        # 初始轮密钥加
        add_round_key(state, self.round_keys[0])

        # 前7轮：SubBytes → ShiftRows → MixColumns → AddRoundKey
        for round_num in range(1, 8):
            self._sub_bytes(state)
            self._shift_rows(state)
            self._mix_columns(state)
            add_round_key(state, self.round_keys[round_num])

        # 最后一轮（第8轮）：没有MixColumns
        self._sub_bytes(state)
        self._shift_rows(state)
        add_round_key(state, self.round_keys[8])

        # 魔改点：末尾额外异或0x5A
        for i in range(16):
            state[i] ^= 0x5A

        return bytes(state)

    # 解密单个16字节分组
    def decrypt_block(self, ciphertext):
        state = list(ciphertext)

        # 魔改点：先撤销末尾的异或0x5A
        for i in range(16):
            state[i] ^= 0x5A

        # 逆向最后一轮
        add_round_key(state, self.round_keys[8])
        self._inv_shift_rows(state)
        self._inv_sub_bytes(state)

        # 逆向第7到第1轮：AddRoundKey → InvMixColumns → InvShiftRows → InvSubBytes
        for round_num in range(7, 0, -1):
            add_round_key(state, self.round_keys[round_num])
            self._inv_mix_columns(state)
            self._inv_shift_rows(state)
            self._inv_sub_bytes(state)

        # 初始轮密钥加
        add_round_key(state, self.round_keys[0])
        return bytes(state)

    # ECB模式加密
    def encrypt_ecb(self, plaintext):
        plaintext = pkcs7_pad(plaintext)
        ciphertext = b""
        # 逐块加密
        for i in range(0, len(plaintext), 16):
            ciphertext += self.encrypt_block(plaintext[i:i+16])
        return ciphertext

    # ECB模式解密
    def decrypt_ecb(self, ciphertext):
        plaintext = b""
        # 逐块解密
        for i in range(0, len(ciphertext), 16):
            plaintext += self.decrypt_block(ciphertext[i:i+16])
        return pkcs7_unpad(plaintext)


# 魔改AES类型3：位反转S盒 + 反转RotWord + 额外置换层
class ModifiedAES_Type3:
    def __init__(self, key):
        self.key = key
        # 生成位反转S盒：先将i的8位二进制反转，再异或0x63
        self.custom_sbox = [0] * 256
        for i in range(256):
            # 位反转：将bit0与bit7交换，bit1与bit6交换，依此类推
            inv = 0
            for j in range(8):
                if (i >> j) & 1:
                    inv |= (1 << (7 - j))
            self.custom_sbox[i] = inv ^ 0x63
        # 生成逆S盒
        self.custom_inv_sbox = [0] * 256
        for i in range(256):
            self.custom_inv_sbox[self.custom_sbox[i]] = i
        # 执行密钥扩展
        self.round_keys = self._key_expansion()

    # 字节代换（SubBytes）：使用位反转S盒
    def _sub_bytes(self, state):
        for i in range(16):
            state[i] = self.custom_sbox[state[i]]

    # 逆字节代换（InvSubBytes）：使用逆S盒
    def _inv_sub_bytes(self, state):
        for i in range(16):
            state[i] = self.custom_inv_sbox[state[i]]

    # 行移位（ShiftRows）：标准操作
    def _shift_rows(self, state):
        state[1], state[5], state[9], state[13] = state[5], state[9], state[13], state[1]
        state[2], state[6], state[10], state[14] = state[10], state[14], state[2], state[6]
        state[3], state[7], state[11], state[15] = state[15], state[3], state[7], state[11]

    # 逆行移位（InvShiftRows）：标准操作
    def _inv_shift_rows(self, state):
        state[1], state[5], state[9], state[13] = state[13], state[1], state[5], state[9]
        state[2], state[6], state[10], state[14] = state[10], state[14], state[2], state[6]
        state[3], state[7], state[11], state[15] = state[7], state[11], state[15], state[3]

    # 列混合（MixColumns）：标准操作
    def _mix_columns(self, state):
        for i in range(0, 16, 4):
            s0, s1, s2, s3 = state[i], state[i+1], state[i+2], state[i+3]
            state[i]   = gmul(2, s0) ^ gmul(3, s1) ^ s2 ^ s3
            state[i+1] = s0 ^ gmul(2, s1) ^ gmul(3, s2) ^ s3
            state[i+2] = s0 ^ s1 ^ gmul(2, s2) ^ gmul(3, s3)
            state[i+3] = gmul(3, s0) ^ s1 ^ s2 ^ gmul(2, s3)

    # 逆列混合（InvMixColumns）：标准操作
    def _inv_mix_columns(self, state):
        for i in range(0, 16, 4):
            s0, s1, s2, s3 = state[i], state[i+1], state[i+2], state[i+3]
            state[i]   = gmul(14, s0) ^ gmul(11, s1) ^ gmul(13, s2) ^ gmul(9, s3)
            state[i+1] = gmul(9, s0) ^ gmul(14, s1) ^ gmul(11, s2) ^ gmul(13, s3)
            state[i+2] = gmul(13, s0) ^ gmul(9, s1) ^ gmul(14, s2) ^ gmul(11, s3)
            state[i+3] = gmul(11, s0) ^ gmul(13, s1) ^ gmul(9, s2) ^ gmul(14, s3)

    # 密钥扩展：标准10轮
    def _key_expansion(self):
        key_size = len(self.key)
        if key_size == 16:
            nk = 4
            nr = 10
        elif key_size == 24:
            nk = 6
            nr = 12
        else:
            nk = 8
            nr = 14

        w = []
        for i in range(nk):
            w.append([self.key[4*i], self.key[4*i+1], self.key[4*i+2], self.key[4*i+3]])

        for i in range(nk, 4 * (nr + 1)):
            temp = w[i-1][:]
            if i % nk == 0:
                # 魔改点：RotWord不是循环左移1位，而是完全反转（reverse）
                temp = temp[::-1]
                for j in range(4):
                    temp[j] = self.custom_sbox[temp[j]]
                # Rcon异或与标准AES一致
                temp[0] ^= rcon[i // nk - 1]
            elif nk > 6 and i % nk == 4:
                for j in range(4):
                    temp[j] = self.custom_sbox[temp[j]]

            w.append([w[i-nk][j] ^ temp[j] for j in range(4)])

        round_keys = []
        for i in range(nr + 1):
            rk = []
            for j in range(4):
                rk.extend(w[i*4+j])
            round_keys.append(rk)

        return round_keys

    # 魔改点：额外添加的置换层，将状态字节按公式 (i*3+7) mod 16 重新排列
    def _permute_state(self, state):
        temp = [0] * 16
        for i in range(16):
            temp[(i * 3 + 7) % 16] = state[i]
        for i in range(16):
            state[i] = temp[i]

    # 逆置换层：计算逆映射，将置换还原
    def _inv_permute_state(self, state):
        temp = [0] * 16
        # 构建逆置换映射表
        inv_perm = [0] * 16
        for i in range(16):
            inv_perm[(i * 3 + 7) % 16] = i
        # 使用逆映射还原状态
        for i in range(16):
            temp[inv_perm[i]] = state[i]
        for i in range(16):
            state[i] = temp[i]

    # 加密单个16字节分组
    def encrypt_block(self, plaintext):
        state = list(plaintext)
        # 魔改点：在加密开始前先进行置换
        self._permute_state(state)
        add_round_key(state, self.round_keys[0])

        # 前9轮：SubBytes → ShiftRows → MixColumns → AddRoundKey
        for round_num in range(1, 10):
            self._sub_bytes(state)
            self._shift_rows(state)
            self._mix_columns(state)
            add_round_key(state, self.round_keys[round_num])

        # 最后一轮：没有MixColumns
        self._sub_bytes(state)
        self._shift_rows(state)
        add_round_key(state, self.round_keys[10])
        # 魔改点：在加密结束后再次进行置换
        self._permute_state(state)

        return bytes(state)

    # 解密单个16字节分组
    def decrypt_block(self, ciphertext):
        state = list(ciphertext)
        # 魔改点：先逆向最后的置换
        self._inv_permute_state(state)
        add_round_key(state, self.round_keys[10])
        self._inv_shift_rows(state)
        self._inv_sub_bytes(state)

        # 逆向第9到第1轮
        for round_num in range(9, 0, -1):
            add_round_key(state, self.round_keys[round_num])
            self._inv_mix_columns(state)
            self._inv_shift_rows(state)
            self._inv_sub_bytes(state)

        add_round_key(state, self.round_keys[0])
        # 魔改点：逆向开始的置换
        self._inv_permute_state(state)

        return bytes(state)

    # ECB模式加密
    def encrypt_ecb(self, plaintext):

        ciphertext = b""
        for i in range(0, len(plaintext), 16):

        for i in range(0, len(ciphertext), 16):
            plaintext += self.decrypt_block(ciphertext[i:i+16])

/* GF(2^8)有限域乘法：AES列混合操作的核心运算
   使用不可约多项式 x^8 + x^4 + x^3 + x + 1 (0x11B) */
static unsigned char gmul(unsigned char a, unsigned char b) {
    unsigned char p = 0;
    for (int i = 0; i < 8; i++) {
        if (b & 1) p ^= a;
        unsigned char hi = a & 0x80;
        a <<= 1;
        if (hi) a ^= 0x1B;
        b >>= 1;
    }
    return p;
}

/* AddRoundKey：状态与轮密钥逐字节异或 */
static void add_round_key(unsigned char state[16], const unsigned char round_key[16]) {
    for (int i = 0; i < 16; i++) {
        state[i] ^= round_key[i];
    }
}

/* PKCS7填充：将数据填充到block_size的整数倍
   填充的每个字节值等于填充长度 */
static void pkcs7_pad(unsigned char *data, int *len, int block_size) {
    int padding = block_size - (*len) % block_size;
    for (int i = 0; i < padding; i++) {
        data[*len + i] = padding;
    }
    *len += padding;
}

/* PKCS7去填充：根据最后一个字节的值去除填充 */
static int pkcs7_unpad(unsigned char *data, int len) {
    int padding = data[len - 1];
    return len - padding;
}

}

/* Type2密钥扩展：魔改的简化版本
   将RotWord+SubWord+Rcon替换为直接异或0xAA，轮数固定为8 */
static void type2_key_expansion(ModifiedAES_Type2 *aes, const unsigned char *key, int key_size) {
    int nk = (key_size == 16) ? 4 : (key_size == 24) ? 6 : 8;

        w[i][0] = key[4*i]; w[i][1] = key[4*i+1];
        w[i][2] = key[4*i+2]; w[i][3] = key[4*i+3];
    }

    for (int i = nk_words; i < 4 * (aes->nr + 1); i++) {
        unsigned char temp[4];
        temp[0] = w[i-1][0]; temp[1] = w[i-1][1];
        temp[2] = w[i-1][2]; temp[3] = w[i-1][3];

        if (i % nk_words == 0) {
            /* 魔改点：标准AES此处为RotWord+SubWord+Rcon，改为直接异或0xAA */
            for (int j = 0; j < 4; j++) {
                temp[j] ^= 0xAA;
            }
        } else if (nk_words > 6 && i % nk_words == 4) {
            for (int j = 0; j < 4; j++) {
                temp[j] = aes->xor_sbox[temp[j]];
            }
        }

        w[i][0] = w[i - nk_words][0] ^ temp[0];
        w[i][1] = w[i - nk_words][1] ^ temp[1];
        w[i][2] = w[i - nk_words][2] ^ temp[2];
        w[i][3] = w[i - nk_words][3] ^ temp[3];
    }

    for (int i = 0; i <= aes->nr; i++) {
        for (int j = 0; j < 4; j++) {
            aes->round_keys[i][4*j]   = w[i*4+j][0];
            aes->round_keys[i][4*j+1] = w[i*4+j][1];
            aes->round_keys[i][4*j+2] = w[i*4+j][2];
            aes->round_keys[i][4*j+3] = w[i*4+j][3];
        }
    }
}

/* Type2加密单个分组：8轮加密 + 末尾异或0x5A */
static void type2_encrypt_block(ModifiedAES_Type2 *aes, const unsigned char plaintext[16], unsigned char ciphertext[16]) {
    unsigned char state[16];
    memcpy(state, plaintext, 16);

    add_round_key(state, aes->round_keys[0]);

    for (int round = 1; round < 8; round++) {
        type2_sub_bytes(aes, state);
        type2_shift_rows(state);
        type2_mix_columns(state);
        add_round_key(state, aes->round_keys[round]);
    }

    /* 最后一轮没有MixColumns */
    type2_sub_bytes(aes, state);
    type2_shift_rows(state);
    add_round_key(state, aes->round_keys[8]);

    /* 魔改点：末尾额外异或0x5A */
    for (int i = 0; i < 16; i++) {
        state[i] ^= 0x5A;
    }

    memcpy(ciphertext, state, 16);
}

/* Type2解密单个分组：先撤末尾异或，然后逆序执行各轮操作 */
static void type2_decrypt_block(ModifiedAES_Type2 *aes, const unsigned char ciphertext[16], unsigned char plaintext[16]) {
    unsigned char state[16];
    memcpy(state, ciphertext, 16);

    /* 先撤销末尾的异或0x5A */
    for (int i = 0; i < 16; i++) {
        state[i] ^= 0x5A;
    }

    add_round_key(state, aes->round_keys[8]);
    type2_inv_shift_rows(state);
    type2_inv_sub_bytes(aes, state);

    for (int round = 7; round > 0; round--) {
        add_round_key(state, aes->round_keys[round]);
        type2_inv_mix_columns(state);
        type2_inv_shift_rows(state);
        type2_inv_sub_bytes(aes, state);
    }

    add_round_key(state, aes->round_keys[0]);

    memcpy(plaintext, state, 16);
}

/* Type2 ECB模式加密 */
static void type2_encrypt_ecb(ModifiedAES_Type2 *aes, const unsigned char *plaintext, int plaintext_len,
                               unsigned char *ciphertext, int *ciphertext_len) {
    unsigned char padded[512];
    memcpy(padded, plaintext, plaintext_len);
    int padded_len = plaintext_len;
    pkcs7_pad(padded, &padded_len, 16);

    *ciphertext_len = 0;
    for (int i = 0; i < padded_len; i += 16) {
        type2_encrypt_block(aes, padded + i, ciphertext + *ciphertext_len);
        *ciphertext_len += 16;
    }
}

/* Type2 ECB模式解密 */
static void type2_decrypt_ecb(ModifiedAES_Type2 *aes, const unsigned char *ciphertext, int ciphertext_len,
                               unsigned char *plaintext, int *plaintext_len) {
    *plaintext_len = 0;
    for (int i = 0; i < ciphertext_len; i += 16) {
        type2_decrypt_block(aes, ciphertext + i, plaintext + *plaintext_len);
        *plaintext_len += 16;
    }
    *plaintext_len = pkcs7_unpad(plaintext, *plaintext_len);
}

/* 位反转函数：将8位数据的bit0与bit7互换，bit1与bit6互换，依此类推 */
static unsigned char bit_reverse(unsigned char x) {
    unsigned char result = 0;
    for (int i = 0; i < 8; i++) {
        if ((x >> i) & 1) {
            result |= (1 << (7 - i));
        }
    }
    return result;
}

/* 生成位反转S盒：S[i] = bit_reverse(i) ^ 0x63 */

    for (int i = 0; i < 256; i++) {
        sbox[i] = bit_reverse(i) ^ 0x63;
    }
    memset(inv_sbox, 0, 256);
    for (int i = 0; i < 256; i++) {
        inv_sbox[sbox[i]] = i;
    }
}

/* Type3置换层：按照公式 (i * 3 + 7) % 16 对状态字节重新排列
   这是魔改点之一，在加密首尾各执行一次 */
static void type3_permute_state(unsigned char state[16]) {
    unsigned char temp[16];
    for (int i = 0; i < 16; i++) {
        temp[(i * 3 + 7) % 16] = state[i];
    }
    memcpy(state, temp, 16);
}

/* Type3逆置换层：计算逆映射恢复原始排列 */

    unsigned char temp[16];

        inv_perm[(i * 3 + 7) % 16] = i;
    }
    for (int i = 0; i < 16; i++) {
        temp[inv_perm[i]] = state[i];
    }
    memcpy(state, temp, 16);
}

        w[i][1] = w[i - nk_words][1] ^ temp[1];

/* Type3加密单个分组：首尾各加一次置换层 */
static void type3_encrypt_block(ModifiedAES_Type3 *aes, const unsigned char plaintext[16], unsigned char ciphertext[16]) {
    unsigned char state[16];

    /* 魔改点：加密前先执行置换 */

    add_round_key(state, aes->round_keys[0]);

    for (int round = 1; round < aes->nr; round++) {
        type3_sub_bytes(aes, state);
        type3_shift_rows(state);
        type3_mix_columns(state);
        add_round_key(state, aes->round_keys[round]);
    }

    type3_sub_bytes(aes, state);
    type3_shift_rows(state);
    add_round_key(state, aes->round_keys[aes->nr]);

    type3_permute_state(state);

    memcpy(ciphertext, state, 16);
}

/* Type3解密单个分组：先逆置换，再执行逆向各轮操作，最后逆置换 */
static void type3_decrypt_block(ModifiedAES_Type3 *aes, const unsigned char ciphertext[16], unsigned char plaintext[16]) {
    unsigned char state[16];
    memcpy(state, ciphertext, 16);

    /* 先撤销最后的置换 */
    type3_inv_permute_state(state);
    add_round_key(state, aes->round_keys[aes->nr]);
    type3_inv_shift_rows(state);
    type3_inv_sub_bytes(aes, state);

    for (int round = aes->nr - 1; round > 0; round--) {
        add_round_key(state, aes->round_keys[round]);
        type3_inv_mix_columns(state);
        type3_inv_shift_rows(state);
        type3_inv_sub_bytes(aes, state);
    }

    add_round_key(state, aes->round_keys[0]);
    /* 撤销开始时的置换 */
    type3_inv_permute_state(state);

    memcpy(plaintext, state, 16);
}

static void type3_encrypt_ecb(ModifiedAES_Type3 *aes, const unsigned char *plaintext, int plaintext_len,
                               unsigned char *ciphertext, int *ciphertext_len) {

    memcpy(padded, plaintext, plaintext_len);
    int padded_len = plaintext_len;
    pkcs7_pad(padded, &padded_len, 16);

    *ciphertext_len = 0;
    for (int i = 0; i < padded_len; i += 16) {
        type3_encrypt_block(aes, padded + i, ciphertext + *ciphertext_len);
        *ciphertext_len += 16;
    }
}

/* Type3 ECB模式解密 */

                               unsigned char *plaintext, int *plaintext_len) {

        *plaintext_len += 16;
