Base系列算法原理

2026/5/16 算法

Base系列编码原理

2^n 次方系列（Base16 Base32 Base64)

这个系列主要通过位运算来实现编码

以下是常见是Base编码

编码 基数 位数 分组规则
Base16 16(2^4) 4 1字节 -> 2字符
Base32 32(2^5) 5 5字节 -> 8字符
Base64 64(2^6) 6 3字节 -> 4字符

编码原理

以每个字符对应的ASCII码的二进制为一组，取位数个，和对应编码表字符对应

字符串Hello World!对应的ASCII码二进制为

01001000 01100101 01101100 01101100 01101111 00100000 01010111 01101111 01110010 01101100 01100100 00100001

编码方式 二进制取位 加密后效果

Base32 01001 00001 10010 10110 11000 11011 00011 01111 00100 00001 01011 10110 11110 11100 10011 01100 01100 10000 10000 10000 ==== JBSWY3DPEBLW64TMMQQQ====
Base64 010010 000110 010101 101100 011011 000110 111100 100000 010101 110110 111101 110010 011011 000110 010000 100001 SGVsbG8gV29ybGQh

非8的因数编码提醒特别提示

如果出现编码后不够指定位数的例如

xxxxx xxxyy yyyyy y0000 ====

xxxxxx xxyy yyyy00 ====

在后面补0凑够6位然后补=直到够八位

def encode_base64(data):
    # 标准Base64码表
    alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
    # 储存编码字符
    result = ""
    bits = 0
    bits_count = 0
	# 遍历输入数据的每个字节
    for byte in data:
        # 把字节放入缓冲取
        bits = (bits << 8) | byte
        bits_count += 8
        # 当缓冲区至少有6位时取出6位与00111111进行与运算
        while bits_count >= 6:
            bits_count -= 6
            index = (bits >> bits_count) & 0x3f
            # 查表得到Base64字符
            result += alphabet[index]
    
    if bits_count > 0:
        # 剩余位置左移补足不足6位的数据
        index = (bits << (6 - bits_count)) & 0x3f
        result += alphabet[index]

    padding = (4 - len(result) % 4) % 4
    result += "=" * padding

    return result
def decode_base64(encoded):
    # Base64表
    alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
    result = bytearray()
    bits = 0
    bits_count = 0
    for char in encoded:
        # 遇到填充字符停止编码
        if char == '=':
            break
        # 把Base64字符转化为对应数值
        if 'A' <= char <= 'Z':
            val = ord(char) - 65
        elif 'a' <= char <= 'z':
            val = ord(char) - 71
        elif '0' <= char <= '9':
            val = ord(char) + 4
        elif char == '+':
            val = 62
        elif char == '/':
            val = 63
        else:
            #遇到非法字符不处理
            continue
        # 把解码的六位放入缓冲区
        bits = (bits << 6) | val
        bits_count += 6
        # 当缓冲区有8位时取出
        if bits_count >= 8:
            bits_count -= 8  # 为完整字节
            result.append((bits >> bits_count & 0xff))
    # 把bytearray转换为不可变的bytes返回
    return bytes(result)
if __name__ == "__main__":
    # 测试加密
    text ="Hello World!"
    data = text.encode('utf-8')
    # 编码
    encoded = encode_base64(data)
    print(encoded)
    # 解码
    decoded = decode_base64(encoded)
    print(decoded.decode('utf-8'))

// 建立base64解码函数
void encode_base64(unsigned char *data, int data_len, char *output)
{
    // base64码表
    char alphabat[] ="ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
    int i,j;
    int bits = 0;
    int bits_count = 0;
	// 初始化输出位置
    j = 0;
	// 遍历每一字节
    for(i = 0;i < data_len;i++)
    {
        // 先腾出八位空间然后再用位操作写入新的字节
        bits = (bits << 8) | data[i];
        bits_count += 8;
        // 如果写出字节数大于六取出6位与00111111进行与运算
        while (bits_count >= 6)
        {
            bits_count -= 6;
            output[j] = alphabat[(bits >> bits_count) & 0x3f];
            j++;
        }
    // 处理不足六位时的数据
    }
    if(bits_count > 0)
    {
        // 剩余位左移补足六位
        output[j] = alphabat[(bits << (6 - bits_count)) & 0x3f];
        j++;
    }
    // 填充字符
    while (j % 4 != 0)
    {
        output[j] = '=';
        j++;
    }
    // 添加\0结束符
    output[j] = '\0';
}
// 建立解密函数
void decode_base64(char *encoded, unsigned char *output, int *output_len)
{
    // Base64表
    char alphabat[] ="ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
    int i,j;
    int bits = 0;
    int bits_count = 0;
    // 初始化输出位置
    j = 0;
    // 遍历字符串到\0或者=时停止
    for ( i = 0; encoded[i] != '\0' && encoded[i] != '='; i++)
    {
        unsigned char val;
		// 对应Base64的编码表
        if (encoded[i] >= 'A' && encoded[i] <= 'Z')
            val = encoded[i] - 'A';
        else if (encoded[i] >= 'a' && encoded[i] <= 'z')
            val = encoded[i] - 'a' + 26;
        else if (encoded[i] >= '0' && encoded[i] <= '9')
            val = encoded[i] - '0' + 52;
        else if (encoded[i] == '+')
            val = 62;
        else if (encoded[i] == '/')        
            val = 63;
        else
            // 遇到非法字符跳过不处理
            continue;
        // 腾出六位空间放新的数据
        bits = (bits << 6) | val;
        bits_count += 6;
        // 当缓冲区有八位时，提取出一个字节
        if (bits_count >= 8)
        {
            bits_count -= 8;
            output[j] = (bits >> bits_count) & 0xff;
            j++;
        }
    }
    // 将解码数据返回
    *output_len = j;
}

非2^n 系列

编码原理

把一个原始数据视为一个十进制大整数不断除以编码数然后

以下以Base58为例

编码 "Hello" (48656c6c6f 十六进制 = 310939249775 十进制)

310939249775 ÷ 58 = 5361021547 余 41 → 'n'
5361021547 ÷ 58 = 92431405 余 49 → 'x'  
92431405 ÷ 58 = 1593644 余 53 → 'y'
1593644 ÷ 58 = 27476 余 12 → 'E'
27476 ÷ 58 = 473 余 42 → 'o'
473 ÷ 58 = 8 余 9 → 'A'
8 ÷ 58 = 0 余 8 → '9'

结果：9Aonyxn

直到不能继续除为止

python代码演示

def encode_base58(data):
    # Base58编码表
    alphabet = "123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz"
    # 每个字节视为 256 进制的一位，从高位到低位累积
    num = 0
    # 把数值左移八位
    for byte in data:
        num = num * 256 + byte
	# 
    result = ""
    while num > 0:
        num, rem = divmod(num, 58)
        # 返回余数和商
        result = alphabet[rem] + result

    for byte in data:
        if byte == 0:
            # 在结果前面添加一个1
            result = '1' + result
        else:
            # 遇到非零字符停止
            break
    # 返回结果值
    return result

def decode_base58(encoded):
    # Base58表
    alphabet = "123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz"
    num = 0
    
    for char in encoded:
        # 查找字符在编码表的位置，并且在编码表左移一位
        num = num * 58 + alphabet.index(char)
    # 每次除256取余，得到一个字节
    result = bytearray()
    while num > 0:
        num, rem = divmod(num, 256)
        result.insert(0, rem)

    for char in encoded:
        if char == '1':
            # 在结果前面插入0x00字节
            result.insert(0, 0)
        else:
            # 遇到第一个非零值返回
            break

    return bytes(result)

#include <stdio.h>
#include <string.h>

void encode_base58(unsigned char *data, int data_len, char *output)
{
    char alphabet[] = "123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz";
    int i, j, k;
    int carry;
    int digits[256] = {0};
    int digits_len = 1;

    for ( i = 0; i < data_len; i++)
    {
        carry = data[i];
        for ( j = 0; j < digits_len; j++)
        {
            carry += digits[j] * 256;
            digits[j] = carry % 58;
            carry /= 58;
        }
        while (carry > 0)
        {
            digits[digits_len] = carry % 58;
            digits_len++;
            carry /= 58;
        }
        
    }
    k = 0;
    for ( i = 0; i < data_len && data[i] == 0; i++)
    {
        output[k] = '1';
        k++;
    }
    j = digits_len - 1;
    while (j >= 0)
    {
        output[k] = alphabet[digits[j]];
        k++;
        j--;
    }
    output[k] = '\0';
    
}

void decode_base58(char *encoded, unsigned char *output, int *output_len)
{
    char alphabet[] = "123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz";
    int i, j, k;
    int carry;
    int bytes[256] = {0};
    int bytes_len = 1;
    int encoded_len = strlen(encoded);

    for ( i = 0; i < encoded_len; i++)
    {
        int val = 0;
        for ( j = 0; j < 58; j++)
        {
            if (encoded[i] == alphabet[j])
            {
                val = j;
                break;
            }
            
        }
        carry = val;
        for ( j = 0; j < bytes_len; j++)
        {
            carry += bytes[j] * 58;
            bytes[j] = carry %256;
            carry /= 256;
        }
        while (carry > 0)
        {
            bytes[bytes_len] = carry % 256;
            bytes_len++;
            carry /= 256;
        }
        
    }
    k = 0;
    for ( i = 0; i < encoded_len && encoded[i] == '1'; i++)
    {
        output[k] = 0;
        k++;
    }
    j = bytes_len - 1;
    while (j >= 0)
    {
        output[k] = bytes[j];
        k++;
        j--;
    }
    *output_len = k;
}

混合编码

编码原理

通过混合的编码，来实现更高的效率。这里以Base85（Ascii85）为例，它是一种将4字节数据编码为5个可打印ASCII字符的高效编码方式。

编码 基数 字节分组 编码后字符数 效率
Base64 64 3字节 4字符 75%
Base85 85 4字节 5字符 80%

Base85的编码过程：

  1. 将4字节数据视为一个32位无符号整数
  2. 用这个整数不断除以85，得到5个余数
  3. 每个余数加上33（ASCII ‘!’）得到编码字符

以下以字符串”Test”为例：

"Test" 的ASCII码：84 101 115 116
视为32位整数：84*256^3 + 101*256^2 + 115*256 + 116 = 1413829428

1413829428 ÷ 85 = 16633287 余 3 → '!' + 3 = '$'
16633287 ÷ 85 = 195685 余 62 → '!' + 62 = 'a'
195685 ÷ 85 = 2302 余 15 → '!' + 15 = 'P'
2302 ÷ 85 = 27 余 7 → '!' + 7 = '('
27 ÷ 85 = 0 余 27 → '!' + 27 = '"'

结果：$aP("

python代码演示

# -*- coding: utf-8 -*-
def encode_base85(data):
    alphabet = "!\"#$%&'()*+,-./0123456789:;<=>?@ABCDEFGHIJKLMNOPQRSTUVWXYZ[\\]^_`abcdefghijklmnopqrstu"
    result = ""
    i = 0
    data_len = len(data)
    
    while i < data_len:
        # 每次处理4字节
        chunk = data[i:i+4]
        chunk_len = len(chunk)
        
        # 将4字节打包为一个32位整数
        value = 0
        for j in range(chunk_len):
            value = (value << 8) | chunk[j]
        
        # 如果是最后一个块且不足4字节，特殊处理
        if chunk_len < 4:
            # 计算需要的字符数
            chars_needed = chunk_len + 1
            encoded_chars = []
            for j in range(chars_needed):
                encoded_chars.append(alphabet[value % 85])
                value //= 85
            # 反转得到正确顺序
            result += ''.join(reversed(encoded_chars))
            break
        
        # 正常编码4字节为5字符
        temp = []
        for j in range(5):
            temp.append(alphabet[value % 85])
            value //= 85
        # 反转得到正确顺序
        result += ''.join(reversed(temp))
        i += 4
    
    return result

def decode_base85(encoded):
    alphabet = "!\"#$%&'()*+,-./0123456789:;<=>?@ABCDEFGHIJKLMNOPQRSTUVWXYZ[\\]^_`abcdefghijklmnopqrstu"
    result = bytearray()
    value = 0
    count = 0
    
    for char in encoded:
        if char == 'z':
            # 特殊字符z表示4个0字节
            result.extend([0, 0, 0, 0])
            continue
        
        idx = alphabet.index(char)
        value = value * 85 + idx
        count += 1
        
        if count == 5:
            # 5个字符解码为4字节
            for j in range(3, -1, -1):
                result.append((value >> (8 * j)) & 0xff)
            value = 0
            count = 0
    
    # 处理最后不足5个字符的情况
    if count > 0:
        # 补零使最后一个块完整
        for j in range(count, 5):
            value = value * 85 + 84  # 'u' 对应84
        
        # 根据原始长度确定输出字节数
        for j in range(count - 1):
            result.append((value >> (8 * (3 - j))) & 0xff)
    
    return bytes(result)

if __name__ == '__main__':
    # 测试
    text = "Hello World!"
    data = text.encode('utf-8')
    # Base85编码
    encoded = encode_base85(data)
    print(encoded)
    # Base85解码
    decoded = decode_base85(encoded)
    print(decoded.decode('utf-8'))

#include <stdio.h>
#include <string.h>
#include <stdlib.h>

void encode_base85(unsigned char *data, int data_len, char *output)
{
    char alphabet[] = "!\"#$%&'()*+,-./0123456789:;<=>?@ABCDEFGHIJKLMNOPQRSTUVWXYZ[\\]^_`abcdefghijklmnopqrstu";
    int i, j, k = 0;
    unsigned long value;
    
    for (i = 0; i < data_len; i += 4)
    {
        value = 0;
        int bytes_used = (data_len - i < 4) ? (data_len - i) : 4;
        
        for (j = 0; j < bytes_used; j++)
        {
            value = (value << 8) | data[i + j];
        }
        
        if (bytes_used < 4)
        {
            // 处理最后不足4字节的块
            for (j = 0; j < bytes_used + 1; j++)
            {
                output[k + bytes_used - j] = alphabet[value % 85];
                value /= 85;
            }
            k += bytes_used + 1;
            break;
        }
        
        // 正常编码4字节为5字符
        for (j = 4; j >= 0; j--)
        {
            output[k + j] = alphabet[value % 85];
            value /= 85;
        }
        k += 5;
    }
    output[k] = '\0';
}

void decode_base85(char *encoded, unsigned char *output, int *output_len)
{
    char alphabet[] = "!\"#$%&'()*+,-./0123456789:;<=>?@ABCDEFGHIJKLMNOPQRSTUVWXYZ[\\]^_`abcdefghijklmnopqrstu";
    int i, j, k = 0;
    unsigned long value = 0;
    int count = 0;
    int encoded_len = strlen(encoded);
    
    for (i = 0; i < encoded_len; i++)
    {
        if (encoded[i] == 'z')
        {
            // 特殊字符z表示4个0字节
            output[k++] = 0;
            output[k++] = 0;
            output[k++] = 0;
            output[k++] = 0;
            continue;
        }
        
        // 查找字符在编码表中的位置
        int idx = -1;
        for (j = 0; j < 85; j++)
        {
            if (alphabet[j] == encoded[i])
            {
                idx = j;
                break;
            }
        }
        if (idx == -1) continue;
        
        value = value * 85 + idx;
        count++;
        
        if (count == 5)
        {
            // 5个字符解码为4字节
            for (j = 3; j >= 0; j--)
            {
                output[k++] = (value >> (8 * j)) & 0xff;
            }
            value = 0;
            count = 0;
        }
    }
    
    // 处理最后不足5个字符的情况
    if (count > 0)
    {
        for (j = count; j < 5; j++)
        {
            value = value * 85 + 84;  // 'u' 对应84
        }
        for (j = 0; j < count - 1; j++)
        {
            output[k++] = (value >> (8 * (3 - j))) & 0xff;
        }
    }
    
    *output_len = k;
}

}
