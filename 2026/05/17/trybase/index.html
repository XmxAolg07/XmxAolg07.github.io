base系列变表和实战

2^n次方base编码变表

base系列的编码不只是常规的base表，在ctf的实战中往往会使用变表而不是标准表

变表的原理就是把对应的索引值对应的字符变化

例如原本对应是A的位置变成了O导致最终编码不同

我们给出base64变表的代码作为演示

python代码演示

def encode_base64(data, table):
    # 储存编码字符
    result =''
    bits = 0
    bits_count = 0
	# 遍历每个字符串
    for byte in data:
        bits = (bits << 8) | byte
        bits_count += 8
		# 放入缓冲区
        while bits_count >= 6:
            bits_count -= 6
            index = (bits >> bits_count) & 0x3f
            result += table[index]
    
    if bits_count > 0:
        # 剩余位置左移补足不足6位的数据
        index = (bits << (6 - bits_count)) & 0x3f
        result += table[index]

    padding = (4 - len(result) % 4) % 4
    result += '=' * padding

    return result
def decode_base64(encoded, table):
    # 创建可变字节数组
    result = bytearray()
    bits = 0
    bits_count = 0
	# 遍历每个字符
    for char in encoded:
        if char == "=":
            break
        # 在变表中查找字符对应的索引
        index = table.index(char)
        bits = (bits << 6) | index
        bits_count += 6
		# 积累到八位提取一个字节
        if bits_count >= 8:
            bits_count -= 8
            result.append((bits >> bits_count) & 0xff)

    return bytes(result)

if __name__ == '__main__':
    # 创建变表
    table = 'qvEJAfHmUYjBac+u8Ph5n9Od17FrICL/X0gVtM4Qk6T2z3wNSsyoebilxWKGZpRD'
    text = "Hello World!"
    data = text.encode("utf-8")
    # 使用变表编码
    encoded = encode_base64(data, table)
    print(encoded)
    # 变表解码
    decoded = decode_base64(encoded, table)
    print(decoded.decode("utf-8"))
    # 原表编码
    standard_table = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
    encoded_std = encode_base64(data, standard_table)
    print(encoded_std)
    # 原本解码
    decoded_std = decode_base64(encoded_std, standard_table)
    print(decoded_std.decode("utf-8"))

#include <stdio.h>
#include <string.h>
// 建立编码函数
void encode_base64(unsigned char *data, int data_len, char *output, char *table)
{
    int i, j;
    int bits = 0;
    int bits_count = 0;
    // 初始化输出位置
    j = 0;
	// 遍历每一字节
    for ( i = 0; i < data_len; i++)
    {
        // 腾出八位空间之后用位操作写入字符
        bits = (bits << 8) | data[i];
        bits_count += 8;
		// 如果写出字节数大于六就取出6位与00111111做与运算
        while (bits_count >= 6)
        {
            bits_count -= 6;
            output[j] = table[(bits >> bits_count) & 0x3f];
            j++;
        }
    // 不足六位补填充字符
    }
    while (j % 4 != 0)
    {
        output[j] = '=';
        j++;
    }
    output[j] = '\0';
    
}
// 建立在编码表寻找对应索引值的函数
int find_index(char *table, char ch)
{
    int i;
    for ( i = 0; i < 64; i++)
    {
        if (table[i] == ch)
        {
            return i;
        }
        
    }
    return 0;
}
// 建立解码函数
void decode_case64(char *encoded, unsigned char *output, int *output_len, char *table)
{
    int i, j;
    int bits = 0;
    int bits_count = 0;
    // 初始化输出位置
    j = 0;
	// 遍历每一字符到\0或者=停止
    for ( i = 0; encoded[i] != '\0' && encoded[i] != '='; i++)
    {
        // 腾出六位放新的数据
        int index = find_index(table, encoded[i]);
        bits = (bits << 6) | index;
        bits_count += 6;
        // 当缓冲区有八位时，提取出一个字节
        if (bits_count >= 8)
        {
            bits_count -= 8;
            output[j] = (bits >> bits_count) &0xff;
            j++;
        }
    }
    // 返回解码数据
    *output_len = j;
}

int main(void)
{
    // 建立原表和变表
    char table[] = "qvEJAfHmUYjBac+u8Ph5n9Od17FrICL/X0gVtM4Qk6T2z3wNSsyoebilxWKGZpRD";
    char standard_table[] = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
    char *text = "Hello World!";
    int data_len = strlen(text);
    // 创建足够的空间
    char encoded[256];
    unsigned char decoded[256];
    int decoded_len;
    int i;
	// 输出变表编码结果
    encode_base64((unsigned char *)text, data_len, encoded, table);
    printf("%s\n", encoded);
	// 输出变表解码结果
    decode_case64(encoded, decoded, &decoded_len, table);
    for (i = 0; i < decoded_len; i++)
    {
        printf("%c", decoded[i]);
    }
    printf("\n");
	// 输出原表编码结果
    encode_base64((unsigned char *)text, data_len, encoded, standard_table);
    printf("%s\n", encoded);
	// 输出原表解码结果
    decode_case64(encoded, decoded, &decoded_len, standard_table);
    for ( i = 0; i < decoded_len; i++)
    {
        printf("%c", decoded[i]);
    }
    printf("\n");

大数除法变表

和上面的原理差不多，都是把索引值对应的字符更改实现的变表

python代码演示

def encode_base58(data, table):
    num = 0
    # 把数值左移八位
    for byte in data:
        num = num * 256 + byte

    result =''
    while num > 0:
        num, rem = divmod(num, 58)
        # 返回余数和商
        result = table[rem] + result

    for byte in data:
        if byte == 0:
            result = table[0] + result
        else:
            # 遇到非零字符停止
            break

    return result

def decode_base58(encoded, table):
    num = 0
    # 查找字符在编码表的位置
    for char in encoded:
        num = num * 58 + table.index(char)
	# 每次除256取余，得到一个字节
    result =bytearray()
    while num > 0:
        num, rem = divmod(num, 256)
        result.insert(0, rem)

    for char in encoded:
        if char == table[0]:
            # 在结果前插入0x00字节
            result.insert(0, 0)
        else:
            # 遇到第一个非零值返回
            break
    return bytes(result)

if __name__ == "__main__":
    # 创建原表和变表

    custom_table = "ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz123456789"
	# 创建字符串
    text = "Hello World!"

#include <stdio.h>
#include <string.h>
// 建立函数查找索引值对应的字符
int find_index58(char *table, char ch)
{
    int i;
    for (i = 0; i < 58; i++)
    {
        if (table[i] == ch)
            return i;
    }
    return 0;
}
// 建立编码函数
void encode_base58(unsigned char *data, int data_len, char *output, char *table)
{
    int i, j, k;
    int carry;
    int digits[256] = {0};
    int digits_len = 1;
    // 遍历字符串
    for (i = 0; i < data_len; i++)
    {
        carry = data[i];
        // 每个除以58
        for (j = 0; j < digits_len; j++)
        {
            carry += digits[j] * 256;
            digits[j] = carry % 58;
            carry /= 58;
        }
        while (carry > 0)
        {
            digits[digits_len] = carry % 58;
            digits_len++;
            carry /= 58;
        }
    }
    
    k = 0;
    for (i = 0; i < data_len && data[i] == 0; i++)
    {
        output[k] = table[0];
        k++;
    }
    
    j = digits_len - 1;
    while (j >= 0)
    {
        output[k] = table[digits[j]];
        k++;
        j--;
    }
    
    output[k] = '\0';
}
// 建立解码函数
void decode_base58(char *encoded, unsigned char *output, int *output_len, char *table)
{
    int i, j, k;
    int carry;
    int bytes[256] = {0};
    int bytes_len = 1;
    int encoded_len = strlen(encoded);
    // 遍历整个字符串
    for (i = 0; i < encoded_len; i++)
    {
        int val = find_index58(table, encoded[i]);
        
        carry = val;
        for (j = 0; j < bytes_len; j++)
        {
            carry += bytes[j] * 58;
            bytes[j] = carry % 256;
            carry /= 256;
        }
        while (carry > 0)
        {
            bytes[bytes_len] = carry % 256;
            bytes_len++;
            carry /= 256;
        }
    }
    
    k = 0;
    for (i = 0; i < encoded_len && encoded[i] == table[0]; i++)
    {
        output[k] = 0;
        k++;
    }
    
    j = bytes_len - 1;
    while (j >= 0)
    {
        output[k] = bytes[j];
        k++;
        j--;
    }
    
    *output_len = k;
}

{
    // 建立原本和变表
    char standard_table[] = "123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz";
    char custom_table[] = "ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz123456789";
    char *text = "Hello World!";
    // 空出一定字节数给解码编码使用
    int data_len = strlen(text);
    char encoded[256];
    unsigned char decoded[256];
    int decoded_len;
    int i;
    // 原表编码解码
    encode_base58((unsigned char *)text, data_len, encoded, standard_table);
    printf("%s\n", encoded);
    decode_base58(encoded, decoded, &decoded_len, standard_table);
    for (i = 0; i < decoded_len; i++)
    {
        printf("%c", decoded[i]);
    }
    printf("\n");
    // 变表解码编码
    encode_base58((unsigned char *)text, data_len, encoded, custom_table);
    printf("%s\n", encoded);
    decode_base58(encoded, decoded, &decoded_len, custom_table);
    for (i = 0; i < decoded_len; i++)
    {
        printf("%c", decoded[i]);
    }
    printf("\n");
    
    return 0;
}

import base64

if __name__ == '__main__':

    table = 'qvEJAfHmUYjBac+u8Ph5n9Od17FrICL/X0gVtM4Qk6T2z3wNSsyoebilxWKGZpRD'

    # 创建翻译表
    trans = str.maketrans(table, standard_table)
    
    # 将密文用变表翻译成标准base64编码
    standard_b64 = text.translate(trans)
    decoded = base64.b64decode(standard_b64)
    
    print(decoded.decode("utf-8"))

结语

在ctf的比赛中base系列编码是常客，但是很多的reverse的题用的都不是标准的base编码

很多的使用了变表还有和其他算法混用，例如AES,rc4，或者简单的加减移位等等
