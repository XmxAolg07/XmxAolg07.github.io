rc4算法原理和魔改

2026/5/19 CTF解题 算法

rc4算法原理和魔改

rc4原理

第一步

准备一个包含256个字节的数组 S，并按照 S[0]=0, S[1]=1, …, S[255]=255 的顺序填充，初始化S盒

设置一个密钥key把S盒打乱，最后256次后获得一个和密钥相关但是看起来毫无规律的排序

第二步

算法会不断更新两个索引 i 和 j，并交换 S[i] 和 S[j] 的值。然后，从 S 盒的某个位置 t（t = S[i] + S[j]）取出一个字节 S[t]就是当前这一步生成的密钥流字节

这个寻找、交换、输出的过程会不断重复，直到生成的密钥流长度与待加密的明文长度相等为止。S 盒的状态也在每一次循环中被动态更新

第三步

把生成好的密钥流与明文数据每个字节进行异或运算，得到密文

简单流程

加密过程：

明文 xor 密钥流 = 密文

解密过程

密文 xor 密钥流 = 明文

关键变量

密钥流

密钥流的长度和明文的长度是相应的，从不断变化的S盒生成，计算出一个字节就输出一共字节

密钥

一个长度为1-256的数组，决定S盒的初始状态

状态向量

一个长度为256的数，经过密钥的变化，变成了一个看似乱序的排序（0-255各出现一次，顺序打乱）

暂时向量

一个256字节的数组，把密钥反复填满256字节，在 KSA 阶段使用，PRGA 开始后就不再需要

实际演示

python脚本

def rc4_init(key):
    key_length = len(key)
    # 状态向量S
    S = list(range(256))
    # 索引值
    j = 0
    # 遍历S盒每一个位置
    for i in range(256):
        # 当i超过密钥长度时循环使用密钥
        j = (j + S[i] + key[i % key_length]) % 256
        # 打乱S盒的顺序
        S[i], S[j] = S[j], S[i]
        # 返回被打乱的S盒
    return S

def rc4_crypt(S, data):
    # 索引值i，j
    i = 0
    j = 0
    # 存放结果（明文或者密文）
    result = bytearray()
    # 逐字节处理数据
    for byte in data:
        # 生成密钥流
        i = (i + 1) % 256
        j = (j + S[i]) % 256
        # 继续打乱S盒
        S[i], S[j] = S[j], S[i]
        # k就是密钥流字节
        k = S[(S[i] + S[j]) % 256]
        # 加密/解密
        result.append(byte ^ k)
    return bytes(result)

if __name__ == '__main__':
    # 密钥K
    key = b'secret_key'
    # 明文
    plaintext = b'Hello World!'
    # 用密钥初始S盒
    S = rc4_init(key)
    # PRGA + XOR，加密
    ciphertext = rc4_crypt(S, plaintext)
    # 输出密文
    print('start', ciphertext.hex())
    # 用同样的密钥初始S盒
    S = rc4_init(key)
    # PRGA + XOR，解密
    decrypted = rc4_crypt(S, ciphertext)
    # 输出解密后的密文
    print('end',decrypted.decode() )

void swap(unsigned char* a, unsigned char* b)
{
    unsigned char temp = *a;
    *a = *b;
    *b = temp;
}
// 加密/解密函数
void RC4_enc(unsigned char* plaintext, int length, const char* key)
{
    int i, j;
    // 状态向量
    unsigned char* S;
    // 暂时向量
    unsigned char k[256];
    int keylen = 0;
	// 遍历字符串直到遇到"\0"
    while (key[keylen] != '\0')
    {
        keylen++;
    }
    // 构建暂时向量k
    for ( i = 0; i < 256; i++)
    {
        k[i] = key[i % keylen];
    }
    // 分配256字节内存
    S = (unsigned char*)malloc(256 * sizeof(unsigned char));
    for ( i = 0; i < 256; i++)
    {
        S[i] = i;
    }
    // 乱序S盒
    j = 0;
    for ( i = 0; i < 256; i++)
    {
        j = (j + S[i] + k[i]) % 256;
        swap(&S[i], &S[j]);
    }
    // 伪随机生成算法
    i = 0;
    j = 0;
    for (int n = 0; n < length; n++)
    {
        i = (i + 1) % 256;
        j = (j + S[i]) % 256;
        swap(&S[i], &S[j]);
        int t = (S[i] + S[j]) % 256;
        // 异或加密
        plaintext[n] ^= S[t];
    }
    // 释放动态分配的内存
    free(S);
}
// 主函数
int main(void)
{
    // 明文
    unsigned char plaintext[]=
    {
        0x11,0x45,0x14
    };
    // 密钥key
    const char key[] = "key";
    // 调用rc4加密
    RC4_enc(plaintext, sizeof(plaintext), key);
	// 输出加密后的密文
    for (int i = 0; i < sizeof(plaintext); i++)
        printf("%02X", plaintext[i]);
    printf("\n");
    
    return 0;
}

rc4魔改

rc4在ctf的题目中往往不是使用标准的rc4，而是使用了魔改，并且魔改通常不是单独使用，而且混合使用，多种魔改一起使用

改常量

标准的rc4的部分常量是死的，出题人会把常量改掉，让加密逻辑变得完全不同

常见的改变点：

改动位置 标准值 实例
S 盒大小 256 改成 128、64、甚至 50
PRGA 中 i 的步进 i = （i + 1) % 256 改成 i = (i + 3) % 256 或 i = (i + 7) % 128
j 的更新公式 j = (j + S[i]) % 256 改成 j = (j + S[i] + 1) % 256 或 j = (j + S[i] * 2) % 128
密钥流取值位置 S[(S[i] + S[j]) % 256] 改成 S[(S[i] ^ S[j]) % 256] 或 S[(S[i] + S[j] + 1) % 128]

常量的改动不破坏算法的对称性，只要加密和解密用同一套常量，密钥流就完全一致

实际代码演示

# 带常量修改的RC4初始化函数
def rc4_init_constant(key, constant):
    # 获取密钥长度
    key_length = len(key)
    # 初始化状态向量S为0-255
    S = list(range(256))
    # 索引值j
    j = 0
    # KSA阶段：用密钥和常量共同打乱S盒
    for i in range(256):
        # 在标准RC4基础上增加了constant常量
        j = (j + S[i] + key[i % key_length] + constant) % 256
        # 交换S[i]和S[j]
        S[i], S[j] = S[j], S[i]
    # 返回被打乱的S盒
    return S

# 带常量修改的RC4加密/解密函数
def rc4_crypt_constant(S, data, constant):
    # 索引值i，j
    i = 0
    j = 0
    # 存放结果（明文或者密文）
    result = bytearray()
    # 逐字节处理数据
    for byte in data:
        # i的步进改为constant（标准RC4是i = (i + 1) % 256）
        i = (i + constant) % 256
        # j的更新增加了constant常量
        j = (j + S[i] + constant) % 256
        # 继续打乱S盒
        S[i], S[j] = S[j], S[i]
        # 密钥流取值位置也增加了constant常量
        k = S[(S[i] + S[j] + constant) % 256]
        # 加密/解密：异或运算
        result.append(byte ^ k)
    return bytes(result)

if __name__ == '__main__':

    key = b'secret_key'
    # 明文
    plaintext = b'Hello World!'
    # 修改的常量值

    S = rc4_init_constant(key, constant)

    print("public:", decrypted.decode())

{

        // 交换S[i]和S[j]
        unsigned char temp = S[i];
        S[i] = S[j];
        S[j] = temp;

    {
        // i的步进改为constant（标准RC4是i = (i + 1) % 256）

        // 加密/解密：异或运算
        result[n] = data[n] ^ k;
    }

换S盒

标准 RC4 的 S 盒初始值是 [0, 1, 2, …, 255]，我们可以把S盒换成自定义的字节数组

例如，改变顺序，减小S盒等等

但是不管初始 S 盒长什么样，KSA 会把它进一步打乱， 但不同的起点会导致完全不同的 S 盒演化路径，最终产生的密钥流也完全不同

实际代码演示

# 根据种子生成自定义S盒
def generate_custom_sbox(seed):
    # 初始化为0-255
    S = list(range(256))
    # 使用种子打乱S盒顺序
    for i in range(256):
        j = (i + seed + S[i]) % 256
        S[i], S[j] = S[j], S[i]
    return S

# 使用自定义S盒的RC4初始化函数
def rc4_init_custom_sbox(key, seed):
    # 获取密钥长度
    key_length = len(key)
    # 先用种子生成自定义初始S盒
    S = generate_custom_sbox(seed)
    # 索引值j
    j = 0
    # KSA阶段：用密钥进一步打乱自定义S盒
    for i in range(256):
        j = (j + S[i] + key [i % key_length]) % 256
        S[i], S[j] = S[j], S[i]
    return S

# 标准RC4加密/解密函数（PRGA阶段）
def rc4_crypt(S, data):
    # 索引值i，j
    i = 0
    j = 0
    # 存放结果（明文或者密文）
    result = bytearray()
    # 逐字节处理数据
    for byte in data:
        i = (i +1) % 256
        j = (j + S[i]) % 256
        S[i], S[j] = S[j], S[i]
        # k就是密钥流字节
        k = S[(S[i] + S[j]) % 256]
        # 加密/解密：异或运算
        result.append(byte ^ k)
    return bytes(result)

void generate_custom_sbox(unsigned char *S, int seed)
{
    int i, j;

    // 初始化为0-255
    for (i = 0; i < 256; i++)
    {
        S[i] = i;
    }

    for (i = 0; i < 256; i++)
    {
        j = (i + seed + S[i]) % 256;

        unsigned char temp = S[i];
        S[i] = S[j];
        S[j] = temp;
    }
}

// 使用自定义S盒的RC4初始化函数
void rc4_init_custom_sbox(unsigned char *S, const unsigned char *key, int key_length, int seed)
{
    int i, j = 0;

    // 先用种子生成自定义初始S盒
    generate_custom_sbox(S, seed);
    
    // KSA阶段：用密钥进一步打乱自定义S盒
    for (i = 0; i < 256; i++)
    {
        j = (j + S[i] + key[i % key_length]) % 256;

        unsigned char temp = S[i];
        S[i] = S[j];
        S[j] = temp;
    }
}

// 标准RC4加密/解密函数（PRGA阶段）
void rc4_crypt(unsigned char *S, const unsigned char *data, unsigned char *result, int data_length)
{
    // 索引值i，j
    int i = 0, j = 0;

    // 逐字节处理数据
    for (int n = 0; n < data_length; n++)
    {
        i = (i + 1) % 256;
        j = (j + S[i]) % 256;

        // 交换S[i]和S[j]，继续打乱S盒
        unsigned char temp = S[i];
        S[i] = S[j];
        S[j] = temp;

        // k就是密钥流字节
        unsigned char k = S[(S[i] + S[j]) % 256];

        // 加密/解密：异或运算
        result[n] = data[n] ^ k;
    }
}

加可逆变换

标准 RC4 把从 S 盒取出的字节直接作为密钥流与明文异或。出题人在这个环节额外加了一层数学运算，让密钥流字节经过”二次加工”

这个运算必须是可逆的，否则出题人自己也没法解密

常见可逆变换

变换类型 示例 可逆原因
额外异或 k = S[t] ^ 0x55 再异或一次 0x55 就还原
字节循环移位 k = S[t] ^ 0x55 反向移位就还原
固定表替换 k = lut[S[t]] 只要 LUT 是双射（一对一映射），就能建逆表还原
加减常量 k = (S[t] + 0x37) % 256 减回去就还原
与S盒其他元素运算 k = S[t] ^ S[(t+1)%256] 结合 S 盒状态可逆推

解密时，只要用同样的 f 函数处理同样的 S 盒状态，就能生成同样的密钥流，异或回去就还原明文，对称性由 f 的可逆性保证

实际代码演示

# 可逆变换函数：对字节进行加密方向或解密方向的变换
def reversible_transform(byte, forward):
    if forward:
        # 加密方向变换：加37 → 循环左移3位 → 异或0xA5 → 循环左移7位 → 减19
        byte = (byte + 37) & 0xFF
        byte = ((byte << 3) | (byte >> 5)) & 0xFF
        byte = byte ^ 0xA5
        byte = ((byte << 7) | (byte >> 1)) & 0xFF
        byte = (byte - 19) & 0xFF
    else:
        # 解密方向变换：反向操作，与加密顺序相反
        byte = (byte + 19) & 0xFF
        byte = ((byte >> 7) | (byte << 1)) & 0xFF
        byte = byte ^ 0xA5
        byte = ((byte >> 3) | (byte << 5)) & 0xFF
        byte = (byte - 37) & 0xFF   
    return byte

# 标准RC4初始化函数（KSA阶段）
def rc4_init(key):
    key_length = len(key)
    # 初始化状态向量S为0-255
    S = list(range(256))
    j = 0
    for i in range(256):
        j = (j + S[i] + key[i % key_length]) % 256
        S[i], S[j] = S[j], S[i]
    return S

# 带可逆变换的RC4加密函数
def rc4_crypt_transform(S, data):
    # 索引值i，j
    i = 0
    j = 0
    # 存放结果（密文）
    result = bytearray()
    # 逐字节处理数据
    for byte in data:
        i = (i + 1) % 256
        j = (j + S[i]) % 256
        S[i], S[j] = S[j], S[i]
        # k就是密钥流字节
        k = S[(S[i] + S[j]) % 256]
        # 先对明文进行可逆变换，再与密钥流异或
        transformed = reversible_transform(byte, True)
        result.append(transformed ^ k)
    return bytes(result)

# 带可逆变换的RC4解密函数
def rc4_decrypt_transform(S, data):
    # 索引值i，j
    i = 0
    j = 0
    # 存放结果（明文）

unsigned char reversible_transform(unsigned char byte, int forword)
{
    if (forword)
    {
        // 加密方向变换：加37 → 循环左移3位 → 异或0xA5 → 循环左移7位 → 减19
        byte = (byte + 37) & 0xFF;
        byte = ((byte << 3) | (byte >> 5)) & 0xFF;
        byte = byte ^ 0xA5;
        byte = ((byte << 7) | (byte >> 1)) & 0xFF;
        byte = (byte - 19) & 0xFF;
    }
    else
    {
        // 解密方向变换：反向操作，与加密顺序相反
        byte = (byte + 19) & 0xFF;
        byte = ((byte >> 7) | (byte << 1)) & 0xFF;
        byte = byte ^ 0xA5;
        byte = ((byte >> 3) | (byte << 5)) & 0xFF;
        byte = (byte - 37) & 0xFF;
    }
    return byte;

}

{

    int i = 0, j = 0;

    // 逐字节处理数据
    for (int n = 0; n < data_length; n++)
    {
        i = (i + 1) % 256;
        j = (j + S[i]) % 256;
    
        // 交换S[i]和S[j]，继续打乱S盒
        unsigned char temp = S[i];
        S[i] = S[j];

        unsigned char transformed = reversible_transform(data[n], 1);
        result[n] = transformed ^ k;
    }
}
