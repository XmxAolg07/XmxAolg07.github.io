SM4算法和魔改

SM4原理

算法概述

SM4是中国国家密码管理局发布的对称分组密码算法。核心参数：

  • 分组长度：128位
  • 密钥长度：128位
  • 结构：非平衡Feistel网络
  • 轮数：32轮

核心组件

S盒

S盒是8位输入8位输出的非线性查找表，是算法唯一的非线性组件。

将32位字拆分为4个字节分别查表：

输入: a = b0 || b1 || b2 || b3 (各8位)
输出: S(b0) || S(b1) || S(b2) || S(b3)

L变换

对32位字进行线性扩散：

L(B) = B ⊕ (B <<< 2) ⊕ (B <<< 10) ⊕ (B <<< 18) ⊕ (B <<< 24)

L’变换

用于密钥扩展：

L'(B) = B ⊕ (B <<< 13) ⊕ (B <<< 23)

合成置换

T(.) = L(τ(.))     // τ为S盒替换
T'(.) = L'(τ(.))

密钥扩展

128位密钥扩展为32个32位轮密钥。

系统参数FK：4个32位常量

固定参数CK：32个32位常量，CK[i] = (4i,4i+1,4i+2,4i+3)各8位

扩展过程：

K[0..3] = MK[0..3] ⊕ FK[0..3]

K[i+4] = K[i] ⊕ T'(K[i+1] ⊕ K[i+2] ⊕ K[i+3] ⊕ CK[i])

rk[i] = K[i+4]   (i = 0..31)

加密过程

128位明文分为4个32位字 X[0..3]，进行32轮迭代。

轮函数：

X[i+4] = X[i] ⊕ T(X[i+1] ⊕ X[i+2] ⊕ X[i+3] ⊕ rk[i])

反序变换（最后一轮后）：

密文 = X[35] || X[34] || X[33] || X[32]

解密过程

解密与加密结构完全相同，仅轮密钥使用顺序相反：

加密: rk[0], rk[1], ..., rk[31]
解密: rk[31], rk[30], ..., rk[0]

    def __init__(self, key):
        # 密钥必须是16字节
        if len(key) != 16:
            raise ValueError("Key must be 16 bytes")
        
        # 把16字节密钥分成4个32位数
        MK = []
        for i in range(4):
            MK.append(int.from_bytes(key[i*4:(i+1)*4], 'big'))
        
        # 生成32个轮密钥
        self.rk = self._generate_round_keys(MK)
    
    def _rotl(self, x, n):
        # 循环左移
        return ((x << n) | (x >> (32 - n))) & 0xFFFFFFFF
    
    def _sbox_substitution(self, a):
        # 按字节查S盒替换
        b = 0
        for i in range(4):
            byte_val = (a >> (24 - i*8)) & 0xFF
            replaced = self.SBOX[byte_val]
            b |= replaced << (24 - i*8)
        return b
    
    def _L_tau(self, b):
        # τ变换，就是S盒替换
        return self._sbox_substitution(b)
    
    def _L(self, b):
        # L线性变换，用于轮函数
        return b ^ self._rotl(b, 2) ^ self._rotl(b, 10) ^ self._rotl(b, 18) ^ self._rotl(b, 24)
    
    def _L_prime(self, b):
        # L'线性变换，用于密钥扩展
        return b ^ self._rotl(b, 13) ^ self._rotl(b, 23)
    
    def _T(self, b):
        # T合成置换，先S盒再L变换
        return self._L(self._L_tau(b))
    
    def _T_prime(self, b):
        # T'合成置换，先S盒再L'变换
        return self._L_prime(self._L_tau(b))
    
    def _generate_round_keys(self, MK):
        # 密钥扩展，生成32个轮密钥
        K = []
        for i in range(4):
            K.append(MK[i] ^ self.FK[i])
        
        rk = []
        for i in range(32):
            t = K[i+1] ^ K[i+2] ^ K[i+3] ^ self.CK[i]
            K.append(K[i] ^ self._T_prime(t))
            rk.append(K[i+4])
        
        return rk
    
    def _round_function(self, X):
        # 32轮加密迭代
        for i in range(32):
            t = X[i+1] ^ X[i+2] ^ X[i+3] ^ self.rk[i]
            X.append(X[i] ^ self._T(t))
        return X
    
    def encrypt(self, plaintext):
        # 加密，输入16字节明文
        if len(plaintext) != 16:
            raise ValueError("Plaintext must be 16 bytes")
        
        # 明文分成4个32位字
        X = []
        for i in range(4):
            X.append(int.from_bytes(plaintext[i*4:(i+1)*4], 'big'))
        
        # 32轮迭代
        X = self._round_function(X)
        
        # 反序输出
        Y = [X[35], X[34], X[33], X[32]]
        
        # 组合成16字节密文
        ciphertext = b''
        for y in Y:
            ciphertext += y.to_bytes(4, 'big')
        
        return ciphertext
    
    def decrypt(self, ciphertext):
        # 解密，和加密一样，只是轮密钥顺序反过来
        if len(ciphertext) != 16:
            raise ValueError("Ciphertext must be 16 bytes")
        
        X = []
        for i in range(4):
            X.append(int.from_bytes(ciphertext[i*4:(i+1)*4], 'big'))
        
        # 轮密钥反转
        self.rk = self.rk[::-1]
        X = self._round_function(X)
        Y = [X[35], X[34], X[33], X[32]]
        
        plaintext = b''
        for y in Y:
            plaintext += y.to_bytes(4, 'big')
        
        # 恢复轮密钥顺序
        self.rk = self.rk[::-1]
        return plaintext

// SM4上下文，存放轮密钥
typedef struct {
    uint32_t rk[32];  // 32个轮密钥
} SM4_CTX;

// 循环左移
static uint32_t rotl(uint32_t x, int n) {
    return (x << n) | (x >> (32 - n));
}

// S盒替换，对32位数每个字节查S盒
static uint32_t sbox_sub(uint32_t a) {
    uint32_t b = 0;
    b |= (uint32_t)SBOX[(a >> 24) & 0xFF] << 24;
    b |= (uint32_t)SBOX[(a >> 16) & 0xFF] << 16;
    b |= (uint32_t)SBOX[(a >> 8) & 0xFF] << 8;
    b |= (uint32_t)SBOX[a & 0xFF];
    return b;
}

// L线性变换，用于轮函数
static uint32_t L(uint32_t b) {
    return b ^ rotl(b, 2) ^ rotl(b, 10) ^ rotl(b, 18) ^ rotl(b, 24);
}

// L'线性变换，用于密钥扩展
static uint32_t L_prime(uint32_t b) {
    return b ^ rotl(b, 13) ^ rotl(b, 23);
}

// T合成置换，先S盒再L变换
static uint32_t T(uint32_t b) {
    return L(sbox_sub(b));
}

// T'合成置换，先S盒再L'变换
static uint32_t T_prime(uint32_t b) {
    return L_prime(sbox_sub(b));
}

// 密钥扩展，生成32个轮密钥
void sm4_set_key(SM4_CTX *ctx, const uint8_t key[16]) {
    uint32_t MK[4];   // 原始密钥
    uint32_t K[36];   // 密钥扩展数组
    
    // 把16字节密钥转成4个32位数
    for (int i = 0; i < 4; i++) {
        MK[i] = ((uint32_t)key[i*4] << 24) |
                ((uint32_t)key[i*4+1] << 16) |
                ((uint32_t)key[i*4+2] << 8) |
                ((uint32_t)key[i*4+3]);
    }
    
    // K0~K3 = MK异或FK
    for (int i = 0; i < 4; i++) {
        K[i] = MK[i] ^ FK[i];
    }
    
    // 生成32个轮密钥
    for (int i = 0; i < 32; i++) {
        uint32_t t = K[i+1] ^ K[i+2] ^ K[i+3] ^ CK[i];
        K[i+4] = K[i] ^ T_prime(t);
        ctx->rk[i] = K[i+4];
    }
}

// 加密，输入16字节明文，输出16字节密文
void sm4_encrypt(SM4_CTX *ctx, const uint8_t plaintext[16], uint8_t ciphertext[16]) {
    uint32_t X[36];  // 36个32位状态字
    
    // 明文分成4个32位字
    for (int i = 0; i < 4; i++) {
        X[i] = ((uint32_t)plaintext[i*4] << 24) |
               ((uint32_t)plaintext[i*4+1] << 16) |
               ((uint32_t)plaintext[i*4+2] << 8) |
               ((uint32_t)plaintext[i*4+3]);
    }
    
    // 32轮迭代
    for (int i = 0; i < 32; i++) {
        uint32_t t = X[i+1] ^ X[i+2] ^ X[i+3] ^ ctx->rk[i];
        X[i+4] = X[i] ^ T(t);
    }
    
    // 反序输出
    uint32_t Y[4] = {X[35], X[34], X[33], X[32]};
    
    // 转成16字节密文
    for (int i = 0; i < 4; i++) {
        ciphertext[i*4] = (Y[i] >> 24) & 0xFF;
        ciphertext[i*4+1] = (Y[i] >> 16) & 0xFF;
        ciphertext[i*4+2] = (Y[i] >> 8) & 0xFF;
        ciphertext[i*4+3] = Y[i] & 0xFF;
    }
}

// 解密，和加密一样只是轮密钥反着用
void sm4_decrypt(SM4_CTX *ctx, const uint8_t ciphertext[16], uint8_t plaintext[16]) {
    uint32_t rk_backup[32];
    memcpy(rk_backup, ctx->rk, sizeof(ctx->rk));  // 备份轮密钥
    
    // 反转轮密钥顺序
    for (int i = 0; i < 32; i++) {
        ctx->rk[i] = rk_backup[31 - i];
    }
    
    // 用反转的轮密钥加密就是解密
    sm4_encrypt(ctx, ciphertext, plaintext);
    
    memcpy(ctx->rk, rk_backup, sizeof(ctx->rk));  // 恢复轮密钥
}

// 打印十六进制数据
void print_hex(const char *label, const uint8_t *data, int len) {
    printf("%s", label);
    for (int i = 0; i < len; i++) {
        printf("%02X", data[i]);
    }
    printf("\n");
}

SM4魔改原理

SM4魔改通常保持整体框架不变，修改其中的可调参数和组件。常见魔改点：

S盒替换

S盒是SM4唯一的非线性组件，替换S盒可直接改变算法性质。

# 原始S盒
sbox[256] = {0xD6, 0x90, 0xE9, ...}

# 魔改：自定义S盒
sbox[256] = {0x3E, 0x72, 0x5B, ...}

L变换参数

修改循环移位位数可改变扩散特性。

原始: L(B) = B ⊕ (B <<< 2) ⊕ (B <<< 10) ⊕ (B <<< 18) ⊕ (B <<< 24)
魔改: L(B) = B ⊕ (B <<< 3) ⊕ (B <<< 9) ⊕ (B <<< 21) ⊕ (B <<< 27)

L’变换参数

原始: L'(B) = B ⊕ (B <<< 13) ⊕ (B <<< 23)
魔改: L'(B) = B ⊕ (B <<< 11) ⊕ (B <<< 19)

轮数修改

原始: 32轮
魔改: 24轮 或 40轮

系统参数修改

修改FK常量或CK常量。

# 原始FK
FK = [0xA3B1BAC6, 0x56AA3350, 0x677D9197, 0xB27022DC]

# 魔改：自定义FK
FK = [0x01234567, 0x89ABCDEF, 0xFEDCBA98, 0x76543210]

轮函数修改

改变轮函数内部结构。

原始: X[i+4] = X[i] ⊕ T(X[i+1] ⊕ X[i+2] ⊕ X[i+3] ⊕ rk[i])

魔改: X[i+4] = X[i] ⊕ T(X[i+1] ⊕ X[i+2] ⊕ X[i+3] ⊕ rk[i] ⊕ 0x5A5A5A5A)

反序变换修改

原始: 输出 [X35, X34, X33, X32]

魔改: 输出 [X34, X33, X32, X35]   // 循环移位

字节序修改

原始: 大端序

魔改: 小端序

输入输出变换

在加密前后添加额外异或操作。

def encrypt(plaintext):
    plaintext ^= 0xAA  # 前置异或
    ciphertext = sm4_encrypt(plaintext)
    ciphertext ^= 0x55  # 后置异或
    return ciphertext

密钥扩展修改

在轮密钥生成中加入额外操作。

原始: K[i+4] = K[i] ⊕ T'(K[i+1] ⊕ K[i+2] ⊕ K[i+3] ⊕ CK[i])

魔改: K[i+4] = K[i] ⊕ T'(K[i+1] ⊕ K[i+2] ⊕ K[i+3] ⊕ CK[i])
rk[i] = K[i+4] ⊕ CK[i]  // 轮密钥与CK再次异或

魔改识别要点

逆向分析时关注：

  • S盒内容是否与标准一致
  • L变换的移位位数（2,10,18,24）
  • L’变换的移位位数（13,23）
  • 轮数是否为32
  • FK/CK常量是否标准
  • 反序变换模式

# 循环左移
def rotl(x, n):
    return ((x << n) | (x >> (32 - n))) & 0xFFFFFFFF

# S盒替换，按字节查表
def sbox_sub(a):
    b = 0
    b |= SBOX[(a >> 24) & 0xFF] << 24
    b |= SBOX[(a >> 16) & 0xFF] << 16
    b |= SBOX[(a >> 8) & 0xFF] << 8
    b |= SBOX[a & 0xFF]
    return b

# L线性变换，用于轮函数
def L(b):
    return b ^ rotl(b, 2) ^ rotl(b, 10) ^ rotl(b, 18) ^ rotl(b, 24)

# L'线性变换，用于密钥扩展
def L_prime(b):
    return b ^ rotl(b, 13) ^ rotl(b, 23)

# T合成置换，先S盒再L变换
def T(b):
    return L(sbox_sub(b))

# T'合成置换，先S盒再L'变换
def T_prime(b):
    return L_prime(sbox_sub(b))

# 密钥扩展，生成32个轮密钥
def generate_round_keys(MK):
    K = []
    # K0~K3 = MK异或FK
    for i in range(4):
        K.append(MK[i] ^ FK[i])
    
    # 生成32个轮密钥
    rk = []
    for i in range(ROUNDS):
        t = K[i+1] ^ K[i+2] ^ K[i+3] ^ CK[i]
        K.append(K[i] ^ T_prime(t))
        rk.append(K[i+4])
    
    return rk

class SM4_MODIFIED:
    def __init__(self, key):
        # 密钥必须是16字节
        if len(key) != 16:
            raise ValueError("Key must be 16 bytes")
        
        # 16字节密钥转4个32位数
        MK = []
        for i in range(4):
            MK.append(int.from_bytes(key[i*4:(i+1)*4], 'big'))
        
        # 生成轮密钥
        self.rk = generate_round_keys(MK)
    
    def encrypt_block(self, plaintext):
        # 加密一个16字节分组
        if len(plaintext) != 16:
            raise ValueError("Plaintext must be 16 bytes")
        
        # 明文转4个32位字（小端序）
        X = []
        for i in range(4):
            X.append(int.from_bytes(plaintext[i*4:(i+1)*4], 'little'))
        
        # 32轮迭代
        for i in range(ROUNDS):
            t = X[i+1] ^ X[i+2] ^ X[i+3] ^ self.rk[i]
            X.append(X[i] ^ T(t))
        
        # 反序输出【修改点：顺序不同】
        Y = [X[ROUNDS+3], X[ROUNDS+2], X[ROUNDS+1], X[ROUNDS+0]]
        
        # 转成16字节密文（小端序）
        ciphertext = b''
        for y in Y:
            ciphertext += y.to_bytes(4, 'little')
        
        return ciphertext
    
    def decrypt_block(self, ciphertext):
        # 解密一个16字节分组
        if len(ciphertext) != 16:
            raise ValueError("Ciphertext must be 16 bytes")
        
        # 密文转4个32位字（小端序）
        X = []
        for i in range(4):
            X.append(int.from_bytes(ciphertext[i*4:(i+1)*4], 'little'))
        
        # 轮密钥反转，解密就是反着用轮密钥
        self.rk = self.rk[::-1]
        
        # 32轮迭代
        for i in range(ROUNDS):
            t = X[i+1] ^ X[i+2] ^ X[i+3] ^ self.rk[i]
            X.append(X[i] ^ T(t))
        
        # 反序输出
        Y = [X[ROUNDS+3], X[ROUNDS+2], X[ROUNDS+1], X[ROUNDS+0]]
        
        # 转成明文
        plaintext = b''
        for y in Y:
            plaintext += y.to_bytes(4, 'little')
        
        # 恢复轮密钥顺序
        self.rk = self.rk[::-1]
        
        return plaintext
    
    def pad(self, data):
        # PKCS7填充，补到16字节的倍数
        pad_len = 16 - len(data) % 16
        return data + bytes([pad_len] * pad_len)
    
    def unpad(self, data):
        # 去掉PKCS7填充
        pad_len = data[-1]
        return data[:-pad_len]
    
    def encrypt(self, plaintext):
        # 加密任意长度数据（ECB模式）
        padded = self.pad(plaintext)  # 先填充
        ciphertext = b''
        # 逐块加密
        for i in range(0, len(padded), 16):
            block = padded[i:i+16]
            ciphertext += self.encrypt_block(block)
        return ciphertext
    
    def decrypt(self, ciphertext):
        # 解密任意长度数据（ECB模式）
        if len(ciphertext) % 16 != 0:
            raise ValueError("Ciphertext length must be multiple of 16")
        
        plaintext = b''
        # 逐块解密
        for i in range(0, len(ciphertext), 16):
            block = ciphertext[i:i+16]
            plaintext += self.decrypt_block(block)
        
        # 去掉填充
        return self.unpad(plaintext)

static uint32_t sbox_sub(uint32_t a) {

    b |= (uint32_t)SBOX[(a >> 24) & 0xFF] << 24;
    b |= (uint32_t)SBOX[(a >> 16) & 0xFF] << 16;
    b |= (uint32_t)SBOX[(a >> 8) & 0xFF] << 8;

// 密钥扩展，生成32个轮密钥
void sm4_set_key(SM4_CTX *ctx, const uint8_t key[16]) {
    uint32_t MK[4];   // 原始密钥
    uint32_t K[36];   // 密钥扩展数组
    
    // 16字节密钥转4个32位数（大端序）
    for (int i = 0; i < 4; i++) {
        MK[i] = ((uint32_t)key[i*4] << 24) |
                ((uint32_t)key[i*4+1] << 16) |
                ((uint32_t)key[i*4+2] << 8) |
                ((uint32_t)key[i*4+3]);
    }
    
    // K0~K3 = MK异或FK
    for (int i = 0; i < 4; i++) {
        K[i] = MK[i] ^ FK[i];
    }
    
    // 生成32个轮密钥
    for (int i = 0; i < ROUNDS; i++) {
        uint32_t t = K[i+1] ^ K[i+2] ^ K[i+3] ^ CK[i];
        K[i+4] = K[i] ^ T_prime(t);
        ctx->rk[i] = K[i+4];
    }
}

// 加密一个16字节分组
void sm4_encrypt_block(SM4_CTX *ctx, const uint8_t plaintext[16], uint8_t ciphertext[16]) {
    uint32_t X[36];  // 36个32位状态字
    
    // 明文转4个32位字（小端序）
    for (int i = 0; i < 4; i++) {
        X[i] = ((uint32_t)plaintext[i*4+3] << 24) |
               ((uint32_t)plaintext[i*4+2] << 16) |
               ((uint32_t)plaintext[i*4+1] << 8) |
               ((uint32_t)plaintext[i*4]);
    }
    
    // 32轮迭代
    for (int i = 0; i < ROUNDS; i++) {
        uint32_t t = X[i+1] ^ X[i+2] ^ X[i+3] ^ ctx->rk[i];
        X[i+4] = X[i] ^ T(t);
    }
    
    // 反序输出
    uint32_t Y[4] = {X[35], X[34], X[33], X[32]};
    
    // 转成16字节密文（小端序）【修改点】
    for (int i = 0; i < 4; i++) {
        ciphertext[i*4+3] = (Y[i] >> 24) & 0xFF;
        ciphertext[i*4+2] = (Y[i] >> 16) & 0xFF;
        ciphertext[i*4+1] = (Y[i] >> 8) & 0xFF;
        ciphertext[i*4] = Y[i] & 0xFF;
    }
}

// 解密一个16字节分组，就是轮密钥反过来用
void sm4_decrypt_block(SM4_CTX *ctx, const uint8_t ciphertext[16], uint8_t plaintext[16]) {
    uint32_t rk_backup[ROUNDS];
    memcpy(rk_backup, ctx->rk, sizeof(ctx->rk));  // 备份轮密钥
    
    // 反转轮密钥顺序
    for (int i = 0; i < ROUNDS; i++) {
        ctx->rk[i] = rk_backup[ROUNDS - 1 - i];
    }
    
    // 用反转的轮密钥加密就是解密
    sm4_encrypt_block(ctx, ciphertext, plaintext);
    
    memcpy(ctx->rk, rk_backup, sizeof(ctx->rk));  // 恢复轮密钥
}

// PKCS7填充，补到16字节倍数
void pkcs7_pad(const uint8_t *data, int data_len, uint8_t *padded, int *padded_len) {
    int pad_len = 16 - data_len % 16;
    memcpy(padded, data, data_len);
    // 填充值等于填充长度
    for (int i = 0; i < pad_len; i++) {
        padded[data_len + i] = pad_len;
    }
    *padded_len = data_len + pad_len;
}

// 去掉PKCS7填充
int pkcs7_unpad(uint8_t *data, int data_len) {
    int pad_len = data[data_len - 1];  // 最后一个字节就是填充长度
    return data_len - pad_len;
}

// 加密任意长度数据（ECB模式）
void sm4_encrypt(SM4_CTX *ctx, const uint8_t *plaintext, int plaintext_len, 
                 uint8_t *ciphertext, int *ciphertext_len) {
    uint8_t padded[1024];
    int padded_len;
    
    // 先填充
    pkcs7_pad(plaintext, plaintext_len, padded, &padded_len);
    
    // 逐块加密
    for (int i = 0; i < padded_len; i += 16) {
        sm4_encrypt_block(ctx, padded + i, ciphertext + i);
    }
    
    *ciphertext_len = padded_len;
}

// 解密任意长度数据（ECB模式）
int sm4_decrypt(SM4_CTX *ctx, const uint8_t *ciphertext, int ciphertext_len, 
                uint8_t *plaintext) {
    // 逐块解密
    for (int i = 0; i < ciphertext_len; i += 16) {
        sm4_decrypt_block(ctx, ciphertext + i, plaintext + i);
    }
    
    // 去掉填充
    return pkcs7_unpad(plaintext, ciphertext_len);
}

// 打印十六进制
void print_hex(const char *label, const uint8_t *data, int len) {
    printf("%s", label);
    for (int i = 0; i < len; i++) {
        printf("%02X", data[i]);
    }
    printf("\n");
}
